<description>
Bicycle engineering is a sophisticated multidisciplinary field integrating principles of mechanical engineering, materials science, dynamics, and biomechanics. At its core is the frame, a truss structure, typically a diamond frame, designed to maximise stiffness-to-weight ratio by arranging tubes to handle tensile and compressive forces efficiently. The geometry—defined by parameters like head tube angle, seat tube angle, and chainstay length—is critical, dictating the bicycle's handling, stability, and rider ergonomics. For instance, the 'trail' dimension, a function of the head tube angle and fork offset, generates a self-aligning torque (caster effect) that is fundamental to the vehicle's stability. The drivetrain is an exercise in mechanical advantage, using a system of gears (chainrings and cogs) to modulate the ratio between pedal cadence and wheel revolution, optimising for power transfer across varying terrains. Materials science plays a pivotal role, with choices ranging from high-tensile steel and lightweight aluminium alloys to advanced carbon fibre composites, each offering a unique balance of strength, weight, vibration damping, and fatigue life. The entire system is designed to function as an efficient human-machine interface, converting biomechanical energy into kinetic motion with minimal loss.

    <script>
        // --- p5.js Sketch ---
        let scene = 0;
        let sceneProgress = 0;
        let animationStartTime = 0;
        const sceneDurations = [3000, 7000, 7000, 7000, 7000, 5000]; // Durations for intro, frame, drivetrain, wheels, stability, outro
        let bike;

        // --- Narration and Synthesis ---
        const synth = window.speechSynthesis;
        let voices;
        let narrationQueue = [];
        let currentNarrationIndex = 0;
        let isNarrating = false;
        const podcastButton = document.getElementById('podcast');

        const narrationScript = [
            { text: "Welcome to our exploration of Bicycle Engineering, where we dissect the synergy of physics and design.", voice: 0, highlight: 'none' },
            { text: "Let's begin with the heart of the bicycle: the frame. It's a marvel of structural engineering, designed as a truss for maximum rigidity.", voice: 1, highlight: 'frame' },
            { text: "Precisely. The triangular geometry distributes forces efficiently. Key angles, like the head tube angle, dictate steering responsiveness and stability.", voice: 0, highlight: 'angles' },
            { text: "Now, for the engine: the drivetrain. This system converts the rider's effort into forward motion through mechanical advantage.", voice: 1, highlight: 'drivetrain' },
            { text: "It's all about gear ratios. A larger front chainring and a smaller rear cog give you speed, while the opposite provides the torque needed to climb steep hills.", voice: 0, highlight: 'gears' },
            { text: "The wheels are not just for rolling; they are complex dynamic systems. Spokes handle tension to keep the rim true, while bearings minimise friction.", voice: 1, highlight: 'wheels' },
            { text: "And brakes, whether disc or rim, use friction to convert the bicycle's kinetic energy into heat, providing controlled deceleration.", voice: 0, highlight: 'brakes' },
            { text: "But what makes a bicycle so stable at speed? The secret lies in the front-end geometry.", voice: 1, highlight: 'fork' },
            { text: "This is the 'caster effect', created by a parameter called 'trail'. This dimension provides a self-aligning torque that naturally straightens the front wheel.", voice: 0, highlight: 'trail' },
            { text: "All these components integrate to form an incredibly efficient human-powered machine, a perfect synergy of physics, materials science, and biomechanics.", voice: 1, highlight: 'all' },
        ];
        
        function loadVoices() {
            voices = synth.getVoices().filter(v => v.lang.startsWith('en'));
        }

        function playNarration() {
            if (!isNarrating || currentNarrationIndex >= narrationQueue.length) {
                stopNarration();
                return;
            }
            const utterance = narrationQueue[currentNarrationIndex];
            utterance.onend = () => {
                currentNarrationIndex++;
                playNarration();
            };
            synth.speak(utterance);
        }

        function startNarration() {
            if (synth.speaking) {
                stopNarration();
            }
            resetAnimation();
            isNarrating = true;
            podcastButton.innerHTML = '⏸️';
            
            narrationQueue = narrationScript.map(item => {
                const utterance = new SpeechSynthesisUtterance(item.text);
                if (voices && voices.length > 0) {
                    if(item.voice === 0){ // Male-like
                        utterance.voice = voices.find(v => v.name.includes('David') || v.name.includes('Google US English')) || voices[0];
                        utterance.pitch = 0.8;
                        utterance.rate = 0.95;
                    } else { // Female-like
                        utterance.voice = voices.find(v => v.name.includes('Zira') || v.name.includes('Google UK English Female')) || voices[1] || voices[0];
                        utterance.pitch = 1.2;
                        utterance.rate = 0.95;
                    }
                }
                return utterance;
            });
            
            currentNarrationIndex = 0;
            playNarration();
        }

        function stopNarration() {
            synth.cancel();
            isNarrating = false;
            podcastButton.innerHTML = '🔊';
        }

        function resetAnimation() {
            scene = 0;
            sceneProgress = 0;
            animationStartTime = millis();
            if (bike) bike.reset();
        }

        // --- Bicycle Component Classes ---
        class Bicycle {
            constructor(x, y, scale) {
                this.x = x;
                this.y = y;
                this.s = scale;
                this.highlight = 'none';

                // Geometric points relative to a central point
                this.p = {
                    bb: { x: 0, y: 0 }, // Bottom Bracket
                    seatTop: { x: -70, y: -130 },
                    headTop: { x: 130, y: -110 },
                    headBottom: { x: 120, y: -80 },
                    rearAxle: { x: -160, y: 30 },
                    frontAxle: { x: 180, y: 30 }
                };
                
                // Wheel
                this.wheelRadius = 100;
                
                // Drivetrain
                this.chainringRadius = 30;
                this.cogRadius = 12;
                this.pedalAngle = 0;

                this.reset();
            }

            reset() {
                this.frameAlpha = 0;
                this.drivetrainAlpha = 0;
                this.wheelsAlpha = 0;
                this.forkAlpha = 0;
                this.brakesAlpha = 0;
                this.trailAlpha = 0;
                this.highlight = 'none';
            }

            update(progress, currentScene) {
                if (isNarrating) {
                   this.highlight = narrationScript[currentNarrationIndex]?.highlight || 'none';
                } else {
                   this.highlight = 'none';
                }

                this.pedalAngle += 0.03;

                switch (currentScene) {
                    case 1: // Frame
                        this.frameAlpha = progress;
                        if (this.highlight === 'angles') this.frameAlpha = 1;
                        break;
                    case 2: // Drivetrain
                        this.frameAlpha = 1;
                        this.drivetrainAlpha = progress;
                        break;
                    case 3: // Wheels & Brakes
                        this.frameAlpha = 1;
                        this.drivetrainAlpha = 1;
                        this.wheelsAlpha = progress;
                        this.brakesAlpha = progress > 0.5 ? (progress - 0.5) * 2 : 0;
                        break;
                    case 4: // Stability & Fork
                        this.frameAlpha = 1;
                        this.drivetrainAlpha = 1;
                        this.wheelsAlpha = 1;
                        this.brakesAlpha = 1;
                        this.forkAlpha = progress;
                        this.trailAlpha = progress > 0.5 ? (progress - 0.5) * 2 : 0;
                        break;
                    case 5: // Outro
                        this.frameAlpha = 1;
                        this.drivetrainAlpha = 1;
                        this.wheelsAlpha = 1;
                        this.brakesAlpha = 1;
                        this.forkAlpha = 1;
                        this.trailAlpha = 0;
                        break;
                }
            }

            draw() {
                push();
                translate(this.x, this.y);
                scale(this.s);
                
                this.drawWheels();
                this.drawFrame();
                this.drawDrivetrain();
                this.drawForkAndBars();
                this.drawBrakes();
                this.drawTrail();

                pop();
            }

            isHighlighted(part) {
                return this.highlight === part || this.highlight === 'all';
            }
            
            getGlowColor(alpha) {
                return color(0, 150, 255, alpha * 255);
            }

            getPrimaryColor(alpha) {
                return color(240, 240, 240, alpha * 255);
            }
            
            drawPart(partName, alpha, drawFunc) {
                if (alpha > 0) {
                    if (this.isHighlighted(partName)) {
                        strokeWeight(4);
                        stroke(this.getGlowColor(alpha));
                        fill(this.getGlowColor(alpha * 0.5));
                        drawFunc();
                        strokeWeight(2); // reset for other parts
                    } else {
                        stroke(this.getPrimaryColor(alpha));
                        fill(this.getPrimaryColor(alpha * 0.3));
                        drawFunc();
                    }
                }
            }

            drawFrame() {
                this.drawPart('frame', this.frameAlpha, () => {
                    strokeWeight(8);
                    noFill();
                    // Main triangle
                    line(this.p.bb.x, this.p.bb.y, this.p.seatTop.x, this.p.seatTop.y); // Seat tube
                    line(this.p.bb.x, this.p.bb.y, this.p.headBottom.x, this.p.headBottom.y); // Down tube
                    line(this.p.seatTop.x, this.p.seatTop.y, this.p.headTop.x, this.p.headTop.y); // Top tube
                    line(this.p.headTop.x, this.p.headTop.y, this.p.headBottom.x, this.p.headBottom.y); // Head tube
                    // Rear triangle
                    strokeWeight(6);
                    line(this.p.seatTop.x, this.p.seatTop.y, this.p.rearAxle.x, this.p.rearAxle.y); // Seat stays
                    line(this.p.bb.x, this.p.bb.y, this.p.rearAxle.x, this.p.rearAxle.y); // Chain stays
                });
                 
                if (this.isHighlighted('angles') && this.frameAlpha > 0.5) {
                    push();
                    stroke(255, 100, 100, 200);
                    strokeWeight(2);
                    // Head tube angle
                    line(this.p.headTop.x, this.p.headTop.y, this.p.headBottom.x-50, this.p.headBottom.y+20);
                    line(this.p.headBottom.x-50, this.p.headBottom.y+20, this.p.headTop.x+50, this.p.headBottom.y+20);
                    noFill();
                    arc(this.p.headBottom.x, this.p.headBottom.y, 40, 40, 0.45, PI);
                    pop();
                }

                // Saddle and post
                 this.drawPart('frame', this.frameAlpha, () => {
                    strokeWeight(4);
                    line(this.p.seatTop.x, this.p.seatTop.y, this.p.seatTop.x, this.p.seatTop.y-20);
                    fill(this.getPrimaryColor(this.frameAlpha));
                    quad(this.p.seatTop.x-25, this.p.seatTop.y-20, this.p.seatTop.x+20, this.p.seatTop.y-20, this.p.seatTop.x+15, this.p.seatTop.y-28, this.p.seatTop.x-20, this.p.seatTop.y-28);
                });
            }

            drawDrivetrain() {
                this.drawPart('drivetrain', this.drivetrainAlpha, () => {
                    strokeWeight(2);
                    fill(this.getPrimaryColor(this.drivetrainAlpha));
                    // Chainring
                    circle(this.p.bb.x, this.p.bb.y, this.chainringRadius * 2);
                    // Cranks and pedals
                    let crankX1 = this.p.bb.x + 40 * cos(this.pedalAngle);
                    let crankY1 = this.p.bb.y + 40 * sin(this.pedalAngle);
                    line(this.p.bb.x, this.p.bb.y, crankX1, crankY1);
                    rect(crankX1 - 10, crankY1 - 2.5, 20, 5);
                    let crankX2 = this.p.bb.x + 40 * cos(this.pedalAngle + PI);
                    let crankY2 = this.p.bb.y + 40 * sin(this.pedalAngle + PI);
                    line(this.p.bb.x, this.p.bb.y, crankX2, crankY2);
                    rect(crankX2 - 10, crankY2 - 2.5, 20, 5);
                });

                this.drawPart('gears', this.drivetrainAlpha, () => {
                     // Cog
                    stroke(this.isHighlighted('gears') ? this.getGlowColor(this.drivetrainAlpha) : this.getPrimaryColor(this.drivetrainAlpha));
                    fill(this.isHighlighted('gears') ? this.getGlowColor(this.drivetrainAlpha*0.5) : this.getPrimaryColor(this.drivetrainAlpha*0.3));
                    circle(this.p.rearAxle.x, this.p.rearAxle.y, this.cogRadius * 2);
                    // Chain
                    noFill();
                    strokeWeight(2);
                    beginShape();
                    vertex(this.p.bb.x, this.p.bb.y - this.chainringRadius);
                    vertex(this.p.rearAxle.x, this.p.rearAxle.y - this.cogRadius);
                    endShape();
                    beginShape();
                    vertex(this.p.bb.x, this.p.bb.y + this.chainringRadius);
                    vertex(this.p.rearAxle.x, this.p.rearAxle.y + this.cogRadius);
                    endShape();
                });
            }
            
            drawWheels() {
                this.drawPart('wheels', this.wheelsAlpha, () => {
                    noFill();
                    strokeWeight(4);
                    // Rear wheel
                    circle(this.p.rearAxle.x, this.p.rearAxle.y, this.wheelRadius * 2);
                    // Front wheel
                    circle(this.p.frontAxle.x, this.p.frontAxle.y, this.wheelRadius * 2);
                    
                    strokeWeight(1);
                    for (let i = 0; i < 12; i++) {
                        let angle = TWO_PI / 12 * i;
                        line(this.p.rearAxle.x, this.p.rearAxle.y, this.p.rearAxle.x + this.wheelRadius * cos(angle), this.p.rearAxle.y + this.wheelRadius * sin(angle));
                        line(this.p.frontAxle.x, this.p.frontAxle.y, this.p.frontAxle.x + this.wheelRadius * cos(angle), this.p.frontAxle.y + this.wheelRadius * sin(angle));
                    }
                });
            }

            drawForkAndBars() {
                this.drawPart('fork', this.forkAlpha, () => {
                    strokeWeight(6);
                    line(this.p.headBottom.x, this.p.headBottom.y, this.p.frontAxle.x, this.p.frontAxle.y);
                    line(this.p.headTop.x, this.p.headTop.y, this.p.frontAxle.x, this.p.frontAxle.y);

                    strokeWeight(4);
                    // Stem and handlebars
                    let stemX = this.p.headTop.x + 30;
                    let stemY = this.p.headTop.y - 15;
                    line(this.p.headTop.x, this.p.headTop.y, stemX, stemY);
                    line(stemX - 30, stemY, stemX + 30, stemY);
                });
            }
            
            drawBrakes() {
                this.drawPart('brakes', this.brakesAlpha, () => {
                    fill(180, 50, 50, this.brakesAlpha*255);
                    noStroke();
                    // Disc brake calipers
                    circle(this.p.frontAxle.x + 10, this.p.frontAxle.y - 20, 15);
                    circle(this.p.rearAxle.x - 10, this.p.rearAxle.y - 15, 15);
                });
            }

            drawTrail() {
                 if (this.isHighlighted('trail') && this.trailAlpha > 0) {
                     push();
                     let alpha = this.trailAlpha * 255;
                     
                     // Steering axis line
                     let dx = this.p.headTop.x - this.p.headBottom.x;
                     let dy = this.p.headTop.y - this.p.headBottom.y;
                     let mag = sqrt(dx * dx + dy * dy);
                     let ux = dx/mag;
                     let uy = dy/mag;
                     
                     let groundY = this.wheelRadius;
                     let t = (groundY - this.p.headBottom.y) / uy;
                     let intersectX = this.p.headBottom.x + t * ux;

                     stroke(255, 200, 0, alpha);
                     strokeWeight(2);
                     drawingContext.setLineDash([5, 5]);
                     line(this.p.headTop.x, this.p.headTop.y, intersectX, groundY);
                     
                     // Vertical line from axle
                     stroke(0, 255, 255, alpha);
                     line(this.p.frontAxle.x, this.p.frontAxle.y, this.p.frontAxle.x, groundY);
                     
                     // Trail measurement
                     stroke(255, 255, 255, alpha);
                     strokeWeight(3);
                     drawingContext.setLineDash([]);
                     line(intersectX, groundY + 10, this.p.frontAxle.x, groundY + 10);
                     
                     fill(255,255,255, alpha);
                     noStroke();
                     textAlign(CENTER);
                     textSize(16);
                     text("Trail", (intersectX + this.p.frontAxle.x) / 2, groundY + 30);
                     
                     pop();
                 }
            }
        }

        function setup() {
            createCanvas(windowWidth, windowHeight);
            frameRate(30);
            
            if (synth.onvoiceschanged !== undefined) {
                synth.onvoiceschanged = loadVoices;
            }
            loadVoices();
            
            bike = new Bicycle(width / 2, height / 2 + 100, 1.5);
            
            document.getElementById('replay').addEventListener('click', () => {
                if(isNarrating) stopNarration();
                resetAnimation();
            });
            
            podcastButton.addEventListener('click', () => {
                if (isNarrating) {
                    stopNarration();
                } else {
                    startNarration();
                }
            });
            
            resetAnimation();
        }

        function draw() {
            background(color(getComputedStyle(document.documentElement).getPropertyValue('--bg-color')));
            
            let elapsedTime = millis() - animationStartTime;
            let totalDuration = sceneDurations.reduce((a, b) => a + b, 0);

            let cumulativeDuration = 0;
            for (let i = 0; i < sceneDurations.length; i++) {
                if (elapsedTime < cumulativeDuration + sceneDurations[i]) {
                    scene = i;
                    sceneProgress = (elapsedTime - cumulativeDuration) / sceneDurations[i];
                    break;
                }
                cumulativeDuration += sceneDurations[i];
            }
            if (elapsedTime >= totalDuration) {
                scene = sceneDurations.length - 1;
                sceneProgress = 1;
            }

            // Title management
            const titleEl = document.getElementById('title');
            let currentTitle = "";
            switch(scene) {
                case 0: currentTitle = "Bicycle Engineering"; break;
                case 1: currentTitle = "1. Frame Geometry & Structure"; break;
                case 2: currentTitle = "2. Drivetrain & Mechanical Advantage"; break;
                case 3: currentTitle = "3. Wheels & Braking Dynamics"; break;
                case 4: currentTitle = "4. Steering Geometry & Stability"; break;
                case 5: currentTitle = "An Integrated System"; break;
            }
            if (titleEl.innerText !== currentTitle) {
                titleEl.style.opacity = 0;
                setTimeout(() => {
                    titleEl.innerText = currentTitle;
                    titleEl.style.opacity = 1;
                }, 500);
            }

            bike.update(sceneProgress, scene);
            bike.draw();
        }

        function windowResized() {
            resizeCanvas(windowWidth, windowHeight);
            bike.x = width / 2;
            bike.y = height / 2 + 100;
        }

    </script>
