<description>
A car diagram is a scientific schematic representation of a vehicle’s structural and mechanical systems. It details core automotive components such as the engine, transmission, drivetrain, suspension, steering mechanism, braking system, fuel system, electrical system, and chassis frame. At a graduate-level understanding, the diagram captures not only the placement but the interaction between subsystems—highlighting thermodynamic cycles in the engine, torque transfer in drivetrains, fluid dynamics in hydraulic braking, and signal routing in embedded electronic control units (ECUs). Such diagrams are essential for diagnostics, prototyping, and optimization in automotive engineering. Modern representations may include vector-based layers to show different systems in operation. They assist in evaluating performance trade-offs, safety constraints, and energy efficiency, critical for both ICE (internal combustion engine) and EV (electric vehicle) platforms.
</description>

  <script src="https://cdn.jsdelivr.net/npm/p5@1.6.0/lib/p5.min.js"></script>
  <script>
    let stage = 0;
    let frameCounter = 0;
    const totalStages = 6;
    const fps = 30;
    let isNarrating = false;
    let voices = [];
    let synth = window.speechSynthesis;
    let narrationIndex = 0;

    const narration = [
      "Let’s explore the car diagram stage by stage.",
      "This is the chassis, the structural frame of the car.",
      "Here we add the internal combustion engine at the front.",
      "Now we connect the drivetrain and transmission system.",
      "Suspension and wheels are mounted for motion control.",
      "Finally, we complete with the fuel and electrical systems."
    ];

    function setup() {
      frameRate(fps);
      createCanvas(windowWidth, windowHeight);
      stroke(0, 255, 255);
      strokeWeight(1.5);
      noFill();
    }

    function draw() {
      background(10, 10, 16, 220);
      translate(width/2 - 150, height/2 - 100);
      drawBlueprintGrid();

      if (frameCounter < fps * 3) stage = 0;
      else if (frameCounter < fps * 6) stage = 1;
      else if (frameCounter < fps * 9) stage = 2;
      else if (frameCounter < fps * 12) stage = 3;
      else if (frameCounter < fps * 15) stage = 4;
      else if (frameCounter < fps * 18) stage = 5;

      drawSlide(stage);
      frameCounter++;
    }

    function drawBlueprintGrid() {
      stroke(0, 123, 255, 40);
      for (let x = 0; x < width; x += 40) {
        line(x - width/2 + 150, 0, x - width/2 + 150, height);
      }
      for (let y = 0; y < height; y += 40) {
        line(0, y - height/2 + 100, width, y - height/2 + 100);
      }
    }

    function drawSlide(s) {
      stroke(0, 255, 255);
      if (s >= 0) {
        rect(0, 0, 300, 150); // Chassis
      }
      if (s >= 1) {
        fill(0, 255, 255, 60);
        rect(50, 40, 60, 50); // Engine
        noFill();
      }
      if (s >= 2) {
        line(110, 65, 200, 65); // Transmission line
        ellipse(200, 65, 40, 40); // Gearbox
      }
      if (s >= 3) {
        ellipse(60, 160, 40); // Wheel 1
        ellipse(240, 160, 40); // Wheel 2
        rect(30, 130, 240, 30); // Suspension bar
      }
      if (s >= 4) {
        beginShape(); // Fuel tank
        vertex(250, 30);
        vertex(280, 30);
        vertex(280, 60);
        vertex(250, 60);
        endShape(CLOSE);
        line(250, 45, 200, 65); // Fuel line
      }
      if (s >= 5) {
        for (let i = 0; i < 5; i++) {
          point(150 + i * 10, 40 + sin(frameCount * 0.2 + i) * 10); // Electric signal
        }
      }
    }

    document.getElementById('replay').onclick = () => {
      frameCounter = 0;
    };

    document.getElementById('podcast').onclick = () => {
      if (isNarrating) {
        synth.cancel();
        isNarrating = false;
      } else {
        frameCounter = 0;
        narrationIndex = 0;
        isNarrating = true;
        playNarration();
      }
    };

    function playNarration() {
      if (!isNarrating || narrationIndex >= narration.length) {
        isNarrating = false;
        return;
      }

      const utter = new SpeechSynthesisUtterance(narration[narrationIndex]);
      const vList = synth.getVoices().filter(v => v.lang.startsWith('en'));
      utter.voice = vList[narrationIndex % vList.length];
      utter.pitch = 1 + (narrationIndex % 2) * 0.3;
      utter.rate = 1.0;
      synth.speak(utter);

      utter.onend = () => {
        narrationIndex++;
        playNarration();
      };
    }

    window.speechSynthesis.onvoiceschanged = () => {
      voices = synth.getVoices();
    };
    
    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }
  </script>
</body>
