<description>
The human heart is a remarkable, four-chambered muscular organ designed to efficiently pump blood throughout the body, ensuring oxygen and nutrient delivery while simultaneously removing metabolic wastes. This vital function is orchestrated through a precise, cyclical process involving two interconnected circulatory systems: the pulmonary and systemic circuits.

Deoxygenated blood, returning from the body via the superior and inferior vena cava, first enters the heart's upper right chamber, the right atrium (RA). During atrial systole, it's propelled through the tricuspid valve into the lower right chamber, the right ventricle (RV). Upon ventricular contraction, the RV pumps this blood through the pulmonary valve into the pulmonary artery, which branches to carry it to the lungs. In the pulmonary capillaries of the lungs, carbon dioxide is offloaded, and oxygen is absorbed, transforming the blood from deoxygenated to oxygenated.

This now oxygen-rich blood returns to the heart's upper left chamber, the left atrium (LA), via the pulmonary veins. From the LA, it passes through the mitral (bicuspid) valve into the highly muscular left ventricle (LV). The LV, the strongest chamber, then contracts forcefully, ejecting the oxygenated blood through the aortic valve into the aorta. The aorta, the body's largest artery, branches extensively to distribute oxygenated blood to all tissues and organs in the systemic circulation. As tissues utilize oxygen, the blood becomes deoxygenated, completing the systemic loop and returning to the right atrium, restarting the entire cardiac cycle. This continuous, rhythmic pumping action, driven by the heart's intrinsic electrical system and regulated by a complex network of valves, ensures consistent blood flow essential for life.
</description>

        // Helper to draw a vessel path
        function drawVessel(pathDef) {
            let pointsToDraw = getPathPointsForDrawing(pathDef);
            noFill();
            stroke(glassBorder);
            strokeWeight(2);
            beginShape();
            for (let p of pointsToDraw) {

        // Helper for Bezier points calculation
        function getBezierPoint(p0, p1, p2, p3, t) {
            let x = pow(1 - t, 3) * p0.x + 3 * pow(1 - t, 2) * t * p1.x + 3 * (1 - t) * pow(t, 2) * p2.x + pow(t, 3) * p3.x;
            let y = pow(1 - t, 3) * p0.y + 3 * pow(1 - t, 2) * t * p1.y + 3 * (1 - t) * pow(t, 2) * p2.y + pow(t, 3) * p3.y;
            return createVector(x, y);
        }

        // Get points along a path for drawing (more dense for smooth curves)
        function getPathPointsForDrawing(pathDef) {
            let points = [];
            if (pathDef.type === 'linear' && pathDef.points.length === 2) {
                points.push(pathDef.points[0]);
                points.push(pathDef.points[1]);
            } else if (pathDef.type === 'bezier' && pathDef.points.length === 4) {
                for (let i = 0; i <= 20; i++) { // 20 segments for drawing
                    points.push(getBezierPoint(pathDef.points[0], pathDef.points[1], pathDef.points[2], pathDef.points[3], i / 20));
                }
            } else if (pathDef.points.length > 2) { // Handle paths with more than 2 linear points (e.g., path.points = [a,b,c])
                for(let i=0; i < pathDef.points.length; i++) {
                    points.push(pathDef.points[i]);
                }
            }
            return points;
        }

        // Blood Particle Class
        class BloodParticle {
            constructor(id) {
                this.id = id;
                this.reset();
            }

            reset() {
                this.currentPathIndex = 0;
                // Start particles at varied points along the first segment
                this.segmentProgress = random(0, 1);
                this.color = color(50, 100, 255, 200); // Initial deoxygenated blue
            }

            update() {
                const currentPathDef = PATH_DEFINITIONS[this.currentPathIndex];
                const speed = PARTICLE_SPEED_FACTOR / (currentPathDef.durationMultiplier || 1);
                this.segmentProgress += speed * (1 + sin(frameCount * 0.05 + this.id) * 0.5); // Vary speed slightly

                if (this.segmentProgress >= 1) {
                    this.segmentProgress = 0;
                    this.currentPathIndex = (this.currentPathIndex + 1) % PATH_DEFINITIONS.length;
                    const nextPathDef = PATH_DEFINITIONS[this.currentPathIndex];
                    if (nextPathDef.colorChange === 0) { // Becomes deoxygenated
                        this.color = color(50, 100, 255, 200);
                    } else if (nextPathDef.colorChange === 1) { // Becomes oxygenated
                        this.color = color(255, 50, 50, 200);
                    }
                }

                // Calculate position along current segment
                const points = currentPathDef.points;
                let pos;
                if (currentPathDef.type === 'linear' && points.length === 2) {
                    pos = p5.Vector.lerp(points[0], points[1], this.segmentProgress);
                } else if (currentPathDef.type === 'bezier' && points.length === 4) {
                    pos = getBezierPoint(points[0], points[1], points[2], points[3], this.segmentProgress);
                } else { // Fallback, just for robustness
                    pos = points[0];
                }
                this.x = pos.x;
                this.y = pos.y;
            }

            display() {
                noStroke();
                fill(this.color);
                ellipse(this.x, this.y, heartScale * 0.02); // Particle size relative to heart scale
            }
        }

        // Update and draw all blood particles
        function drawBloodFlow() {
            for (let particle of bloodParticles) {
                particle.update();
                particle.display();
            }
        }

        // Speech Synthesis and Narration Logic
        function loadVoices() {
            // Attempt to load voices immediately
            voices = speechSynthesis.getVoices();
            assignVoices();

            // If voices aren't immediately available, wait for the 'voiceschanged' event
            if (voices.length === 0) {
                speechSynthesis.onvoiceschanged = () => {
                    voices = speechSynthesis.getVoices();
                    assignVoices();
                };
            }
        }

        function assignVoices() {
            // Prefer Google US English voices if available
            maleVoice = voices.find(voice => voice.name.includes('Google US English') && voice.name.includes('Male')) ||
                        voices.find(voice => voice.lang === 'en-US' && !voice.name.includes('Female'));
            femaleVoice = voices.find(voice => voice.name.includes('Google US English') && voice.name.includes('Female')) ||
                          voices.find(voice => voice.lang === 'en-US' && voice.name.includes('Female'));

            // Fallback if specific voices not found
            if (!maleVoice && voices.length > 0) maleVoice = voices[0];
            if (!femaleVoice && voices.length > 1) femaleVoice = voices[1];
            else if (!femaleVoice && voices.length > 0) femaleVoice = voices[0]; // If only one voice, use it for both
        }

        function startPodcast() {
            if (speaking) return; // Prevent multiple simultaneous narrations
            speaking = true;
            narrationIndex = 0;
            speakNextPhrase();
        }

        function speakNextPhrase() {
            if (narrationIndex < narrationData.length && speaking) {
                const currentPhrase = narrationData[narrationIndex];
                highlightedElement = currentPhrase.visualCue;

                const utterance = new SpeechSynthesisUtterance(currentPhrase.text);
                utterance.pitch = currentPhrase.pitch;
                utterance.rate = currentPhrase.rate;

                if (currentPhrase.voiceType === 'male' && maleVoice) {
                    utterance.voice = maleVoice;
                } else if (currentPhrase.voiceType === 'female' && femaleVoice) {
                    utterance.voice = femaleVoice;
                } else {
                    utterance.voice = voices[0]; // Fallback to first available voice
                }

                utterance.onend = () => {
                    narrationIndex++;
                    // Add a small delay between phrases for conversational feel
                    setTimeout(speakNextPhrase, 500);
                };

                utterance.onerror = (event) => {
                    console.error('Speech synthesis error:', event.error);
                    speaking = false;
                    highlightedElement = null;
                };

                speechSynthesis.speak(utterance);
            } else {
                speaking = false;
                highlightedElement = null; // Clear highlight at end of narration
            }
        }

                endShape(CLOSE);
            } else if (highlightedElement === 'lungs' && heartElements.lungs_rect) {
                rect(heartElements.lungs_rect.x, heartElements.lungs_rect.y, heartElements.lungs_rect.w, heartElements.lungs_rect.h, 15);
            } else if (highlightedElement === 'body' && heartElements.body_rect) {
                rect(heartElements.body_rect.x, heartElements.body_rect.y, heartElements.body_rect.w, heartElements.body_rect.h, 15);
            } else if (['pulmonaryArtery', 'pulmonaryVeins', 'aorta'].includes(highlightedElement)) {
                // Find the correct path definition based on highlight name
                let pathDefToHighlight = PATH_DEFINITIONS.find(p => p.highlight === highlightedElement);
                if (pathDefToHighlight) {
                    let pointsToDraw = getPathPointsForDrawing(pathDefToHighlight);
                    beginShape();
                    noFill();
                    for (let p of pointsToDraw) {
                        curveVertex(p.x, p.y);
                    }
                    endShape();
                }
            } else if (highlightedElement === 'heart') {
                // Highlight the entire heart structure (a bounding box around chambers)
                const allHeartPoints = [
                    ...heartElements.RA_shape, ...heartElements.RV_shape,
                    ...heartElements.LA_shape, ...heartElements.LV_shape
                ];
                let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
                for(let p of allHeartPoints) {
                    minX = min(minX, p.x);
                    minY = min(minY, p.y);
                    maxX = max(maxX, p.x);
                    maxY = max(maxY, p.y);
                }
                const padding = heartScale * 0.05; // Small padding for the highlight box
                rect(minX - padding, minY - padding, (maxX - minX) + 2 * padding, (maxY - minY) + 2 * padding, 20);
            }
            pop();
        }
    </script>
