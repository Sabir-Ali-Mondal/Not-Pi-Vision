<description>
The Earth's core is the innermost layer of our planet, composed primarily of an iron-nickel alloy, alongside trace amounts of lighter elements like sulfur, oxygen, and silicon. It is divided into two distinct parts: the outer core and the inner core.

The **Outer Core**, extending from approximately 2,890 km to 5,150 km below the Earth's surface, is a liquid layer. Its immense heat, estimated to be between 4,400°C and 6,100°C, keeps the iron-nickel molten. This liquid metal undergoes vigorous convection currents, driven by the cooling of the core and the rotation of the Earth. As the conductive molten iron moves, it generates electrical currents, which in turn produce Earth's geodynamo – the mechanism responsible for creating our planet's protective magnetic field. This field is crucial for deflecting harmful solar radiation and maintaining our atmosphere.

Beneath the outer core lies the **Inner Core**, a solid sphere with a radius of about 1,220 km, reaching temperatures up to 6,200°C. Despite being hotter than the outer core, the inner core remains solid due to the extreme pressures (estimated at 3.6 million atmospheres) exerted by the overlying layers, which prevent the iron atoms from moving freely. Seismic studies indicate that the inner core rotates slightly faster than the rest of the planet. The boundary between the liquid outer core and the solid inner core is known as the Lehmann Discontinuity. The Earth's core is a dynamic and essential component of our planet, influencing geological processes, surface conditions, and habitability.
</description>

    <script>
        let coreSketch;

        const colors = {
            bgColor: '#0a0a10',
            primaryColor: '#007bff',
            glowColor: 'rgba(0, 123, 255, 0.7)',
            textColor: '#f0f0f0',
            glassBg: 'rgba(25, 25, 40, 0.3)',
            glassBorder: 'rgba(255, 255, 255, 0.15)'
        };

        const sketch = (p) => {
            // Animation States
            const STATES = {
                INITIAL: 0,
                ZOOM_INTO_CORE: 1,
                OUTER_CORE_EMPHASIS: 2,
                INNER_CORE_EMPHASIS: 3,
                LEHMANN_DISCONTINUITY: 4,
                MAGNETIC_FIELD: 5,
                INNER_CORE_ROTATION: 6,
                OVERVIEW: 7,
                COMPLETE: 8
            };
            let currentState = STATES.INITIAL;
            let animationStartTime = 0;
            let currentNarrationIndex = 0;

            // Core parameters
            const earthRadius = 400; // Conceptual outer Earth/Mantle for initial view
            const outerCoreRadius = 250; 
            const innerCoreRadius = 120; 

            // Camera parameters
            let camX, camY, camZ;
            let targetCamX, targetCamY, targetCamZ;
            let camRotX, camRotY, camRotZ;
            let targetCamRotX, targetCamRotY, targetCamRotZ;

            // Material properties (alpha for visibility and fade effects)
            let mantleAlpha = 255;
            let outerCoreAlpha = 0; 
            let innerCoreAlpha = 0;
            let magneticFieldAlpha = 0;

            // Convection particles for outer core
            let convectionParticles = [];
            const numParticles = 400; 
            const particleSize = 3;

            // Magnetic field lines
            let magneticFieldLines = [];
            const numFieldLines = 15;

            // Speech Synthesis Variables
            let voices = [];
            let maleVoice;
            let femaleVoice;
            let currentUtterance = null;
            let narrationActive = false;

            const narrationScript = [
                {
                    text: "Welcome to the Earth's core, a truly extreme environment hidden deep beneath our feet!",
                    voice: 'male', pitch: 1.0, rate: 0.9,
                    syncAction: () => {
                        currentState = STATES.INITIAL;
                        animationStartTime = p.millis();
                        targetCamX = 0; targetCamY = 0; targetCamZ = earthRadius * 1.5;
                        targetCamRotX = 0; targetCamRotY = 0; targetCamRotZ = 0;
                        mantleAlpha = 255; // Show initial Earth/Mantle
                        outerCoreAlpha = 0;
                        innerCoreAlpha = 0;
                        magneticFieldAlpha = 0;
                    }
                },
                {
                    text: "Indeed! It's not just a solid ball, but a complex, dynamic system crucial for life on Earth.",
                    voice: 'female', pitch: 1.1, rate: 1.0,
                    syncAction: () => {
                        currentState = STATES.ZOOM_INTO_CORE;
                        animationStartTime = p.millis();
                        targetCamZ = outerCoreRadius * 2.5; // Zoom closer
                        targetCamRotX = p.PI / 10;
                        mantleAlpha = 100; // Make mantle transparent to reveal core
                    }
                },
                {
                    text: "Let's first look at the Outer Core. It's a vast, superheated layer, mostly liquid iron and nickel.",
                    voice: 'male', pitch: 1.0, rate: 0.9,
                    syncAction: () => {
                        currentState = STATES.OUTER_CORE_EMPHASIS;
                        animationStartTime = p.millis();
                        targetCamZ = outerCoreRadius * 2.2; 
                        targetCamRotX = p.PI / 8;
                        mantleAlpha = 0; // Hide mantle completely
                        outerCoreAlpha = 200; // Reveal outer core
                        convectionParticles = []; // Initialize convection particles
                        for (let i = 0; i < numParticles; i++) {
                            convectionParticles.push({
                                pos: p.createVector(p.random(-outerCoreRadius + 20, outerCoreRadius - 20), p.random(-outerCoreRadius + 20, outerCoreRadius - 20), p.random(-outerCoreRadius + 20, outerCoreRadius - 20)),
                                noiseOffset: p.createVector(p.random(1000), p.random(1000), p.random(1000))
                            });
                        }
                    }
                },
                {
                    text: "Imagine molten metal churning like a furious ocean! These powerful convection currents are what make our planet so special.",
                    voice: 'female', pitch: 1.1, rate: 1.0,
                    syncAction: () => {
                        currentState = STATES.OUTER_CORE_EMPHASIS; 
                    }
                },
                {
                    text: "That's right. Because of these moving conductive metals, the outer core acts like a gigantic dynamo, generating Earth's protective magnetic field.",
                    voice: 'male', pitch: 1.0, rate: 0.9,
                    syncAction: () => {
                        currentState = STATES.MAGNETIC_FIELD;
                        animationStartTime = p.millis();
                        magneticFieldAlpha = 200; // Reveal magnetic field lines
                        magneticFieldLines = []; // Generate new field lines
                        for (let i = 0; i < numFieldLines; i++) {
                            magneticFieldLines.push({
                                start: p.createVector(p.random(-outerCoreRadius * 0.7, outerCoreRadius * 0.7), p.random(-outerCoreRadius * 0.7, outerCoreRadius * 0.7), p.random(-outerCoreRadius * 0.7, outerCoreRadius * 0.7)),
                                end: p.createVector(p.random(-outerCoreRadius * 1.5, outerCoreRadius * 1.5), p.random(-outerCoreRadius * 1.5, outerCoreRadius * 1.5), p.random(-outerCoreRadius * 1.5, outerCoreRadius * 1.5))
                            });
                        }
                    }
                },
                {
                    text: "This field deflects harmful solar radiation, making our atmosphere and surface habitable. Truly vital!",
                    voice: 'female', pitch: 1.1, rate: 1.0,
                    syncAction: () => {
                        currentState = STATES.MAGNETIC_FIELD; 
                    }
                },
                {
                    text: "Now, let's dive even deeper, right into the center, to the Inner Core.",
                    voice: 'male', pitch: 1.0, rate: 0.9,
                    syncAction: () => {
                        currentState = STATES.INNER_CORE_EMPHASIS;
                        animationStartTime = p.millis();
                        targetCamZ = innerCoreRadius * 3; // Zoom into inner core
                        targetCamRotX = p.PI / 6;
                        outerCoreAlpha = 100; // Make outer core more transparent
                        innerCoreAlpha = 255; // Reveal inner core
                        magneticFieldAlpha = 50; // Dim magnetic field lines slightly
                    }
                },
                {
                    text: "Despite temperatures hotter than the surface of the sun, the immense pressure here solidifies the iron and nickel into a rigid sphere.",
                    voice: 'female', pitch: 1.1, rate: 1.0,
                    syncAction: () => {
                        currentState = STATES.INNER_CORE_EMPHASIS;
                    }
                },
                {
                    text: "This boundary between the liquid outer and solid inner core is known as the Lehmann Discontinuity.",
                    voice: 'male', pitch: 1.0, rate: 0.9,
                    syncAction: () => {
                        currentState = STATES.LEHMANN_DISCONTINUITY;
                        animationStartTime = p.millis();
                    }
                },
                {
                    text: "And get this, the inner core actually rotates independently from the rest of the planet, slightly faster than the surface!",
                    voice: 'female', pitch: 1.1, rate: 1.0,
                    syncAction: () => {
                        currentState = STATES.INNER_CORE_ROTATION;
                        animationStartTime = p.millis();
                    }
                },
                {
                    text: "It's a remarkable testament to the Earth's internal forces, shaping everything from geology to life itself.",
                    voice: 'male', pitch: 1.0, rate: 0.9,
                    syncAction: () => {
                        currentState = STATES.OVERVIEW;
                        animationStartTime = p.millis();
                        targetCamZ = outerCoreRadius * 3.0; // Pull back for overview
                        targetCamRotX = p.PI / 10;
                        outerCoreAlpha = 180;
                        innerCoreAlpha = 255;
                        magneticFieldAlpha = 150;
                    }
                },
                {
                    text: "A hidden powerhouse, constantly working to keep our planet alive and thriving!",
                    voice: 'female', pitch: 1.1, rate: 1.0,
                    syncAction: () => {
                        currentState = STATES.OVERVIEW;
                    }
                }
            ];

            p.setup = () => {
                p.createCanvas(p.windowWidth, p.windowHeight, p.WEBGL);
                p.setAttributes({ antialias: true }); 
                p.setFrameRate(30);

                camX = targetCamX = 0;
                camY = targetCamY = 0;
                camZ = targetCamZ = earthRadius * 1.5;
                camRotX = targetCamRotX = 0;
                camRotY = targetCamRotY = 0;
                camRotZ = targetCamRotZ = 0;

                p.pixelDensity(p.displayDensity()); 

                if ('speechSynthesis' in window) {
                    window.speechSynthesis.onvoiceschanged = () => {
                        voices = window.speechSynthesis.getVoices();
                        // Prioritize Google US English voices, then generic en-US, then any en voice
                        maleVoice = voices.find(v => v.lang === 'en-US' && v.name.includes('Google US English') && v.name.toLowerCase().includes('male')) ||
                                    voices.find(v => v.lang === 'en-US' && v.name.toLowerCase().includes('male')) ||
                                    voices.find(v => v.lang === 'en'); 
                        femaleVoice = voices.find(v => v.lang === 'en-US' && v.name.includes('Google US English') && v.name.toLowerCase().includes('female')) ||
                                      voices.find(v => v.lang === 'en-US' && v.name.toLowerCase().includes('female')) ||
                                      voices.find(v => v.lang === 'en');
                    };
                    if (window.speechSynthesis.getVoices().length > 0) {
                        window.speechSynthesis.onvoiceschanged(); // Trigger if voices already loaded
                    }
                } else {
                    console.warn("Speech Synthesis API not supported in this browser.");
                }

                // Initial state setup
                p.resetAnimation();
                narrationScript[0].syncAction(); // Set initial visual state without starting narration
            };

            p.draw = () => {
                const bg = p.color(colors.bgColor);
                p.background(bg);

                const lerpFactor = 0.05;
                camX = p.lerp(camX, targetCamX, lerpFactor);
                camY = p.lerp(camY, targetCamY, lerpFactor);
                camZ = p.lerp(camZ, targetCamZ, lerpFactor);
                camRotX = p.lerp(camRotX, targetCamRotX, lerpFactor);
                camRotY = p.lerp(camRotY, targetCamRotY, lerpFactor);
                camRotZ = p.lerp(camRotZ, targetCamRotZ, lerpFactor);

                p.camera(camX, camY, camZ, 0, 0, 0, 0, 1, 0); 

                p.ambientLight(50, 50, 50);
                p.directionalLight(255, 255, 255, -0.5, 0.8, -1);
                p.pointLight(p.color(255, 150, 0), 0, 0, 0); // Light from the center of core

                p.push();
                p.rotateY(p.frameCount * 0.001); // Global slow rotation
                p.rotateX(p.frameCount * 0.0005);

                // Earth or Mantle (for initial cross-section view)
                if (mantleAlpha > 0) {
                    p.noStroke();
                    p.specularMaterial(p.color(20, 80, 200, mantleAlpha)); // Blue-ish for mantle
                    p.shininess(10);
                    p.sphere(earthRadius); 
                }

                // Outer Core
                if (outerCoreAlpha > 0) {
                    p.push();
                    p.rotateY(p.frameCount * 0.0008); // Outer core slow rotation
                    p.specularMaterial(255, 180, 0, outerCoreAlpha); // Gold/Orange with alpha
                    p.shininess(100);
                    p.noStroke();
                    p.sphere(outerCoreRadius);

                    // Convection particles for outer core
                    if (currentState >= STATES.OUTER_CORE_EMPHASIS) {
                        p.push();
                        p.stroke(p.color(255, 200, 50, 200)); 
                        p.strokeWeight(particleSize);
                        p.noFill(); 
                        for (let particle of convectionParticles) {
                            let t = p.frameCount * 0.005;
                            let newX = particle.pos.x + p.map(p.noise(particle.noiseOffset.x + t), 0, 1, -5, 5);
                            let newY = particle.pos.y + p.map(p.noise(particle.noiseOffset.y + t), 0, 1, -5, 5);
                            let newZ = particle.pos.z + p.map(p.noise(particle.noiseOffset.z + t), 0, 1, -5, 5);

                            let currentPos = p.createVector(newX, newY, newZ);
                            // Keep particles within outer core bounds
                            if (currentPos.mag() > outerCoreRadius - particleSize) {
                                currentPos.setMag(outerCoreRadius - particleSize);
                            }
                            if (currentPos.mag() < innerCoreRadius + particleSize) {
                                currentPos.setMag(innerCoreRadius + particleSize);
                            }
                            particle.pos = currentPos;
                            p.point(particle.pos.x, particle.pos.y, particle.pos.z);
                        }
                        p.pop();
                    }
                    p.pop();
                }

                // Inner Core
                if (innerCoreAlpha > 0) {
                    p.push();
                    if (currentState === STATES.INNER_CORE_ROTATION) {
                        p.rotateY(p.frameCount * 0.005); // Faster rotation for emphasis
                    } else {
                        p.rotateY(p.frameCount * 0.001); // Slower default rotation
                    }
                    p.specularMaterial(255, 255, 150, innerCoreAlpha); 
                    p.shininess(200);
                    p.emissiveMaterial(255, 200, 0, innerCoreAlpha * 0.5); // Internal glow
                    p.sphere(innerCoreRadius);
                    p.pop();
                }

                // Lehmann Discontinuity (between outer and inner core)
                if (currentState === STATES.LEHMANN_DISCONTINUITY) {
                    p.push();
                    p.noFill();
                    p.stroke(p.color(255, 255, 255, p.map(p.sin(p.frameCount * 0.1), -1, 1, 100, 255))); // Pulsating white line
                    p.strokeWeight(2);
                    p.sphere(innerCoreRadius + 2); // Slightly larger sphere for boundary
                    p.pop();
                }

                // Magnetic Field Lines
                if (magneticFieldAlpha > 0) {
                    p.push();
                    p.stroke(p.color(colors.primaryColor, magneticFieldAlpha)); 
                    p.strokeWeight(2);
                    p.noFill();
                    for (let i = 0; i < numFieldLines; i++) {
                        let angleOffset = p.frameCount * 0.002;
                        let startRadius = outerCoreRadius * 0.7;
                        let endRadius = outerCoreRadius * 1.5;

                        // Calculate start and end points for a field line arc
                        let x1 = startRadius * p.cos(angleOffset + i * p.TWO_PI / numFieldLines);
                        let y1 = startRadius * p.sin(angleOffset + i * p.TWO_PI / numFieldLines);
                        let z1 = p.map(p.noise(i * 0.1 + p.frameCount * 0.01), 0, 1, -startRadius * 0.8, startRadius * 0.8);

                        let x2 = endRadius * p.cos(angleOffset + i * p.TWO_PI / numFieldLines + p.PI / 2); // Shift phase for arc
                        let y2 = endRadius * p.sin(angleOffset + i * p.TWO_PI / numFieldLines + p.PI / 2);
                        let z2 = p.map(p.noise(i * 0.1 + p.frameCount * 0.01 + 100), 0, 1, -endRadius * 1.2, endRadius * 1.2);

                        // Control points for bezier curve, extending outwards
                        let cp1x = x1 * 1.5; let cp1y = y1 * 1.5; let cp1z = z1 * 1.5;
                        let cp2x = x2 * 0.8; let cp2y = y2 * 0.8; let cp2z = z2 * 0.8;
                        
                        p.bezier(x1, y1, z1, cp1x, cp1y, cp1z, cp2x, cp2y, cp2z, x2, y2, z2);
                    }
                    p.pop();
                }

                p.pop(); 
            };

            p.windowResized = () => {
                p.resizeCanvas(p.windowWidth, p.windowHeight);
            };

            p.resetAnimation = () => {
                window.speechSynthesis.cancel();
                narrationActive = false;
                currentNarrationIndex = 0;
                currentState = STATES.INITIAL;
                animationStartTime = p.millis(); 

                camX = targetCamX = 0;
                camY = targetCamY = 0;
                camZ = targetCamZ = earthRadius * 1.5;
                camRotX = targetCamRotX = 0;
                camRotY = targetCamRotY = 0;
                camRotZ = targetCamRotZ = 0;

                mantleAlpha = 255;
                outerCoreAlpha = 0;
                innerCoreAlpha = 0;
                magneticFieldAlpha = 0;
                convectionParticles = [];
                magneticFieldLines = [];
            };

            p.startNarration = () => {
                p.resetAnimation(); 
                narrationActive = true;
                p.speakNarrationStep();
            };

            p.speakNarrationStep = () => {
                if (!narrationActive || currentNarrationIndex >= narrationScript.length) {
                    narrationActive = false;
                    currentState = STATES.COMPLETE; 
                    return;
                }

                const step = narrationScript[currentNarrationIndex];
                currentUtterance = new SpeechSynthesisUtterance(step.text);

                if (step.voice === 'male' && maleVoice) {
                    currentUtterance.voice = maleVoice;
                } else if (step.voice === 'female' && femaleVoice) {
                    currentUtterance.voice = femaleVoice;
                } else if (voices.length > 0) { 
                    currentUtterance.voice = voices[0];
                } else {
                    console.warn("No suitable voice found. Speech synthesis may not work.");
                }

                currentUtterance.pitch = step.pitch;
                currentUtterance.rate = step.rate;

                currentUtterance.onstart = () => {
                    step.syncAction(); 
                };

                currentUtterance.onend = () => {
                    currentNarrationIndex++;
                    p.speakNarrationStep(); 
                };

                currentUtterance.onerror = (event) => {
                    console.error('Speech synthesis error:', event);
                };

                window.speechSynthesis.speak(currentUtterance);
            };

            // Expose functions to global scope for button access
            p.resetAnimation = p.resetAnimation;
            p.startNarration = p.startNarration;
            p.narrationScript = narrationScript; // Expose for initial visual state
        };

        coreSketch = new p5(sketch);

        document.getElementById('replay').addEventListener('click', () => {
            coreSketch.resetAnimation();
            coreSketch.narrationScript[0].syncAction(); // Reset to initial visual state
        });

        document.getElementById('podcast').addEventListener('click', () => {
            coreSketch.startNarration();
        });

    </script>
