<description>
The "A-Z alphabet book, one by one alphabet will come" concept represents a systematic pedagogical approach to foundational linguistic and cognitive development. Scientifically, it's a meticulously structured sequential information delivery system. Each alphabet letter functions as a discrete graphemic unit, presented individually to optimize cognitive load and facilitate focused attention. This method leverages principles of chunking, where complex information (the entire alphabet) is broken down into manageable, easily digestible components.

The process typically involves multimodal sensory input: visual recognition of the grapheme, auditory processing of its phoneme, and often associative learning by linking the letter to a familiar object (e.g., 'A' for 'Apple'). This multi-sensory integration stimulates various neural pathways, including the visual cortex for pattern recognition, the auditory cortex for sound-symbol correspondence, and the hippocampus for memory encoding. The sequential presentation, coupled with repetition and reinforcement as new letters are introduced, strengthens synaptic connections and consolidates knowledge into long-term semantic memory. This incremental acquisition builds essential phonological awareness and letter knowledge, critical precursors for reading fluency and broader language acquisition. It exemplifies a scaffolded learning environment, progressively building complexity upon established foundational units.

    <script>
        let alphabetData = [];

        let animationStage = 0; // 0: intro/setup, 1: letter grow, 2: object appear, 3: sound wave, 4: consolidate
        let stageStartTime;
        const STAGE_DURATION = 1500; // Duration for each major animation stage (ms)
        const LETTER_TRANSITION_OUT_DURATION = 1000; // Duration for letter consolidating (ms)

        let particles = [];
        const NUM_PARTICLES = 100;

        // Narration variables
        let voices = [];
        let maleVoices = [];
        let femaleVoices = [];
        let currentUtterance = null;
        let podcastActive = false;
        let narrationIndex = 0;

        const podcastScript = [

        function preload() {
            // Populate alphabet data
            for (let i = 0; i < 26; i++) {
                let letter = String.fromCharCode(65 + i);
                let object = getObjectForLetter(letter); // Custom function to get objects
                alphabetData.push({ letter: letter, object: object, processed: false, pos: null, size: 0, opacity: 0, objectOpacity: 0, soundOpacity: 0 });
            }
        }

        function getObjectForLetter(letter) {
            switch (letter) {
                case 'A': return 'Apple';
                case 'B': return 'Ball';
                case 'C': return 'Cat';
                case 'D': return 'Dog';
                case 'E': return 'Egg';
                case 'F': return 'Fish';
                case 'G': return 'Grape';
                case 'H': return 'Hat';
                case 'I': return 'Ice';
                case 'J': return 'Jar';
                case 'K': return 'Kite';
                case 'L': return 'Lion';
                case 'M': return 'Moon';
                case 'N': return 'Nest';
                case 'O': return 'Owl';
                case 'P': return 'Pig';
                case 'Q': return 'Queen';
                case 'R': return 'Ring';
                case 'S': return 'Sun';
                case 'T': return 'Tree';
                case 'U': return 'Umbrella';
                case 'V': return 'Van';
                case 'W': return 'Wolf';
                case 'X': return 'Xylophone';
                case 'Y': return 'Yoyo';
                case 'Z': return 'Zebra';
                default: return 'Object';
            }
        }

        function setup() {
            let canvas = createCanvas(windowWidth, windowHeight);
            canvas.parent('p5-container');
            frameRate(30);
            resetAnimation();
            populateParticles();
            
            // Get voices for narration
            if (speechSynthesis) {
                speechSynthesis.onvoiceschanged = () => {
                    voices = speechSynthesis.getVoices();
                    // Attempt to find specific English voices
                    maleVoices = voices.filter(voice => voice.lang === 'en-US' && (voice.name.includes('Google US English Male') || voice.name.includes('Alex')));
                    femaleVoices = voices.filter(voice => voice.lang === 'en-US' && (voice.name.includes('Google US English Female') || voice.name.includes('Samantha')));

                    // Fallback to any English voice if specific ones not found
                    if (maleVoices.length === 0) maleVoices = voices.filter(v => v.lang.startsWith('en') && (v.name.includes('Zira') || v.name.includes('David') || v.name.includes('Google')));
                    if (femaleVoices.length === 0) femaleVoices = voices.filter(v => v.lang.startsWith('en') && (v.name.includes('Zira') || v.name.includes('David') || v.name.includes('Google')));

                    // Final fallback to any voice
                    if (maleVoices.length === 0 && femaleVoices.length === 0) {
                        maleVoices = voices;
                        femaleVoices = voices;
                    }
                };
                // For some browsers, voices might be available immediately on page load
                voices = speechSynthesis.getVoices();
                if (voices.length > 0) {
                     speechSynthesis.onvoiceschanged(); // Trigger populate if voices already there
                }
            }

            document.getElementById('replay').addEventListener('click', () => {
                stopPodcast();
                resetAnimation();
            });
            document.getElementById('podcast').addEventListener('click', startPodcast);
        }

        function resetAnimation() {
            stopPodcast(); // Ensure no active narration
            currentAlphabetIndex = -1;
            animationStage = 0;
            stageStartTime = millis();
            for (let i = 0; i < alphabetData.length; i++) {
                alphabetData[i].processed = false;
                // Reset visual properties for all letters
                alphabetData[i].pos = createVector(width / 2, height / 2);
                alphabetData[i].size = 0;
                alphabetData[i].opacity = 0;
                alphabetData[i].objectOpacity = 0;
                alphabetData[i].soundOpacity = 0;
            }
            populateParticles();
        }

        function populateParticles() {
            particles = [];
            for (let i = 0; i < NUM_PARTICLES; i++) {
                particles.push({
                    pos: createVector(random(width), random(height)),
                    vel: p5.Vector.random2D().mult(random(0.5, 1.5)),
                    size: random(2, 5),
                    color: color(100, 150, 255, 150), // Light blue with transparency
                    target: null // Target for molecular attraction
                });
            }
        }

        function startPodcast() {
            stopPodcast(); // Stop any ongoing narration first
            resetAnimation(); // Reset visuals
            podcastActive = true;
            narrationIndex = 0;
            stageStartTime = millis(); // Ensure stageStartTime is reset for visuals
            playNextNarration();
        }

        function stopPodcast() {
            if (currentUtterance) {
                speechSynthesis.cancel();
            }
            podcastActive = false;
            currentUtterance = null;
        }

        function playNextNarration() {
            if (!podcastActive || narrationIndex >= podcastScript.length) {
                podcastActive = false;
                currentAlphabetIndex = -1; // Reset to intro state after podcast ends
                animationStage = 0;
                return;
            }

            let script = podcastScript[narrationIndex];
            currentUtterance = new SpeechSynthesisUtterance(script.text);
            
            let voicePool = (script.voice === 'male' && maleVoices.length > 0) ? maleVoices : (femaleVoices.length > 0 ? femaleVoices : voices);
            if (voicePool.length > 0) {
                currentUtterance.voice = voicePool[Math.floor(random(voicePool.length))];
            }
            currentUtterance.pitch = script.pitch;
            currentUtterance.rate = script.rate;

            currentUtterance.onend = () => {
                narrationIndex++;
                playNextNarration();
            };
            currentUtterance.onerror = (event) => {
                console.error("SpeechSynthesisUtterance error:", event.error);
                narrationIndex++; // Still try to advance
                playNextNarration();
            };

            speechSynthesis.speak(currentUtterance);
        }

        function draw() {
            background(getCSSVariable('--bg-color').toString());

            updateParticles();
            drawParticles();

            let elapsed = millis() - stageStartTime;
            let currentScript;

            if (podcastActive) {
                currentScript = podcastScript[narrationIndex];
                if (currentScript && currentScript.visualCue !== undefined) {
                    if (currentAlphabetIndex !== currentScript.visualCue) {
                        currentAlphabetIndex = currentScript.visualCue;
                        animationStage = currentScript.stage || 1; // Default to stage 1 if not specified
                        stageStartTime = millis(); // Reset timer for new visual cue
                    }
                    if (currentScript.stage !== undefined && animationStage !== currentScript.stage) {
                        animationStage = currentScript.stage;
                        stageStartTime = millis();
                    }
                }
            } else {
                // Manual progression if no podcast or podcast is finished
                if (currentAlphabetIndex === -1 && animationStage === 0) {
                    // Initial state, waiting for input
                    displayIntroText();
                    return;
                } else if (currentAlphabetIndex === -1 && animationStage > 0) { // If it's already started without podcast
                    currentAlphabetIndex = 0;
                    animationStage = 1;
                    stageStartTime = millis();
                }

                if (currentAlphabetIndex >= 0 && currentAlphabetIndex < alphabetData.length) {
                    if (animationStage < 4 && elapsed > STAGE_DURATION) {
                        animationStage++;
                        stageStartTime = millis();
                    } else if (animationStage === 4 && elapsed > LETTER_TRANSITION_OUT_DURATION) {
                        // Letter consolidation finished, move to next
                        alphabetData[currentAlphabetIndex].processed = true;
                        currentAlphabetIndex++;
                        animationStage = 0; // Reset stage for next letter
                        stageStartTime = millis();
                    }
                } else if (currentAlphabetIndex >= alphabetData.length) {
                    // Animation completed, reset to initial state after a delay
                    if (millis() - stageStartTime > 3000) { // Wait 3 seconds after last letter
                        resetAnimation();
                    }
                    return;
                }
            }

            if (currentAlphabetIndex === -1) {
                displayIntroText(); // Display intro text if just started
                return;
            }
            
            // Draw main content area - glassmorphism effect
            let panelWidth = width * 0.7;
            let panelHeight = height * 0.6;
            let panelX = width / 2;
            let panelY = height / 2;

            noStroke();
            fill(getCSSVariable('--glass-bg'));
            rectMode(CENTER);
            rect(panelX, panelY, panelWidth, panelHeight, 20); // Rounded rectangle
            stroke(getCSSVariable('--glass-border'));
            strokeWeight(1);
            noFill();
            rect(panelX, panelY, panelWidth, panelHeight, 20);

            // Draw current letter
            let currentLetterData = alphabetData[currentAlphabetIndex];
            if (currentLetterData) {
                let currentLetter = currentLetterData.letter;
                let currentObject = currentLetterData.object;

                let letterScale = 0;
                let letterAlpha = 0;
                let objectAlpha = 0;
                let soundAlpha = 0;

                let progress = 0;
                if (animationStage < 4) {
                    progress = min(1, elapsed / STAGE_DURATION); // Progress through current stage
                } else if (animationStage === 4) {
                    progress = min(1, elapsed / LETTER_TRANSITION_OUT_DURATION); // For consolidation stage
                }

                switch (animationStage) {
                    case 0: // Intro or waiting state
                        letterScale = 0; // Letter not visible
                        letterAlpha = 0;
                        break;
                    case 1: // Letter growing, fully visible
                        letterScale = map(progress, 0, 1, 0.5, 1);
                        letterAlpha = map(progress, 0, 1, 0, 255);
                        break;
                    case 2: // Object appears
                        letterScale = 1;
                        letterAlpha = 255;
                        objectAlpha = map(progress, 0, 1, 0, 255);
                        break;
                    case 3: // Sound wave
                        letterScale = 1;
                        letterAlpha = 255;
                        objectAlpha = 255;
                        soundAlpha = map(progress, 0, 1, 0, 255);
                        break;
                    case 4: // Consolidating / fading out
                        letterScale = map(progress, 0, 1, 1, 0.5);
                        letterAlpha = map(progress, 0, 1, 255, 0);
                        objectAlpha = map(progress, 0, 1, 255, 0);
                        soundAlpha = map(progress, 0, 1, 255, 0);
                        
                        // Move letter to memory bank
                        let targetX = map(currentAlphabetIndex, 0, alphabetData.length -1, 50, width - 50);
                        let targetY = 50; // Top of the screen for memory bank

                        currentLetterData.pos.x = lerp(width/2, targetX, progress);
                        currentLetterData.pos.y = lerp(height/2, targetY, progress);
                        currentLetterData.size = lerp(120, 20, progress);
                        currentLetterData.opacity = lerp(255, 50, progress);
                        currentLetterData.objectOpacity = lerp(255, 0, progress); // Object fades completely
                        currentLetterData.soundOpacity = lerp(255, 0, progress); // Sound fades completely
                        break;
                }

                push();
                translate(width / 2, height / 2);
                textAlign(CENTER, CENTER);
                textFont('Segoe UI', BOLD);
                
                // Letter
                fill(getCSSVariable('--text-color', letterAlpha));
                textSize(120 * letterScale);
                text(currentLetter, 0, -40);

                // Associated Object
                if (objectAlpha > 0) {
                    fill(getCSSVariable('--primary-color', objectAlpha));
                    textSize(40 * letterScale);
                    text(currentObject, 0, 60);
                }

                // Sound Wave
                if (soundAlpha > 0) {
                    noFill();
                    stroke(getCSSVariable('--glow-color', soundAlpha));
                    strokeWeight(3);
                    let waveFactor = map(progress, 0, 1, 0.5, 1.5);
                    for (let i = 0; i < 3; i++) {
                        let radius = map(sin(frameCount * 0.1 + i * PI / 4), -1, 1, 80, 120) * waveFactor;
                        ellipse(0, 0, radius * 2, radius * 0.8);
                    }
                }
                pop();

                // Target particles to the letter if it's active
                if (animationStage > 0 && animationStage < 4) {
                    let targetPos = createVector(width / 2, height / 2 - 40);
                    particles.forEach(p => p.target = targetPos);
                } else {
                    particles.forEach(p => p.target = null); // No target, just drift
                }
            }

            // Draw processed letters in memory bank
            drawMemoryBank();
        }

        function drawMemoryBank() {
            for (let i = 0; i < alphabetData.length; i++) {
                let data = alphabetData[i];
                if (data.processed || (currentAlphabetIndex === i && animationStage === 4)) {
                    let alpha = data.opacity || 50; // Faded if processed, or current if fading out
                    let size = data.size || 20;
                    let x = data.pos.x || map(i, 0, alphabetData.length - 1, 50, width - 50); // Default position
                    let y = data.pos.y || 50; // Fixed row for simplicity

                    // Ensure x,y are within bounds
                    x = constrain(x, 20, width - 20);
                    y = constrain(y, 20, height - 20);

                    push();
                    textAlign(CENTER, CENTER);
                    fill(getCSSVariable('--text-color', alpha));
                    textSize(size);
                    text(data.letter, x, y);
                    pop();
                }
            }
        }

        function updateParticles() {
            for (let p of particles) {
                if (p.target) {
                    let force = p5.Vector.sub(p.target, p.pos);
                    force.setMag(0.1); // Attraction strength
                    p.vel.add(force);
                    p.vel.mult(0.98); // Drag
                } else {
                    // Drift randomly
                    p.vel.x += random(-0.1, 0.1);
                    p.vel.y += random(-0.1, 0.1);
                    p.vel.limit(2);
                }
                
                p.pos.add(p.vel);

                // Wrap around edges
                if (p.pos.x < 0) p.pos.x = width;
                if (p.pos.x > width) p.pos.x = 0;
                if (p.pos.y < 0) p.pos.y = height;
                if (p.pos.y > height) p.pos.y = 0;
            }
        }

        function drawParticles() {
            noStroke();
            for (let p of particles) {
                fill(p.color);
                ellipse(p.pos.x, p.pos.y, p.size);
            }

            // Draw connections between nearby particles
            stroke(getCSSVariable('--glow-color', 50));
            strokeWeight(1);
            for (let i = 0; i < particles.length; i++) {
                for (let j = i + 1; j < particles.length; j++) {
                    let d = dist(particles[i].pos.x, particles[i].pos.y, particles[j].pos.x, particles[j].pos.y);
                    if (d < 70) {
                        line(particles[i].pos.x, particles[i].pos.y, particles[j].pos.x, particles[j].pos.y);
                    }
                }
            }
        }

        function getCSSVariable(name, alpha = 255) {
            let colorStr = getComputedStyle(document.documentElement).getPropertyValue(name).trim();
            if (colorStr.startsWith('rgba')) {
                let parts = colorStr.match(/rgba\((\d+),\s*(\d+),\s*(\d+),\s*([\d.]+)\)/);
                if (parts) {
                    let r = parseInt(parts[1]);
                    let g = parseInt(parts[2]);
                    let b = parseInt(parts[3]);
                    // If an alpha value is passed, use it, otherwise use the one from the CSS variable
                    return color(r, g, b, alpha !== 255 ? alpha : parseFloat(parts[4]) * 255);
                }
            }
            // For rgb, hex, or named colors, convert to p5 color with given alpha
            let c = color(colorStr);
            c.setAlpha(alpha);
            return c;
        }

        function displayIntroText() {
            push();
            textAlign(CENTER, CENTER);
            fill(getCSSVariable('--text-color', 200));
            textSize(40);
            text("Alphabet Learning System", width / 2, height / 2 - 50);
            textSize(24);
            fill(getCSSVariable('--text-color', 150));
            text("Click 'Podcast' for narrated scientific explanation\nor 'Replay' for visual demonstration.", width / 2, height / 2 + 50);
            pop();
        }

        function windowResized() {
            resizeCanvas(windowWidth, windowHeight);
            populateParticles(); // Reposition particles for new canvas size
            resetAnimation(); // Reset the visual layout
        }
    </script>
