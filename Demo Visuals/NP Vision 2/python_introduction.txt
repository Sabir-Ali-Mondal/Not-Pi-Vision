<description>
Python, introduced by Guido van Rossum in 1991, is a high-level, general-purpose, interpreted programming language renowned for its emphasis on code readability and a syntax that allows programmers to express concepts in fewer lines of code than in languages like C++ or Java. Its design philosophy, encapsulated in "The Zen of Python" (PEP 20), promotes principles such as "Explicit is better than implicit" and "Readability counts." Architecturally, Python code is compiled into bytecode, which is then executed by the Python Virtual Machine (PVM) – primarily CPython, the reference implementation. This interpretation step, while offering platform independence and rapid development cycles, can incur performance overhead, especially for CPU-bound tasks, a limitation often exacerbated by the Global Interpreter Lock (GIL) in CPython, which restricts concurrent execution of native threads.

Python employs a dynamic, strong typing system, meaning variable types are determined at runtime, and implicit type coercion between incompatible types is minimized. It fully supports multiple programming paradigms, including object-oriented (classes, inheritance, polymorphism), imperative, and functional programming. Its extensive standard library, alongside a vast ecosystem of third-party packages (accessible via PyPI), provides robust functionalities for diverse domains such as web development (Django, Flask), scientific computing (NumPy, SciPy, Pandas), machine learning (TensorFlow, PyTorch, Scikit-learn), data analysis, automation, and system administration. Python's versatility and a large, supportive community have cemented its status as a cornerstone language in modern software development and data science.

        // Draws an arrow that animates its length based on progress
        function drawArrow(x1, y1, x2, y2, animProgress, animStart, animEnd, isHighlight, animationDuration = 1000) {
            let segmentProgress = map(millis() - slideStartTime, 0, animationDuration, 0, 1);
            segmentProgress = constrain(segmentProgress, animStart, animEnd);

            if (segmentProgress === 0) return;

            let currentX = lerp(x1, x2, segmentProgress);
            let currentY = lerp(y1, y2, segmentProgress);

            push();
            if (isHighlight) {
                stroke(c_glowColor);
                strokeWeight(3);
                drawingContext.shadowBlur = 15;
                drawingContext.shadowColor = c_glowColor.toString();
            } else {
                stroke(c_primaryColor);
                strokeWeight(2);
            }
            line(x1, y1, currentX, currentY);

            // Draw arrowhead only when the line is almost fully drawn
            if (segmentProgress >= animEnd * 0.95) { // Threshold for arrow head to appear
                let arrowSize = 10;
                let angle = atan2(y2 - y1, x2 - x1);
                push();
                translate(currentX, currentY); // Use currentX/Y for dynamic arrow head position
                rotate(angle);
                triangle(0, 0, -arrowSize, arrowSize / 2, -arrowSize, -arrowSize / 2);
                pop();
            }
            drawingContext.shadowBlur = 0;
            pop();
        }

        // Easing function for smoother transitions
        function easeInOutQuad(t) {
            return t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;
        }

        }

        function stopNarration() {
            if (!isNarrating) return;

            synth.cancel();
            isNarrating = false;
            currentHighlightKey = null;
            if (speakingUtterance) {
                speakingUtterance.onend = null;
                speakingUtterance = null;
            }
        }

        function speakNextUtterance() {
            if (!isNarrating || utteranceIndex >= narrationScript.length) {
                stopNarration();
                return;
            }

            const { text, voiceIndex, slide, highlight } = narrationScript[utteranceIndex];

            // Trigger slide change if necessary
            if (slide !== currentSlide) {
                currentSlide = slide;
                slideStartTime = millis();
                isTransitioning = true;
            } else {
                slideStartTime = millis(); // Reset start time for internal animations of current slide
                isTransitioning = false;
            }

            currentHighlightKey = highlight || null; // Set highlight key

            speakingUtterance = new SpeechSynthesisUtterance(text);
            if (voices.length > 0) {
                speakingUtterance.voice = voices[voiceIndex % voices.length];
                speakingUtterance.pitch = map(voiceIndex % 2, 0, 1, 0.9, 1.1);
                speakingUtterance.rate = map(voiceIndex % 2, 0, 1, 1.0, 1.05);
            }

            speakingUtterance.onend = () => {
                const nextUtterance = narrationScript[utteranceIndex + 1];
                if (!nextUtterance || nextUtterance.highlight !== currentHighlightKey) {
                    currentHighlightKey = null; // Clear highlight if next utterance doesn't need it
                }
                utteranceIndex++;
                speakNextUtterance();
            };

            speakingUtterance.onerror = (event) => {
                console.error('SpeechSynthesisUtterance.onerror', event);
                // Attempt to proceed if an error occurs
                utteranceIndex++;
                speakNextUtterance();
            };

            synth.speak(speakingUtterance);
        }

        // --- Button Event Listeners ---
        document.getElementById('replay').addEventListener('click', () => {
            stopNarration();
            resetAnimation();
        });

        document.getElementById('podcast').addEventListener('click', () => {
            if (isNarrating) {
                stopNarration();
            } else {
                startNarration();
            }
        });
