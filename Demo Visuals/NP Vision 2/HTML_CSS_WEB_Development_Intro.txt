<description>
Web development, particularly the foundational "HTML CSS WEB Development Intro," involves the systematic construction of interactive digital interfaces accessed via the World Wide Web. At its core lies HyperText Markup Language (HTML), a declarative markup language that defines the semantic structure and content of a web page. HTML elements, comprising tags such as `<div>`, `<p>`, `<a>`, and `<img>`, encapsulate textual, multimedia, and interactive components. The browser's rendering engine parses the HTML document, constructing a Document Object Model (DOM) tree, an in-memory representation of the page's hierarchical structure.

Cascading Style Sheets (CSS) serve as a complementary declarative language, dictating the visual presentation and layout of HTML elements. CSS rules, defined through selectors and declarations, enable precise control over typography, colors, spacing, positioning, and responsiveness across various devices. The "cascading" aspect refers to the prioritization rules governing style application when multiple conflicting styles are declared. Together, HTML and CSS embody the principle of "separation of concerns," distinguishing content and structure from its aesthetic presentation. Modern web development frequently leverages client-side scripting (e.g., JavaScript) for dynamic behavior and server-side logic for data management, culminating in a sophisticated client-server architecture. The browser's rendering pipeline then combines the DOM and CSS Object Model (CSSOM) to paint pixels onto the screen, resulting in the final user interface.

        let currentStage = 0;

        // Narration script
        const narrationScript = [
            { text: "Greetings! Let's embark on a journey to understand how websites are built.", voice: 'female', pitch: 1.1, rate: 0.9, visualStage: 0, highlightElement: 'client-server' },

        // p5.js sketch
        let sketch = function(p) {
            let clientPos, serverPos;
            let htmlFilePos, cssFilePos;
            let htmlElements = [];
            let cssParticles = [];
            let domNodes = [];
            let renderRect;
            
            p.setup = function() {
                p.createCanvas(p.windowWidth, p.windowHeight);
                p.frameRate(30);
                p.angleMode(p.DEGREES);

                p.initVoices(); // Load voices
                p.resetAnimation(); // Initialize positions and states
            };

            p.initVoices = function() {
                if ('speechSynthesis' in window) {
                    speechSynthesis.onvoiceschanged = () => {
                        voices = speechSynthesis.getVoices();
                        // Attempt to find specific voices, fall back to gender-based, then first available
                        maleVoice = voices.find(v => v.name.includes('Google US English') && (v.name.includes('Male') || v.name.includes('en-US-Standard-C'))) || voices.find(v => v.gender === 'male' || v.lang === 'en-US') || voices[0];
                        femaleVoice = voices.find(v => v.name.includes('Google US English') && (v.name.includes('Female') || v.name.includes('en-US-Standard-A'))) || voices.find(v => v.gender === 'female' || v.lang === 'en-US') || voices[1];
                        if (!maleVoice && voices.length > 0) maleVoice = voices[0];
                        if (!femaleVoice && voices.length > 1) femaleVoice = voices[1];
                        if (!femaleVoice && voices.length > 0) femaleVoice = voices[0]; // Fallback to first if only one voice
                    };
                } else {
                    console.warn("SpeechSynthesis API not supported.");
                }
            };

            p.resetAnimation = function() {
                currentStage = 0;
                animationProgress = 0;
                narrationIndex = 0;
                isAnimating = true;
                speaking = false;
                highlightElement = null;

                // Stop any ongoing speech
                if ('speechSynthesis' in window && speaking) {
                    speechSynthesis.cancel();
                    speaking = false;
                }

                // Initialize positions for all elements for smooth restart
                clientPos = p.createVector(p.width * 0.15, p.height / 2);
                serverPos = p.createVector(p.width * 0.85, p.height / 2);
                htmlFilePos = p.createVector(serverPos.x, serverPos.y); // Starts at server
                cssFilePos = p.createVector(serverPos.x, serverPos.y + 100); // Starts at server area

                htmlElements = [];
                // Simple HTML structure: body, div, p, img
                // Coordinates are relative to p.width/2, p.height/2 for easier scaling/translation
                htmlElements.push({ id: 'html-body', x: 0, y: 0, w: p.width * 0.5, h: p.height * 0.6, type: 'body', label: '<body>', parent: null });
                htmlElements.push({ id: 'html-div', x: 0, y: -p.height * 0.1, w: p.width * 0.4, h: p.height * 0.2, type: 'div', label: '<div>', parentId: 'html-body' });
                htmlElements.push({ id: 'html-p', x: 0, y: p.height * 0.05, w: p.width * 0.3, h: p.height * 0.08, type: 'p', label: '<p>', parentId: 'html-div' });
                htmlElements.push({ id: 'html-img', x: 0, y: p.height * 0.18, w: p.width * 0.2, h: p.height * 0.15, type: 'img', label: '<img>', parentId: 'html-div' });


                domNodes = [];
                // DOM Nodes will be constructed visually on stage 3
                domNodes.push({id: 'dom-document', label: 'Document', level: 0, xOffset: 0});
                domNodes.push({id: 'dom-html', label: 'HTML', level: 1, xOffset: 0});
                domNodes.push({id: 'dom-head', label: 'HEAD', level: 2, xOffset: -0.1});
                domNodes.push({id: 'dom-body', label: 'BODY', level: 2, xOffset: 0.1});
                domNodes.push({id: 'dom-div', label: 'DIV', level: 3, xOffset: -0.05});
                domNodes.push({id: 'dom-p', label: 'P', level: 4, xOffset: -0.05});
                domNodes.push({id: 'dom-img', label: 'IMG', level: 4, xOffset: 0.05});


                cssParticles = [];
                renderRect = { x: p.width / 2, y: p.height / 2, w: 0, h: 0 };
            };

            p.draw = function() {
                p.background(bgColor[0], bgColor[1], bgColor[2]);

                if (!isAnimating) return;

                // Animate progress within a stage if not podcasting
                if (!isPodcastActive) {
                    animationProgress += 0.01; // Increase progress for replay mode
                    if (animationProgress >= 1.0) {
                        animationProgress = 0;
                        currentStage++;
                        if (currentStage > 5) { // Loop back after final stage
                            p.resetAnimation();
                        }
                    }
                }
                
                // --- Stage 0: Client-Server Request ---
                p.push();
                p.drawGlassBox(clientPos.x, clientPos.y, p.width * 0.2, p.height * 0.3, 'Client', highlightElement === 'client' || highlightElement === 'client-server');
                p.drawGlassBox(serverPos.x, serverPos.y, p.width * 0.2, p.height * 0.3, 'Server', highlightElement === 'server' || highlightElement === 'client-server');

                if (currentStage === 0) {
                    let progress = animationProgress;
                    if (isPodcastActive && narrationIndex === 1) progress = p.min(1, p.frameCount * 0.01); // Short burst for request
                    
                    let arrowStartX = clientPos.x + p.width * 0.1;
                    let arrowEndX = p.lerp(arrowStartX, serverPos.x - p.width * 0.1, progress);
                    let arrowY = p.height / 2;
                    p.drawArrow(arrowStartX, arrowY, arrowEndX, arrowY, primaryColor, 10, p.map(progress, 0, 1, 0, 255));
                    
                    if (progress > 0.3 && progress < 0.8) {
                        p.fill(textColor[0], textColor[1], textColor[2], p.map(progress, 0.3, 0.8, 0, 255));
                        p.noStroke();
                        p.textAlign(p.CENTER, p.CENTER);
                        p.textSize(16);
                        p.text('HTTP Request', (arrowStartX + arrowEndX) / 2, arrowY - 20);
                    }
                }
                p.pop();

                // --- Stage 1: HTML Structure ---
                if (currentStage >= 1) {
                    // Animate HTML file coming from server
                    p.push();
                    let targetHtmlFilePos = p.createVector(p.width * 0.3, p.height * 0.2);
                    let currentHtmlFileX, currentHtmlFileY;

                    if (currentStage === 1) {
                        currentHtmlFileX = p.lerp(serverPos.x, targetHtmlFilePos.x, animationProgress);
                        currentHtmlFileY = p.lerp(serverPos.y, targetHtmlFilePos.y, animationProgress);
                    } else {
                        currentHtmlFileX = targetHtmlFilePos.x;
                        currentHtmlFileY = targetHtmlFilePos.y;
                    }
                    p.drawGlassBox(currentHtmlFileX, currentHtmlFileY, p.width * 0.1, p.height * 0.1, 'HTML File', highlightElement === 'html-file');
                    p.pop();

                    // Draw HTML elements
                    p.push();
                    let baseHtmlX = p.width / 2;
                    let baseHtmlY = p.height / 2;
                    let scaleFactor = p.map(animationProgress, 0, 1, 0.5, 1);
                    if (currentStage > 1) scaleFactor = 1; // Keep full size after stage 1

                    p.translate(baseHtmlX, baseHtmlY);
                    p.scale(scaleFactor);

                    htmlElements.forEach(el => {
                        let isHighlighted = highlightElement === el.id || highlightElement === 'html-elements' || highlightElement === 'html-tags';
                        p.drawGlassRect(el.x, el.y, el.w, el.h, glassBg[3], isHighlighted);
                        
                        p.fill(textColor[0], textColor[1], textColor[2], p.map(animationProgress, 0, 1, 0, 255));
                        p.noStroke();
                        p.textAlign(p.CENTER, p.CENTER);
                        p.textSize(14);
                        p.text(el.label, el.x, el.y);

                        if (el.parentId) {
                            let parentEl = htmlElements.find(e => e.id === el.parentId);
                            if (parentEl) {
                                p.stroke(textColor[0], textColor[1], textColor[2], p.map(animationProgress, 0, 1, 0, 150));
                                p.strokeWeight(1);
                                p.line(el.x, el.y - el.h / 2, parentEl.x, parentEl.y + parentEl.h / 2);
                            }
                        }
                    });
                    p.pop();
                }

                // --- Stage 2: CSS Styling ---
                if (currentStage >= 2) {
                    p.push();
                    let targetCssFilePos = p.createVector(p.width * 0.7, p.height * 0.2);
                    let currentCssFileX, currentCssFileY;

                    if (currentStage === 2) {
                        currentCssFileX = p.lerp(serverPos.x, targetCssFilePos.x, animationProgress);
                        currentCssFileY = p.lerp(serverPos.y + 100, targetCssFilePos.y, animationProgress);
                    } else {
                        currentCssFileX = targetCssFilePos.x;
                        currentCssFileY = targetCssFilePos.y;
                    }
                    p.drawGlassBox(currentCssFileX, currentCssFileY, p.width * 0.1, p.height * 0.1, 'CSS File', highlightElement === 'css-file');
                    p.pop();

                    // Animate CSS application as particles
                    if (currentStage === 2) {
                        if (cssParticles.length < 50 && p.frameCount % 5 === 0) {
                            let randomHtmlEl = htmlElements[p.floor(p.random(htmlElements.length))];
                            cssParticles.push({
                                x: currentCssFileX + p.random(-20, 20),
                                y: currentCssFileY + p.random(-20, 20),
                                targetX: (p.width / 2) + randomHtmlEl.x + p.random(-20, 20),
                                targetY: (p.height / 2) + randomHtmlEl.y + p.random(-20, 20),
                                hue: p.random(255),
                                life: 1,
                                maxLife: 60
                            });
                        }

                        for (let i = cssParticles.length - 1; i >= 0; i--) {
                            let p_ = cssParticles[i];
                            p_.x = p.lerp(p_.x, p_.targetX, 0.05);
                            p_.y = p.lerp(p_.y, p_.targetY, 0.05);
                            p_.life -= 1;

                            p.colorMode(p.HSB, 255);
                            p.fill(p_.hue, 200, 255, p.map(p_.life, 0, p_.maxLife, 0, 150));
                            p.noStroke();
                            p.ellipse(p_.x, p_.y, 8);
                            p.colorMode(p.RGB, 255);

                            if (p_.life <= 0) {
                                cssParticles.splice(i, 1);
                            }
                        }

                        // Apply dynamic styling to HTML elements
                        p.push();
                        let baseHtmlX = p.width / 2;
                        let baseHtmlY = p.height / 2;
                        p.translate(baseHtmlX, baseHtmlY);
                        htmlElements.forEach(el => {
                            let isHighlighted = highlightElement === 'css-properties' || highlightElement === 'css-application';

                            let dynamicHue = p.map(p.sin(p.frameCount * 2 + el.x), -1, 1, 0, 255);
                            p.colorMode(p.HSB, 255);
                            let fillCol = p.color(dynamicHue, 150, 200, p.map(animationProgress, 0, 1, 0, 100)); // Apply color gradually
                            p.colorMode(p.RGB, 255);
                            
                            p.drawGlassRect(el.x, el.y, el.w, el.h, glassBg[3], isHighlighted, fillCol);
                            p.fill(textColor[0], textColor[1], textColor[2], 255);
                            p.noStroke();
                            p.textAlign(p.CENTER, p.CENTER);
                            p.textSize(14);
                            p.text(el.label, el.x, el.y);
                            
                            if (el.parentId) { // Re-draw connections for visibility
                                let parentEl = htmlElements.find(e => e.id === el.parentId);
                                if (parentEl) {
                                    p.stroke(textColor[0], textColor[1], textColor[2], 150);
                                    p.strokeWeight(1);
                                    p.line(el.x, el.y - el.h / 2, parentEl.x, parentEl.y + parentEl.h / 2);
                                }
                            }
                        });
                        p.pop();
                    }
                }

                // --- Stage 3: DOM Tree Construction ---
                if (currentStage >= 3) {
                    p.push();
                    let domTreeBaseX = p.width / 2;
                    let domTreeBaseY = p.height / 2;
                    let nodeGapY = p.height * 0.1;

                    domNodes.forEach((node, index) => {
                        let targetX = domTreeBaseX + node.xOffset * p.width;
                        let targetY = domTreeBaseY - p.height * 0.25 + node.level * nodeGapY;
                        
                        let isHighlighted = highlightElement === node.id || highlightElement === 'dom-tree' || highlightElement === 'dom-nodes';

                        // Animate nodes appearing
                        let currentAlpha = p.map(animationProgress, 0, 1, 0, 255);
                        if (!isPodcastActive && animationProgress * domNodes.length < index) {
                            currentAlpha = 0; // Stagger appearance in replay
                        } else if (isPodcastActive && currentStage !== 3) {
                            currentAlpha = 255; // Keep visible if passed stage 3
                        }


                        if (currentAlpha > 0) {
                            p.drawGlassNode(targetX, targetY, node.label, isHighlighted, currentAlpha);
                        }
                    });

                    // Draw connections based on hierarchy and animation progress
                    const connections = [
                        { parent: 'dom-document', child: 'dom-html' },
                        { parent: 'dom-html', child: 'dom-head' },
                        { parent: 'dom-html', child: 'dom-body' },
                        { parent: 'dom-body', child: 'dom-div' },
                        { parent: 'dom-div', child: 'dom-p' },
                        { parent: 'dom-div', child: 'dom-img' }
                    ];

                    connections.forEach((conn, index) => {
                        let parentNode = domNodes.find(n => n.id === conn.parent);
                        let childNode = domNodes.find(n => n.id === conn.child);
                        if (parentNode && childNode) {
                            let parentX = domTreeBaseX + parentNode.xOffset * p.width;
                            let parentY = domTreeBaseY - p.height * 0.25 + parentNode.level * nodeGapY;
                            let childX = domTreeBaseX + childNode.xOffset * p.width;
                            let childY = domTreeBaseY - p.height * 0.25 + childNode.level * nodeGapY;

                            let currentAlpha = p.map(animationProgress, 0, 1, 0, 255);
                            if (!isPodcastActive && animationProgress * domNodes.length < index + domNodes.indexOf(parentNode)) {
                                currentAlpha = 0; // Stagger connection appearance
                            } else if (isPodcastActive && currentStage !== 3) {
                                currentAlpha = 255;
                            }

                            if (currentAlpha > 0) {
                                p.drawConnection({x: parentX, y: parentY}, {x: childX, y: childY}, currentAlpha);
                            }
                        }
                    });
                    p.pop();
                }

                // --- Stage 4: Rendering to Screen ---
                if (currentStage >= 4) {
                    p.push();
                    let targetRenderW = p.width * 0.6;
                    let targetRenderH = p.height * 0.7;

                    if (currentStage === 4) {
                        renderRect.w = p.lerp(0, targetRenderW, animationProgress);
                        renderRect.h = p.lerp(0, targetRenderH, animationProgress);
                    } else {
                        renderRect.w = targetRenderW;
                        renderRect.h = targetRenderH;
                    }

                    p.drawGlassRect(p.width / 2, p.height / 2, renderRect.w, renderRect.h, glassBg[3], highlightElement === 'rendering');

                    if (currentStage === 4 && animationProgress > 0.3) {
                        // Simulate pixels or elements appearing
                        p.noStroke();
                        p.fill(primaryColor[0], primaryColor[1], primaryColor[2], p.map(animationProgress, 0.3, 1, 0, 150));
                        for (let i = 0; i < 50; i++) {
                            let rx = p.random(p.width / 2 - renderRect.w / 2 + 10, p.width / 2 + renderRect.w / 2 - 10);
                            let ry = p.random(p.height / 2 - renderRect.h / 2 + 10, p.height / 2 + renderRect.h / 2 - 10);
                            p.ellipse(rx, ry, p.random(5, 15));
                        }
                    }
                    p.pop();
                }

                // --- Stage 5: Final Web Page Displayed ---
                if (currentStage >= 5) {
                    p.push();
                    let finalPageW = p.width * 0.6;
                    let finalPageH = p.height * 0.7;
                    p.drawGlassRect(p.width / 2, p.height / 2, finalPageW, finalPageH, glassBg[3], highlightElement === 'final-page');
                    
                    p.fill(textColor[0], textColor[1], textColor[2]);
                    p.noStroke();
                    p.textAlign(p.CENTER, p.CENTER);
                    p.textSize(28);
                    p.text('Your Web Page', p.width / 2, p.height / 2 - finalPageH / 4);
                    p.textSize(20);
                    p.text('Interactive Content Rendered!', p.width / 2, p.height / 2 + finalPageH / 4);
                    p.pop();
                }
            };

            // Helper function to draw a glassmorphic box (centered)
            p.drawGlassBox = function(x, y, w, h, label, isHighlighted = false) {
                p.rectMode(p.CENTER);
                p.noStroke();

                // Inner glow for highlight
                if (isHighlighted) {
                    p.fill(glowColor[0], glowColor[1], glowColor[2], p.map(p.sin(p.frameCount * 5), -1, 1, 0, glowColor[3]));
                    p.rect(x, y, w + 10, h + 10, 15);
                    p.fill(glowColor[0], glowColor[1], glowColor[2], p.map(p.sin(p.frameCount * 5), -1, 1, 0, glowColor[3] * 0.5));
                    p.rect(x, y, w + 20, h + 20, 20);
                }
                
                // Glass effect
                p.fill(glassBg[0], glassBg[1], glassBg[2], glassBg[3]);
                p.stroke(glassBorder[0], glassBorder[1], glassBorder[2], glassBorder[3]);
                p.strokeWeight(1.5);
                p.rect(x, y, w, h, 10);

                p.fill(textColor);
                p.noStroke();
                p.textAlign(p.CENTER, p.CENTER);
                p.textSize(18);
                p.text(label, x, y);
            };

            // Helper for drawing a general glassmorphic rectangle (centered)
            p.drawGlassRect = function(x, y, w, h, alpha = glassBg[3], isHighlighted = false, fillColor = null) {
                p.rectMode(p.CENTER);
                p.noStroke();

                if (isHighlighted) {
                    p.fill(glowColor[0], glowColor[1], glowColor[2], p.map(p.sin(p.frameCount * 5), -1, 1, 0, glowColor[3]));
                    p.rect(x, y, w + 10, h + 10, 15);
                }

                if (fillColor) {
                    p.fill(fillColor[0], fillColor[1], fillColor[2], fillColor[3]);
                } else {
                    p.fill(glassBg[0], glassBg[1], glassBg[2], alpha);
                }
                p.stroke(glassBorder[0], glassBorder[1], glassBorder[2], glassBorder[3]);
                p.strokeWeight(1.5);
                p.rect(x, y, w, h, 10);
            };

            p.drawGlassNode = function(x, y, label, isHighlighted = false, alpha = glassBg[3]) {
                p.ellipseMode(p.CENTER);
                p.noStroke();
                
                if (isHighlighted) {
                    p.fill(glowColor[0], glowColor[1], glowColor[2], p.map(p.sin(p.frameCount * 5), -1, 1, 0, glowColor[3]));
                    p.ellipse(x, y, 60, 60);
                }

                p.fill(glassBg[0], glassBg[1], glassBg[2], alpha);
                p.stroke(glassBorder[0], glassBorder[1], glassBorder[2], alpha * (glassBorder[3]/glassBg[3])); // Scale border alpha too
                p.strokeWeight(1.5);
                p.ellipse(x, y, 50, 50);

                p.fill(textColor[0], textColor[1], textColor[2], p.min(255, alpha * 3)); // Make text brighter
                p.noStroke();
                p.textAlign(p.CENTER, p.CENTER);
                p.textSize(12);
                p.text(label, x, y);
            };

            p.drawArrow = function(x1, y1, x2, y2, color, weight, alpha = 255) {
                p.stroke(color[0], color[1], color[2], alpha);
                p.strokeWeight(weight);
                p.line(x1, y1, x2, y2);
                p.push();
                p.translate(x2, y2);
                let angle = p.atan2(y2 - y1, x2 - x1);
                p.rotate(angle);
                p.noStroke();
                p.fill(color[0], color[1], color[2], alpha);
                p.triangle(0, weight / 2, 0, -weight / 2, 15, 0);
                p.pop();
            };

            p.drawConnection = function(node1, node2, alpha) {
                p.stroke(textColor[0], textColor[1], textColor[2], alpha);
                p.strokeWeight(1);
                p.line(node1.x, node1.y + 25, node2.x, node2.y - 25);
            };

            // Speech functions
            p.startNarration = function() {
                if ('speechSynthesis' in window) {
                    if (speaking) {
                        speechSynthesis.cancel();
                    }
                    narrationIndex = 0;
                    speakNextLine();
                } else {
                    console.warn("SpeechSynthesis API not supported.");
                }
            };

            function speakNextLine() {
                if (narrationIndex < narrationScript.length) {
                    let line = narrationScript[narrationIndex];
                    let utterance = new SpeechSynthesisUtterance(line.text);

                    utterance.voice = (line.voice === 'male' && maleVoice) ? maleVoice : femaleVoice;
                    utterance.pitch = line.pitch;
                    utterance.rate = line.rate;

                    utterance.onstart = () => {
                        speaking = true;
                        currentStage = line.visualStage; // Update visual stage immediately on start
                        animationProgress = 0; // Reset animation progress for new stage
                        highlightElement = line.highlightElement;
                    };

                    utterance.onend = () => {
                        speaking = false;
                        narrationIndex++;
                        if (narrationIndex < narrationScript.length) {
                            speakNextLine();
                        } else {
                            // End of narration, reset for next podcast or replay
                            isPodcastActive = false;
                            highlightElement = null; // Clear highlight
                            setTimeout(() => p.resetAnimation(), 3000); // Reset animation after narration ends
                        }
                    };

                    utterance.onerror = (event) => {
                        console.error('Speech synthesis error:', event.error);
                        speaking = false;
                    };

                    speechSynthesis.speak(utterance);
                }
            }

            p.windowResized = function() {
                p.resizeCanvas(p.windowWidth, p.windowHeight);
                p.resetAnimation(); // Re-initialize positions based on new canvas size
            };
        };

        let myp5 = new p5(sketch);

        // Button event listeners
        document.getElementById('replay').addEventListener('click', () => {
            myp5.resetAnimation();
            if ('speechSynthesis' in window) {
                speechSynthesis.cancel(); // Ensure narration stops
            }
            myp5.isPodcastActive = false; // Disable podcast mode
        });

        document.getElementById('podcast').addEventListener('click', () => {
            myp5.resetAnimation(); // Reset visuals
            myp5.isPodcastActive = true; // Enable podcast mode
            myp5.startNarration(); // Start narration
        });
    </script>
