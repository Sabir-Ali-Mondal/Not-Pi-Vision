<description>
The "CH3CH2CHCHCHCH2 STRUCTURE" refers to Hexa-1,3-diene, an unsaturated hydrocarbon with the molecular formula C6H10. This molecule is specifically classified as a conjugated diene, characterized by the presence of two carbon-carbon double bonds separated by a single carbon-carbon bond within its chain.

The linear six-carbon backbone of hexa-1,3-diene can be systematically numbered from the end nearest to the first double bond. Following IUPAC nomenclature, the structure is CH2=CH-CH=CH-CH2-CH3, indicating double bonds between carbons 1 and 2 (C1=C2) and between carbons 3 and 4 (C3=C4).

The carbon atoms involved in the double bonds (C1, C2, C3, and C4) exhibit sp2 hybridization, adopting a trigonal planar geometry around each. This planar arrangement allows the unhybridized p-orbitals on these four adjacent sp2 carbons to overlap extensively, forming a continuous, delocalized pi-electron system above and below the plane of the molecule. This conjugation significantly enhances the molecule's stability by spreading electron density over a larger area. The remaining carbons, C5 (methylene) and C6 (methyl), are sp3 hybridized, forming tetrahedral geometries.

While the C1=C2 double bond cannot exhibit geometric isomerism due to the two identical hydrogen atoms on C1, the C3=C4 double bond can exist as cis or trans isomers. Hexa-1,3-diene, like other alkenes, readily undergoes addition reactions (e.g., hydrogenation, halogenation). As a conjugated diene, it is also a key reactant in important organic transformations such as the Diels-Alder cycloaddition, making it a molecule of considerable synthetic utility and scientific interest.

        let animationStage = 0;
        let frameCounter = 0;
        const FPS = 30;

        // Define molecular structure constants
        const BOND_LENGTH = 70; // Pixel length of a bond
        const C_RADIUS = 25;
        const H_RADIUS = 15;

        let atoms = []; // Stores atom objects: {id, type, x, y, r, hybridization, highlight, glow, orbitalRotation, cisTransAngle, parentC}
        let bonds = []; // Stores bond objects: {c1_idx, c2_idx, type}

        function setup() {
            createCanvas(windowWidth, windowHeight).parent(document.body);
            frameRate(FPS);
            pixelDensity(2); // For high-fidelity rendering on high-DPI screens

            // Get voices for narration

                femaleVoice = voices.find(v => v.name.includes('Google US English') && v.name.includes('Female')) || voices.find(v => v.lang === 'en-US' && v.name.includes('Female')) || voices[1] || voices[0];
            };

            initializeMolecule();

        function initializeMolecule() {
            atoms = [];
            bonds = [];
            const startX = -2.5 * BOND_LENGTH; // Leftmost C (C6 in IUPAC / C5 internally)

            // Define carbons (internal indices C0-C5, corresponding to IUPAC C1-C6 from right to left)
            // C0: CH2= (IUPAC C1) - sp2
            // C1: =CH- (IUPAC C2) - sp2
            // C2: -CH= (IUPAC C3) - sp2
            // C3: =CH- (IUPAC C4) - sp2
            // C4: -CH2- (IUPAC C5) - sp3
            // C5: -CH3 (IUPAC C6) - sp3

            const carbonsData = [
                { id: 'C0', x: startX + 5 * BOND_LENGTH, y: 0, type: 'C', hybridization: 'sp2', hydrogens: 2 },
                { id: 'C1', x: startX + 4 * BOND_LENGTH, y: 0, type: 'C', hybridization: 'sp2', hydrogens: 1 },
                { id: 'C2', x: startX + 3 * BOND_LENGTH, y: 0, type: 'C', hybridization: 'sp2', hydrogens: 1 },
                { id: 'C3', x: startX + 2 * BOND_LENGTH, y: 0, type: 'C', hybridization: 'sp2', hydrogens: 1 },
                { id: 'C4', x: startX + 1 * BOND_LENGTH, y: 0, type: 'C', hybridization: 'sp3', hydrogens: 2 },
                { id: 'C5', x: startX + 0 * BOND_LENGTH, y: 0, type: 'C', hybridization: 'sp3', hydrogens: 3 }
            ];

            carbonsData.forEach(c => {
                atoms.push({ id: c.id, type: 'C', x: c.x, y: c.y, r: C_RADIUS, hybridization: c.hybridization, highlight: false, glow: 0, orbitalRotation: 0, cisTransAngle: 0 });
            });

            // Define carbon-carbon bonds (c1_idx, c2_idx refer to internal atoms array index)
            bonds.push({ c1_idx: 5, c2_idx: 4, type: 'single' }); // C6-C5 (IUPAC) / C5-C4 (internal)
            bonds.push({ c1_idx: 4, c2_idx: 3, type: 'single' }); // C5-C4 (IUPAC) / C4-C3 (internal)
            bonds.push({ c1_idx: 3, c2_idx: 2, type: 'double' }); // C4=C3 (IUPAC) / C3=C2 (internal)
            bonds.push({ c1_idx: 2, c2_idx: 1, type: 'single' }); // C3-C2 (IUPAC) / C2-C1 (internal)
            bonds.push({ c1_idx: 1, c2_idx: 0, type: 'double' }); // C2=C1 (IUPAC) / C1=C0 (internal)

            // Add hydrogens around their parent carbons
            let h_idx_counter = 0;
            atoms.filter(a => a.type === 'C').forEach((c, idx) => {
                const numHydrogens = carbonsData[idx].hydrogens;
                for (let i = 0; i < numHydrogens; i++) {
                    let hX = c.x, hY = c.y;
                    let angle_offset = 0; // Base angle for the hydrogen placement
                    const h_dist = C_RADIUS + H_RADIUS * 0.8; // Distance from carbon center

                    if (c.hybridization === 'sp3') { // Tetrahedral approximation
                        if (c.id === 'C5') { // CH3 (IUPAC C6)
                            angle_offset = PI/2; // Start roughly above
                            let angle = angle_offset + map(i, 0, 2, -PI/3, PI/3);
                            hX = c.x + h_dist * cos(angle);
                            hY = c.y + h_dist * sin(angle);
                        } else { // CH2 (IUPAC C5)
                            angle_offset = PI/2;
                            let angle = angle_offset + map(i, 0, 1, -PI/4, PI/4);
                            hX = c.x + h_dist * cos(angle);
                            hY = c.y + h_dist * sin(angle);
                        }
                    } else { // sp2 (Trigonal Planar approximation)
                        if (c.id === 'C0') { // CH2= (IUPAC C1)
                            angle_offset = 0; // Towards right of molecule
                            let angle = angle_offset + map(i, 0, 1, -PI/3, PI/3);
                            hX = c.x + h_dist * cos(angle);
                            hY = c.y + h_dist * sin(angle);
                        } else if (c.id === 'C1' || c.id === 'C3') { // =CH- (IUPAC C2, C4)
                            angle_offset = PI * 0.75; // Towards upper-left
                            hX = c.x + h_dist * cos(angle_offset);
                            hY = c.y + h_dist * sin(angle_offset);
                        } else if (c.id === 'C2') { // -CH= (IUPAC C3)
                            angle_offset = PI * 0.25; // Towards upper-right
                            hX = c.x + h_dist * cos(angle_offset);
                            hY = c.y + h_dist * sin(angle_offset);
                        }
                    }
                    atoms.push({ id: `H${h_idx_counter++}`, type: 'H', x: hX, y: hY, r: H_RADIUS, parentC: c.id, highlight: false, glow: 0 });
                }
            });
        }

        function draw() {

            translate(width / 2, height / 2); // Center the drawing

            switch (animationStage) {
                case 0: // Initial state / Title screen

                    drawText("Press 'Podcast' to start", 0, 50, 20);
                    break;

                case 1: // Molecular Formula and Name

                    break;

                case 2: // Carbon Backbone
                    drawAtoms(atoms.filter(a => a.type === 'C'));
                    drawBonds(bonds, 0); // Only carbons connected
                    break;

                case 3: // Full Structure: Carbons, Hydrogens, Bonds
                    drawAtoms(atoms);
                    drawBonds(bonds, 1); // Draw all bonds including C-H
                    break;

                case 4: // Hybridization (sp2 vs sp3)
                    drawAtoms(atoms);
                    drawBonds(bonds, 1);
                    // Highlight sp2 carbons and indicate planar geometry
                    atoms.forEach(atom => {
                        if (atom.type === 'C' && atom.hybridization === 'sp2') {
                            drawGlow(atom.x, atom.y, atom.r * 2.5, getComputedStyle(document.documentElement).getPropertyValue('--glow-color'), atom.glow);
                            // Draw a trigonal planar indicator (a dashed triangle)
                            push();
                            stroke(getComputedStyle(document.documentElement).getPropertyValue('--primary-color'));
                            strokeWeight(2);
                            drawingContext.setLineDash([5, 5]); // Dashed line
                            noFill();
                            translate(atom.x, atom.y);
                            // Orient the triangle along the bond
                            let neighborC_idx = (atom.id === 'C0' || atom.id === 'C2') ? atom.id === 'C0' ? 1 : 3 : (atom.id === 'C1' || atom.id === 'C3') ? atom.id === 'C1' ? 0 : 2 : -1;
                            if (neighborC_idx !== -1) {
                                let neighborC = atoms[neighborC_idx];
                                let angle = atan2(neighborC.y - atom.y, neighborC.x - atom.x);
                                rotate(angle); // Align triangle with bond direction
                            }
                            triangle(C_RADIUS * 1.5, 0, C_RADIUS * cos(2 * PI / 3), C_RADIUS * sin(2 * PI / 3), C_RADIUS * cos(4 * PI / 3), C_RADIUS * sin(4 * PI / 3));
                            drawingContext.setLineDash([]); // Reset line dash
                            pop();
                        }
                    });
                    break;

                case 5: // Conjugation (p-orbitals and pi-cloud)
                    drawAtoms(atoms);
                    drawBonds(bonds, 1);
                    // Draw p-orbitals on sp2 carbons
                    atoms.forEach(atom => {
                        if (atom.type === 'C' && atom.hybridization === 'sp2') {
                            drawPOribital(atom.x, atom.y, atom.r * 2, atom.orbitalRotation, getComputedStyle(document.documentElement).getPropertyValue('--glow-color'));
                        }
                    });
                    // Draw delocalized pi cloud (simplified) - for C0=C1-C2=C3 chain (internal indices)
                    const conjugatedCarbons = [atoms[0], atoms[1], atoms[2], atoms[3]];
                    if (conjugatedCarbons.length === 4) {
                        let minX = conjugatedCarbons[0].x;
                        let maxX = conjugatedCarbons[3].x;
                        drawPiCloud(minX, maxX, conjugatedCarbons[0].y, atoms[0].highlight ? 255 : 100);
                    }
                    break;

                case 6: // Isomerism (cis/trans) on C3=C4 bond (internal C2=C3)
                    drawAtoms(atoms);
                    drawBonds(bonds, 1);

                    // Simulate a subtle back-and-forth movement for cis/trans illustration
                    let c2 = atoms[2]; // internal C2, IUPAC C3
                    let c3 = atoms[3]; // internal C3, IUPAC C4

                    if (c2 && c3) {
                        // Apply rotation to the atoms that would change position in cis/trans
                        // This is a simplified visual to imply the change, not a precise model
                        push();
                        translate(c2.x, c2.y);
                        rotate(c2.cisTransAngle);
                        drawAtomPart(c2); // Redraw the atom with the rotation
                        pop();

                        push();
                        translate(c3.x, c3.y);
                        rotate(c3.cisTransAngle);
                        drawAtomPart(c3); // Redraw the atom with the rotation
                        pop();

                        // Redraw hydrogens and bonds attached to C2 and C3 with this transformation
                        atoms.filter(a => a.type === 'H' && (a.parentC === c2.id || a.parentC === c3.id)).forEach(h => {
                            let parent = atoms.find(a => a.id === h.parentC);
                            if (parent) {
                                push();
                                translate(parent.x, parent.y);
                                rotate(parent.cisTransAngle);
                                stroke(getComputedStyle(document.documentElement).getPropertyValue('--text-color'));
                                strokeWeight(2);
                                // Adjust H position relative to parent C, then draw
                                let origHX = h.x - parent.x;
                                let origHY = h.y - parent.y;
                                line(0, 0, origHX, origHY);
                                drawAtomPart(h, origHX, origHY);
                                pop();
                            }
                        });

                        // Draw bonds C2=C3 (double bond)
                        drawBond(atoms[2], atoms[3], 'double');
                    }
                    break;

                case 7: // Reactivity / Final Stable Form
                    drawAtoms(atoms);
                    drawBonds(bonds, 1);
                    // Subtle pulse on the entire molecule
                    let pulseScale = map(sin(frameCount * 0.05), -1, 1, 0.98, 1.02);
                    scale(pulseScale);
                    break;
            }

            frameCounter++;
        }

        // --- Drawing Helper Functions ---
        function drawAtom(atom) {
            push();
            fill(atom.type === 'C' ? '#B0B0B0' : '#E0E0E0'); // Carbon: grey, Hydrogen: light grey
            stroke(atom.highlight ? getComputedStyle(document.documentElement).getPropertyValue('--primary-color') : 'none');
            strokeWeight(atom.highlight ? 3 : 0);
            ellipse(atom.x, atom.y, atom.r * 2, atom.r * 2);

            if (atom.glow > 0) {
                drawGlow(atom.x, atom.y, atom.r * 2.5, getComputedStyle(document.documentElement).getPropertyValue('--glow-color'), atom.glow);
            }

            fill(getComputedStyle(document.documentElement).getPropertyValue('--bg-color'));
            noStroke();
            textSize(atom.r * 0.8);
            textAlign(CENTER, CENTER);
            text(atom.type, atom.x, atom.y);
            pop();
        }

        // Used for drawing a part of an atom during transformations (e.g., cis/trans)
        function drawAtomPart(atom, x = 0, y = 0) {
            fill(atom.type === 'C' ? '#B0B0B0' : '#E0E0E0');
            noStroke();
            ellipse(x, y, atom.r * 2, atom.r * 2);
            fill(getComputedStyle(document.documentElement).getPropertyValue('--bg-color'));
            textSize(atom.r * 0.8);
            textAlign(CENTER, CENTER);
            text(atom.type, x, y);
        }

        function drawAtoms(atomList) {
            atomList.forEach(atom => drawAtom(atom));
        }

        function drawBond(c1, c2, type) {
            stroke(getComputedStyle(document.documentElement).getPropertyValue('--text-color'));
            strokeWeight(4);

            let offset = 0;
            if (type === 'double') {
                offset = 3;
            }

            const angle = atan2(c2.y - c1.y, c2.x - c1.x);
            const dx = cos(angle);
            const dy = sin(angle);

            const startX = c1.x + C_RADIUS * dx;
            const startY = c1.y + C_RADIUS * dy;
            const endX = c2.x - C_RADIUS * dx;
            const endY = c2.y - C_RADIUS * dy;

            if (type === 'single') {
                line(startX, startY, endX, endY);
            } else if (type === 'double') {
                line(startX - dy * offset, startY + dx * offset, endX - dy * offset, endY + dx * offset);
                line(startX + dy * offset, startY - dx * offset, endX + dy * offset, endY - dx * offset);
            }
        }

        function drawBonds(bondList, stage) {
            bondList.forEach(bond => {
                let c1_atom = atoms[bond.c1_idx];
                let c2_atom = atoms[bond.c2_idx];

                if (c1_atom && c2_atom && bond.type !== 'H_bond') { // Exclude H_bond type here if I add it
                    drawBond(c1_atom, c2_atom, bond.type);
                }
            });

            // Draw C-H bonds dynamically for any stage >= 1
            if (stage >= 1) {
                atoms.filter(a => a.type === 'H').forEach(h_atom => {
                    let parentC = atoms.find(a => a.id === h_atom.parentC);
                    if (parentC) {
                        stroke(getComputedStyle(document.documentElement).getPropertyValue('--text-color'));
                        strokeWeight(2);
                        line(parentC.x, parentC.y, h_atom.x, h_atom.y);
                    }
                });
            }
        }

        function drawText(txt, x, y, size, color = getComputedStyle(document.documentElement).getPropertyValue('--text-color')) {
            fill(color);
            textSize(size);
            textAlign(CENTER, CENTER);
            text(txt, x, y);
        }

        function drawGlow(x, y, radius, color, alpha) {
            for (let i = 0; i < 5; i++) {
                let currentAlpha = map(i, 0, 4, 0, alpha);
                fill(red(color), green(color), blue(color), currentAlpha);
                noStroke();
                ellipse(x, y, radius + i * 5, radius + i * 5);
            }
        }

        function drawPOribital(x, y, size, rotation, color) {
            push();
            translate(x, y);
            rotate(rotation);
            noStroke();
            fill(red(color), green(color), blue(color), 150); // Semi-transparent color
            ellipse(0, -size / 2, size / 2, size);
            ellipse(0, size / 2, size / 2, size);
            pop();
        }

        function drawPiCloud(startX, endX, y, alpha) {
            push();
            noStroke();
            fill(red(getComputedStyle(document.documentElement).getPropertyValue('--glow-color')), green(getComputedStyle(document.documentElement).getPropertyValue('--glow-color')), blue(getComputedStyle(document.documentElement).getPropertyValue('--glow-color')), alpha);

            let centerX = (startX + endX) / 2;
            let width = endX - startX;
            let height = C_RADIUS * 3; // Approx height of pi cloud

            ellipse(centerX, y - height / 2, width + C_RADIUS, height);
            ellipse(centerX, y + height / 2, width + C_RADIUS, height);
            pop();
        }

        function startNarration() {
            if (narrationPlaying) {
                speechSynthesis.cancel();
            }
            narrationPlaying = true;
            narrationStep = 0;
            speakNextNarrationStep();
        }

        function speakNextNarrationStep() {
            if (narrationStep < narrationScript.length) {
                const step = narrationScript[narrationStep];
                const utterance = new SpeechSynthesisUtterance(step.text);

                utterance.voice = (step.voice === 'male' && maleVoice) ? maleVoice : (step.voice === 'female' && femaleVoice) ? femaleVoice : voices[0];
                utterance.pitch = step.pitch;
                utterance.rate = step.rate;

                utterance.onstart = () => {
                    animationStage = step.visuals.stage;
                    applyVisualHighlights(step.visuals.highlight);
                };

                utterance.onend = () => {
                    setTimeout(() => {
                        narrationStep++;
                        speakNextNarrationStep();
                    }, step.delay); // Add delay between sentences
                };

                speechSynthesis.speak(utterance);
            } else {
                narrationPlaying = false;
                applyVisualHighlights(null); // Clear all highlights at the end
            }
        }

        function applyVisualHighlights(highlightType) {
            atoms.forEach(atom => {
                atom.highlight = false;
                atom.glow = 0;
                atom.orbitalRotation = 0;
                atom.cisTransAngle = 0;
            });

            if (highlightType === 'formula') {
                atoms[0].highlight = true; // Dummy highlight for stage 1 text
            } else if (highlightType === 'C_backbone') {
                atoms.filter(a => a.type === 'C').forEach(c => c.highlight = true);
            } else if (highlightType === 'double_bonds') {
                // Highlight the carbons in the double bonds: C0, C1, C2, C3 (internal indices)
                atoms[0].highlight = true; atoms[1].highlight = true;
                atoms[2].highlight = true; atoms[3].highlight = true;
            } else if (highlightType === 'sp2_carbons') {
                atoms.filter(a => a.type === 'C' && a.hybridization === 'sp2').forEach(c => c.glow = 150);
            } else if (highlightType === 'p_orbitals' || highlightType === 'pi_cloud') {
                atoms.filter(a => a.type === 'C' && a.hybridization === 'sp2').forEach(c => {
                    c.orbitalRotation = frameCount * 0.05; // Animate p-orbitals
                    c.glow = 150;
                });
            } else if (highlightType === 'isomerism') {
                // Apply a fluctuating rotation to the relevant carbons for cis/trans illustration
                // C2 and C3 are the carbons of the C3=C4 bond (IUPAC)
                let c2 = atoms[2];
                let c3 = atoms[3];
                if (c2 && c3) {
                    c2.cisTransAngle = map(sin(frameCount * 0.1), -1, 1, -PI / 8, PI / 8);
                    c3.cisTransAngle = map(sin(frameCount * 0.1), -1, 1, PI / 8, -PI / 8);
                }
            } else if (highlightType === 'reactivity') {
                atoms.forEach(atom => {
                    atom.glow = map(sin(frameCount * 0.1), -1, 1, 50, 100);
                });
            }
            // 'final_structure' will simply clear all highlights and maintain the stable view.
        }

        function resetAnimation() {
            speechSynthesis.cancel();
            narrationPlaying = false;
            animationStage = 0;
            narrationStep = 0;
            frameCounter = 0;
            initializeMolecule(); // Reset molecule positions/states
            applyVisualHighlights(null); // Clear all highlights
        }
