<description>
ताजमहल, जो कि एक अद्वतीय स्थापत्य कला का उदाहरण है, केवल एक इमारत नहीं बल्कि विभिन्न वैज्ञानिक सिद्धांतों का एक उत्कृष्ट संगम है। इसकी नींव, जो यमुना नदी के नरम तलछट पर आधारित है, एक जटिल इंजीनियरिंग उपलब्धि है। प्रारंभिक काल में, इसमें विशाल लकड़ी के लट्ठों (सैकड़ों मीटर गहरे कुओं में डाले गए) का उपयोग किया गया था, जो जल-संतृप्त होने पर अपनी शक्ति बनाए रखते थे। आधुनिक विश्लेषण बताते हैं कि नदी के जल स्तर में उतार-चढ़ाव इसकी दीर्घकालिक स्थिरता के लिए एक संभावित चुनौती प्रस्तुत करता है, जिसके लिए निरंतर निगरानी और संरक्षण की आवश्यकता है।

संरचनात्मक रूप से, ताजमहल में एक 'डबल-डोम' प्रणाली का उपयोग किया गया है। यह प्रणाली बाहरी भव्यता और आंतरिक स्थान के बीच संतुलन स्थापित करती है, साथ ही भूकम्प जैसी प्राकृतिक आपदाओं से बचाव में भी सहायक होती है। इसकी मीनारें, जो मुख्य मकबरे से थोड़ी बाहर की ओर झुकी हुई हैं, एक जानबूझकर की गई अभियांत्रिकी युक्ति है ताकि किसी भी पतन की स्थिति में वे मकबरे से दूर गिरें।

सामग्री विज्ञान की दृष्टि से, इसमें प्रयुक्त सफेद संगमरमर (मकरान से प्राप्त), जो उच्च शुद्धता का कैल्साइट है, न केवल अपनी सौंदर्य क्षमता के लिए चुना गया था बल्कि इसकी थर्मल स्थिरता और परावर्तक गुणों के लिए भी। दिन के विभिन्न प्रहरों और मौसमों में इसकी सतह का रंग बदलना प्रकाशिकी और पदार्थ के बीच जटिल परस्पर क्रिया का परिणाम है। संगमरमर में जड़ी 'पिएत्रा ड्यूरा' कलाकृति, जिसमें हजारों कीमती और अर्ध-कीमती पत्थरों को जटिल ज्यामितीय और पुष्प पैटर्न में सावधानीपूर्वक काटा और फिट किया गया है, सामग्री की कठोरता और सूक्ष्म कारीगरी का प्रमाण है।

दृष्टिगत भ्रम (Optical Illusions) भी इसके डिज़ाइन का अभिन्न अंग हैं। सुलेख (calligraphy) को इस तरह से स्केल किया गया है कि ऊपर से भी अक्षर समान आकार के दिखें, जो परिप्रेक्ष्य सुधार का एक परिष्कृत अनुप्रयोग है। उद्यान और जल प्रणालियाँ, जैसे कि चारबाग योजना और फव्वारे, केवल सौंदर्य के लिए नहीं थे; वे ताप को अवशोषित करके और वाष्पीकरण द्वारा वातावरण को ठंडा करके एक प्रभावी सूक्ष्म-जलवायु नियंत्रण प्रणाली के रूप में भी कार्य करते थे। ताजमहल की वास्तुकला माप की सटीकता, समरूपता और सामग्री के चतुर उपयोग का एक वैज्ञानिक अध्ययन है जो इसे सदियों से एक इंजीनियरिंग चमत्कार बनाए हुए है।
</description>

        function setup() {

            p5Canvas.parent(document.body); // Attach canvas to body
            frameRate(30);
            slideStartTime = millis();

            // Load voices
            speechSynthesis.onvoiceschanged = () => {
                availableVoices = speechSynthesis.getVoices();
                // Try to find Hindi voices first, then general male/female, otherwise fallback
                maleVoice = availableVoices.find(voice => voice.lang === 'hi-IN' && voice.name.includes('Male')) ||

        function draw() {
            background(getComputedStyle(document.documentElement).getPropertyValue('--bg-color'));

            let timeElapsed = millis() - slideStartTime;
            let slideProgress = timeElapsed / 1000; // in seconds

            drawVisuals(currentSlide, slideProgress);
        }

        function windowResized() {
            resizeCanvas(windowWidth * 0.8, windowHeight * 0.8);

        function drawVisuals(slide, progress) {
            push();
            translate(width / 2, height / 2); // Center everything

            const primaryColor = color(getComputedStyle(document.documentElement).getPropertyValue('--primary-color'));
            const textColor = color(getComputedStyle(document.documentElement).getPropertyValue('--text-color'));
            const glowColor = color(getComputedStyle(document.documentElement).getPropertyValue('--glow-color'));

            noFill();
            stroke(textColor);
            strokeWeight(2);
            textAlign(CENTER, CENTER);

            switch (slide) {
                case 0: // Intro: Title and abstract Taj outline
                    if (progress < 2) { // Fade in
                        let alpha = map(progress, 0, 2, 0, 255);
                        stroke(red(textColor), green(textColor), blue(textColor), alpha);
                        fill(red(textColor), green(textColor), blue(textColor), alpha);
                    } else {
                        stroke(textColor);
                        fill(textColor);
                    }
                    textSize(width * 0.05);
                    text("ताजमहल: वैज्ञानिक दृष्टिकोण", 0, -height * 0.2);

                    let outlineScale = map(sin(progress * 0.5), -1, 1, 0.8, 1.2);
                    let outlineAlpha = map(progress, 0, 5, 0, 255, true);

                    push();
                    scale(outlineScale);
                    stroke(red(primaryColor), green(primaryColor), blue(primaryColor), outlineAlpha);
                    strokeWeight(3);
                    // Abstract Taj Mahal outline
                    rect(-width * 0.1, -height * 0.1, width * 0.2, height * 0.2); // Base
                    arc(0, -height * 0.1, width * 0.15, height * 0.15, PI, TWO_PI); // Dome
                    line(-width * 0.15, height * 0.05, -width * 0.15, height * 0.1); // Minaret 1
                    line(width * 0.15, height * 0.05, width * 0.15, height * 0.1); // Minaret 2
                    pop();
                    break;

                case 1: // Foundation: Wells and piles
                    let baseHeight = height * 0.2;
                    let baseY = height * 0.25 - baseHeight / 2;

                    // Riverbed representation
                    fill(40, 60, 80);
                    noStroke();
                    rect(-width / 2, baseY + baseHeight / 2, width, height / 2 - (baseY + baseHeight / 2));

                    // Base structure
                    noFill();
                    stroke(textColor);
strokeWeight(3);
                    rect(-width * 0.15, -baseHeight / 2, width * 0.3, baseHeight);

                    // Piles animation
                    let numPiles = 10;
                    let pileSpacing = (width * 0.3) / (numPiles + 1);
                    let pileLength = map(progress, 0, 3, 0, height * 0.3, true);

                    for (let i = 0; i < numPiles; i++) {
                        let x = -width * 0.15 + pileSpacing * (i + 1);
                        let y = baseHeight / 2;
                        stroke(lerpColor(textColor, glowColor, sin(progress * 2 + i * 0.5) * 0.5 + 0.5));
                        line(x, y, x, y + pileLength);
                        fill(lerpColor(textColor, glowColor, sin(progress * 2 + i * 0.5) * 0.5 + 0.5));
                        ellipse(x, y + pileLength, 5, 5); // Represents end of pile/well
                    }

                    // Water level fluctuation indicator
                    if (progress > 3) {
                        let waterAlpha = map(progress, 3, 6, 0, 100, true);
                        fill(0, 100, 200, waterAlpha);
                        noStroke();
                        let waterLevel = map(sin(progress * 0.8), -1, 1, 0, height * 0.05);
                        rect(-width / 2, baseY + baseHeight / 2 + waterLevel, width, height * 0.01);
                    }
                    break;

                case 2: // Double Dome and Leaning Minarets
                    // Double Dome Cross-section
                    let domeInnerRadius = width * 0.08;
                    let domeOuterRadius = width * 0.12;
                    let domeHeight = height * 0.2;

                    stroke(textColor);
                    strokeWeight(3);
                    noFill();

                    // Inner dome
                    arc(0, -domeHeight / 2, domeInnerRadius * 2, domeHeight * 2, PI, TWO_PI);
                    // Outer dome
                    arc(0, -domeHeight / 2, domeOuterRadius * 2, domeHeight * 2, PI, TWO_PI);

                    // Highlight based on progress
                    if (progress > 0 && progress < 3) { // Inner dome highlight
                        stroke(glowColor);
                        strokeWeight(5);
                        arc(0, -domeHeight / 2, domeInnerRadius * 2, domeHeight * 2, PI, TWO_PI);
                    } else if (progress > 3 && progress < 6) { // Outer dome highlight
                        stroke(glowColor);
                        strokeWeight(5);
                        arc(0, -domeHeight / 2, domeOuterRadius * 2, domeHeight * 2, PI, TWO_PI);
                    }

                    // Minarets
                    let minaretBaseY = height * 0.15;
                    let minaretHeight = height * 0.3;
                    let minaretWidth = width * 0.02;
                    let tiltAngle = map(progress, 6, 9, 0, PI / 40, true); // Max tilt of 4.5 degrees (PI/40)

                    push();
                    translate(-width * 0.25, minaretBaseY); // Left minaret
                    rotate(-tiltAngle);
                    rect(-minaretWidth / 2, -minaretHeight, minaretWidth, minaretHeight);
                    pop();

                    push();
                    translate(width * 0.25, minaretBaseY); // Right minaret
                    rotate(tiltAngle);
                    rect(-minaretWidth / 2, -minaretHeight, minaretWidth, minaretHeight);
                    pop();

                    // Highlight minarets
                    if (progress > 6) {
                        push();
                        translate(-width * 0.25, minaretBaseY);
                        rotate(-tiltAngle);
                        stroke(glowColor);
                        strokeWeight(5);
                        rect(-minaretWidth / 2, -minaretHeight, minaretWidth, minaretHeight);
                        pop();

                        push();
                        translate(width * 0.25, minaretBaseY);
                        rotate(tiltAngle);
                        stroke(glowColor);
                        strokeWeight(5);
                        rect(-minaretWidth / 2, -minaretHeight, minaretWidth, minaretHeight);
                        pop();
                    }
                    break;

                case 3: // Materials Science: Marble and Pietra Dura
                    let marbleSize = width * 0.3;
                    let marbleX = 0;
                    let marbleY = 0;

                    // Main Marble Block
                    fill(240, 240, 240);
                    stroke(textColor);
                    strokeWeight(2);
                    rect(marbleX - marbleSize / 2, marbleY - marbleSize / 2, marbleSize, marbleSize, 10);

                    // Changing color with light (abstract)
                    if (progress > 2 && progress < 5) {
                        let lightHue = map(sin(progress * 0.8), -1, 1, 0, 60); // Yellow to orange hue
                        let lightSaturation = 50;
                        let lightBrightness = map(sin(progress * 1.5), -1, 1, 80, 100);
                        colorMode(HSB, 360, 100, 100);
                        fill(lightHue, lightSaturation, lightBrightness, 0.8);
                        rect(marbleX - marbleSize / 2, marbleY - marbleSize / 2, marbleSize, marbleSize, 10);
                        colorMode(RGB, 255, 255, 255); // Reset color mode
                    }

                    // Pietra Dura Inlay animation
                    if (progress > 5) {
                        let inlayProgress = map(progress, 5, 8, 0, 1, true);
                        let patternSize = marbleSize * 0.05;
                        let startX = marbleX - marbleSize / 2 + patternSize / 2;
                        let startY = marbleY - marbleSize / 2 + patternSize / 2;

                        for (let x = 0; x < marbleSize / patternSize - 1; x++) {
                            for (let y = 0; y < marbleSize / patternSize - 1; y++) {
                                let rectX = startX + x * patternSize;
                                let rectY = startY + y * patternSize;
                                let distFromCenter = dist(rectX, rectY, marbleX, marbleY);
                                let delay = map(distFromCenter, 0, marbleSize / 2, 0, 0.5);

                                if (inlayProgress > delay) {
                                    let fillAlpha = map(inlayProgress, delay, delay + 0.5, 0, 255, true);
                                    let c = color(map(x + y, 0, 10, 0, 255), map(x, 0, 5, 0, 255), map(y, 0, 5, 0, 255), fillAlpha);
                                    fill(c);
                                    noStroke();
                                    rect(rectX, rectY, patternSize, patternSize);
                                    stroke(textColor);
                                    strokeWeight(1);
                                    noFill();
                                    rect(rectX, rectY, patternSize, patternSize);
                                }
                            }
                        }
                    }

                    if (progress > 5) { // Highlight the inlaid area
                        noFill();
                        stroke(glowColor);
                        strokeWeight(5);
                        rect(marbleX - marbleSize / 2, marbleY - marbleSize / 2, marbleSize, marbleSize, 10);
                    }
                    break;

                case 4: // Optical Illusions & Symmetry
                    // Symmetry Grid
                    stroke(textColor);
                    strokeWeight(1);
                    line(0, -height / 2, 0, height / 2); // Vertical symmetry axis
                    line(-width / 2, 0, width / 2, 0);   // Horizontal symmetry axis

                    if (progress < 4) { // Demonstrate symmetry with a simple shape
                        let shapeSize = width * 0.15;
                        let animProgress = map(progress, 0, 3, 0, 1, true);
                        let xOffset = map(animProgress, 0, 1, -width * 0.2, -width * 0.1);
                        let alpha = map(animProgress, 0, 1, 0, 255, true);

                        fill(red(primaryColor), green(primaryColor), blue(primaryColor), alpha);
                        ellipse(xOffset, 0, shapeSize, shapeSize); // Left half
                        fill(red(primaryColor), green(primaryColor), blue(primaryColor), alpha);
                        ellipse(-xOffset, 0, shapeSize, shapeSize); // Right half (symmetrical)

                        // Highlight symmetry axis
                        stroke(glowColor);
                        strokeWeight(4);
                        line(0, -height / 2, 0, height / 2);
                    }

                    // Calligraphy illusion
                    if (progress > 4) {
                        let textY = -height * 0.15;
                        let textHeight = height * 0.05;
                        let textAlpha = map(progress, 4, 6, 0, 255, true);

                        textSize(textHeight);
                        fill(red(textColor), green(textColor), blue(textColor), textAlpha);
                        noStroke();
                        text("بسم الله الرحمن الرحيم", 0, textY); // Example Arabic calligraphy

                        if (progress > 6) { // Illusion explanation
                            textSize(textHeight * 0.6);
                            fill(red(glowColor), green(glowColor), blue(glowColor), textAlpha);
                            text("उच्च दूरी पर अक्षरों का समान आकार", 0, textY + textHeight * 1.5);
                            
                            // Visual representation of perspective correction
                            let correctionProgress = map(progress, 6, 9, 0, 1, true);
                            stroke(glowColor);
                            strokeWeight(3);
                            noFill();

                            // Draw two lines, one straight, one curved to represent perceived vs actual size
                            line(-width * 0.1, textY + textHeight * 2.5, width * 0.1, textY + textHeight * 2.5);
                            
                            // Animated curve showing correction
                            beginShape();
                            for(let i = 0; i <= 10; i++) {
                                let x = map(i, 0, 10, -width * 0.1, width * 0.1);
                                let yCurve = textY + textHeight * 2.5 + sin(i / 10 * PI) * (height * 0.02 * correctionProgress);
                                vertex(x, yCurve);
                            }
                            endShape();
                        }
                    }
                    break;

                case 5: // Hydrology & Climate Control: Gardens and water system
                    let gardenSize = width * 0.6;
                    let charbaghX = -gardenSize / 2;
                    let charbaghY = -gardenSize / 2;

                    noFill();
                    stroke(textColor);
                    strokeWeight(3);
                    rect(charbaghX, charbaghY, gardenSize, gardenSize); // Charbagh outline

                    // Water channels
                    let channelWidth = width * 0.01;
                    fill(0, 100, 200, 150);
                    noStroke();

                    rect(charbaghX, charbaghY + gardenSize / 2 - channelWidth / 2, gardenSize, channelWidth); // Horizontal channel
                    rect(charbaghX + gardenSize / 2 - channelWidth / 2, charbaghY, channelWidth, gardenSize); // Vertical channel

                    // Fountains animation
                    let fountain1X = charbaghX + gardenSize / 4;
                    let fountain1Y = charbaghY + gardenSize / 4;
                    let fountain2X = charbaghX + gardenSize * 3 / 4;
                    let fountain2Y = charbaghY + gardenSize / 4;
                    let fountain3X = charbaghX + gardenSize / 4;
                    let fountain3Y = charbaghY + gardenSize * 3 / 4;
                    let fountain4X = charbaghX + gardenSize * 3 / 4;
                    let fountain4Y = charbaghY + gardenSize * 3 / 4;

                    drawFountain(fountain1X, fountain1Y, progress);
                    drawFountain(fountain2X, fountain2Y, progress);
                    drawFountain(fountain3X, fountain3Y, progress);
                    drawFountain(fountain4X, fountain4Y, progress);

                    // Cooling effect (abstract particles)
                    if (progress > 3) {
                        let particleAlpha = map(progress, 3, 6, 0, 150, true);
                        for (let i = 0; i < 50; i++) {
                            let pX = random(charbaghX, charbaghX + gardenSize);
                            let pY = random(charbaghY, charbaghY + gardenSize);
                            let particleMove = map(progress, 3, 6, 0, -height * 0.1, true);
                            fill(150, 200, 255, particleAlpha);
                            noStroke();
                            ellipse(pX, pY + particleMove, 5, 5);
                        }
                    }
                    break;
                case 6: // Conclusion / Environmental Interaction: Changing colors
                    let buildingWidth = width * 0.4;
                    let buildingHeight = height * 0.4;
                    
                    // Main structure
                    fill(240, 240, 240);
                    stroke(textColor);
                    strokeWeight(3);
                    rect(-buildingWidth / 2, -buildingHeight / 2, buildingWidth, buildingHeight, 10);
                    arc(0, -buildingHeight / 2, buildingWidth * 0.8, buildingHeight * 0.8, PI, TWO_PI); // Dome

                    // Dynamic light source animation
                    let lightCycle = sin(progress * 0.5); // Slow cycle
                    let r = map(lightCycle, -1, 1, 200, 255);
                    let g = map(lightCycle, -1, 1, 200, 240);
                    let b = map(lightCycle, -1, 1, 255, 200);
                    let alpha = map(lightCycle, -1, 1, 100, 200);

                    // Apply overlay to simulate color change
                    fill(r, g, b, alpha);
                    rect(-buildingWidth / 2, -buildingHeight / 2, buildingWidth, buildingHeight, 10);
                    arc(0, -buildingHeight / 2, buildingWidth * 0.8, buildingHeight * 0.8, PI, TWO_PI);
                    
                    // Sun/Moon path (abstract)
                    let sunMoonRadius = width * 0.3;
                    let angle = map(progress, 0, 10, 0, TWO_PI); // Full circle over 10 seconds
                    let sunMoonX = sunMoonRadius * cos(angle);
                    let sunMoonY = sunMoonRadius * sin(angle) - height * 0.2; // Offset for better visual

                    fill(255, 255, 0); // Sun
                    if (sin(angle) < 0) { // Below horizon (night)
                        fill(150, 150, 255); // Moon
                    }
                    noStroke();
                    ellipse(sunMoonX, sunMoonY, width * 0.05, width * 0.05);

                    // Text for conclusion
                    if (progress > 5) {
                        let textAlpha = map(progress, 5, 8, 0, 255, true);
                        fill(red(textColor), green(textColor), blue(textColor), textAlpha);
                        textSize(width * 0.03);
                        text("एक चिरस्थायी इंजीनियरिंग चमत्कार", 0, height * 0.25);
                    }
                    break;
            }
            pop();
        }

        function drawFountain(x, y, progress) {
            let fountainBaseRadius = 10;
            fill(0, 100, 200, 180);
            noStroke();
            ellipse(x, y, fountainBaseRadius * 2, fountainBaseRadius);

            let sprayHeight = map(sin(progress * 5 + x * 0.01 + y * 0.01), -1, 1, 0, height * 0.05);
            let sprayParticles = 10;
            for (let i = 0; i < sprayParticles; i++) {
                let px = x + random(-5, 5);
                let py = y - random(0, sprayHeight);
                let particleSize = random(2, 5);
                ellipse(px, py, particleSize, particleSize);
            }
        }

        // --- Narration Controls ---
        function speakCurrentNarration() {
            if (currentNarrationStepIndex >= narrationSteps.length) {
                stopNarration();
                return;
            }

            const step = narrationSteps[currentNarrationStepIndex];
            const utterance = new SpeechSynthesisUtterance(step.text);
            utterance.pitch = step.pitch;
            utterance.rate = step.rate;

            if (availableVoices.length > 0) {
                // Alternate between male and female voices based on voiceIndex property in narrationSteps
                if (step.voiceIndex === 0) { // Male voice
                    utterance.voice = maleVoice;
                } else { // Female voice
                    utterance.voice = femaleVoice;
                }
            }

            utterance.onend = () => {
                currentNarrationStepIndex++;
                speakCurrentNarration(); // Speak next
            };
            utterance.onerror = (event) => {
                console.error('SpeechSynthesisUtterance.onerror', event);
                // Try to continue if there's an error
                currentNarrationStepIndex++;
                speakCurrentNarration();
            };

            speechSynthesis.cancel(); // Stop any current speech
            speechSynthesis.speak(utterance);
            currentUtterance = utterance;

            // Update slide if necessary
            if (currentSlide !== step.slide) {
                currentSlide = step.slide;
                slideStartTime = millis();
            }
        }

        function startNarration() {
            if (!isNarrating) {
                isNarrating = true;
                currentNarrationStepIndex = 0;
                currentSlide = 0; // Reset visuals for narration
                slideStartTime = millis();
                speakCurrentNarration();
            }
        }

        function stopNarration() {
            if (isNarrating) {
                speechSynthesis.cancel();
                isNarrating = false;
                currentUtterance = null;
            }
        }

        // --- Button Event Listeners ---
        document.getElementById('replay').addEventListener('click', () => {
            stopNarration(); // Stop narration if playing
            currentSlide = 0;
            slideStartTime = millis();
            // No reset of narration index if only replay, podcast button will handle that.
        });

        document.getElementById('podcast').addEventListener('click', () => {
            if (isNarrating) {
                stopNarration();
            } else {
                startNarration();
            }
        });
