<description>
An animal cell is the fundamental eukaryotic unit of life in multicellular organisms, characterized by the absence of a cell wall and chloroplasts, and typically possessing an irregular or rounded shape. Encapsulating the cell is the **plasma membrane**, a selectively permeable phospholipid bilayer adorned with embedded proteins and cholesterol, regulating molecular passage. The internal environment, the **cytoplasm**, comprises the jelly-like **cytosol** and various specialized **organelles**.

Dominating the cell's center is the **nucleus**, the command center, enveloped by a double membrane with **nuclear pores**. It houses **chromatin**, the organized form of genetic material (DNA), and a dense **nucleolus** responsible for ribosomal RNA synthesis. Crucial for protein and lipid synthesis are the **endoplasmic reticulum (ER)**: the **rough ER (RER)**, studded with **ribosomes**, synthesizes and folds proteins destined for secretion or membrane integration, while the **smooth ER (SER)** handles lipid synthesis, detoxification, and calcium ion storage.

Proteins and lipids from the ER are then processed by the **Golgi apparatus**, a stack of flattened sacs that modifies, sorts, and packages these molecules into **vesicles** for transport. The cell's energy powerhouses are the **mitochondria**, double-membraned organelles with folded inner membranes (cristae) where aerobic respiration generates ATP. Waste breakdown and cellular recycling are performed by **lysosomes**, containing hydrolytic enzymes, and **peroxisomes**, which detoxify substances and manage oxidative stress. Maintaining cell shape, facilitating movement, and transporting organelles is the **cytoskeleton**, a dynamic network of microtubules, microfilaments, and intermediate filaments, making the animal cell a highly organized and dynamic biological system.
</description>

        class Particle {
            constructor(x, y, type) {
                this.pos = createVector(x, y);
                this.vel = p5.Vector.random2D().mult(random(0.5, 2));
                this.acc = createVector();
                this.lifespan = 255;
                this.type = type; // 'protein' or 'atp' or 'waste'
                this.r = type === 'protein' ? 4 : (type === 'atp' ? 6 : 5);
                this.color = type === 'protein' ? color(getComputedStyle(document.documentElement).getPropertyValue('--protein-color').trim()) :
                             (type === 'atp' ? color(getComputedStyle(document.documentElement).getPropertyValue('--atp-color').trim()) :
                             color(200, 100, 100, 200)); // Waste color
            }

            update() {
                this.vel.add(this.acc);
                this.pos.add(this.vel);
                this.acc.mult(0);
                this.lifespan -= 2;
                this.r *= 0.98; // Shrink over time
            }

            display() {
                noStroke();
                this.color.setAlpha(this.lifespan);
                fill(this.color);
                ellipse(this.pos.x, this.pos.y, this.r * 2);
            }

            isDead() {
                return this.lifespan < 0 || this.r < 1;
            }
        }

            createMembraneShape(numPoints) {
                this.membranePoints = []; // Clear existing points

                let radiusY = this.h / 2;

                    let y = radiusY * sin(angle);
                    // Add some randomness for an organic shape
                    x += random(-radiusX * 0.08, radiusX * 0.08);
                    y += random(-radiusY * 0.08, radiusY * 0.08);
                    this.membranePoints.push(createVector(x, y));
                }
            }

            update() {

                beginShape();
                vertex(x - w * 0.3, y + h * 0.2);
                bezierVertex(x - w * 0.1, y + h * 0.05,
                             x + w * 0.1, y + h * 0.05,
                             x + w * 0.3, y + h * 0.2);
                endShape();
                beginShape();
                vertex(x - w * 0.25, y - h * 0.1);
                bezierVertex(x, y - h * 0.2,

                                  voices.find(voice => voice.lang === 'en-US' && (voice.gender === 'female' || voice.name.toLowerCase().includes('female')));
                    
                    if (!maleVoice || !femaleVoice) {
                        console.warn("Could not find distinct male/female Google US English voices. Falling back to any available en-US voices.");
                        maleVoice = voices.find(voice => voice.lang === 'en-US' && !voice.name.toLowerCase().includes('female')) || voices.find(voice => voice.lang === 'en-US');
                        femaleVoice = voices.find(voice => voice.lang === 'en-US' && !voice.name.toLowerCase().includes('male')) || voices.find(voice => voice.lang === 'en-US');
                    }
                };
                // Manually trigger voice loading if onvoiceschanged doesn't fire immediately
                if (window.speechSynthesis.getVoices().length > 0) {
                    window.speechSynthesis.onvoiceschanged();
                }
            } else {
                console.warn("Speech Synthesis API not supported by this browser.");
            }

            document.getElementById('replay').addEventListener('click', replayAnimation);
            document.getElementById('podcast').addEventListener('click', startPodcast);

            resetAnimation();
        }

        function draw() {
            background(0, 0, 0, 0); // Transparent background for canvas to show CSS body background
            translate(width / 2, height / 2); // Center cell
            cell.update();
            cell.display();

            if (!isPodcastPlaying) {
                // If not podcast, progress visuals based on a fixed timer
                let elapsed = millis() - animationStartTime;
                let currentVisualDuration = narrationData[currentStep] ? narrationData[currentStep].visualDuration : 5000;
                if (elapsed > currentVisualDuration) {
                    currentStep = (currentStep + 1) % narrationData.length;
                    animationStartTime = millis();
                    cell.setHighlight(narrationData[currentStep].highlight);
                }
            }
        }

        function resetAnimation() {
            if (currentUtterance && window.speechSynthesis.speaking) {
                window.speechSynthesis.cancel();
            }
            isPodcastPlaying = false;
            currentStep = 0;
            animationStartTime = millis();
            cell.setHighlight(null); // No initial highlight
            cell.particles = []; // Clear particles
        }

        function startPodcast() {
            resetAnimation(); // Reset visuals and stop any ongoing speech
            isPodcastPlaying = true;
            speakNextNarrative();
        }

        function speakNextNarrative() {
            if (currentStep < narrationData.length) {
                let data = narrationData[currentStep];
                currentUtterance = new SpeechSynthesisUtterance(data.text);
                currentUtterance.pitch = data.pitch;
                currentUtterance.rate = data.rate;

                if (data.voice === "male" && maleVoice) {
                    currentUtterance.voice = maleVoice;
                } else if (data.voice === "female" && femaleVoice) {
                    currentUtterance.voice = femaleVoice;
                } else {
                    currentUtterance.voice = voices.find(v => v.lang === 'en-US'); // Fallback
                }
                
                // Ensure a voice is selected before speaking
                if (!currentUtterance.voice && voices.length > 0) {
                    currentUtterance.voice = voices[0]; // Take the first available voice as a last resort
                }


                currentUtterance.onstart = () => {
                    cell.setHighlight(data.highlight);
                };

                currentUtterance.onend = () => {
                    currentStep++;
                    if (isPodcastPlaying) {
                        speakNextNarrative();
                    } else {
                        cell.setHighlight(null);
                    }
                };

                currentUtterance.onerror = (event) => {
                    console.error('Speech synthesis error:', event);
                    currentStep++;
                    if (isPodcastPlaying) {
                        speakNextNarrative();
                    }
                };
                
                if (currentUtterance.voice) { // Only attempt to speak if a voice is available
                    window.speechSynthesis.speak(currentUtterance);
                } else {
                    console.warn("No suitable voice found for narration. Skipping step.");
                    currentStep++;
                    speakNextNarrative(); // Try next step
                }
            } else {
                // End of narration
                isPodcastPlaying = false;
                cell.setHighlight(null);
                currentStep = 0; // Reset for replay functionality
                animationStartTime = millis(); // Reset timer for non-podcast loop
            }
        }

        function replayAnimation() {
            resetAnimation(); // Just reset visuals, without starting podcast
        }

        function windowResized() {
            resizeCanvas(windowWidth * 0.8, windowHeight * 0.8);
            cell = new Cell(0, 0, width * 0.8, height * 0.8);
            cell.createMembraneShape(20);
        }
