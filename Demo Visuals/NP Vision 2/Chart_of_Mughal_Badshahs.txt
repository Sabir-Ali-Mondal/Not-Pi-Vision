<description>
The "Chart of Mughal Badshahs" serves as a crucial historical document, systematically mapping the genealogical succession and chronological reign periods of the emperors who governed the vast Mughal Empire (1526-1857 CE) across the Indian subcontinent. Scientifically, it functions as a visual representation akin to a dynastic dendrogram or a temporal Gantt chart. Analysis of such a chart allows scholars to perform quantitative studies on regnal lengths, identifying periods of dynastic stability (e.g., the long reigns of Akbar or Aurangzeb) versus periods of rapid succession and instability (e.g., the post-Aurangzeb 'Later Mughals'). It facilitates the correlation of imperial strength, territorial expansion, and cultural efflorescence with specific rulers or generational shifts. Furthermore, the chart reveals patterns of inter-familial conflict, succession crises, and the strategic alliances forged through marital ties, providing empirical data for sociopolitical modeling of imperial governance and decline. This structured representation is indispensable for diachronic analysis of South Asian geopolitics.
</description>

        // Fetch voices once they are loaded
        synth.onvoiceschanged = () => {
            voices = synth.getVoices();
            // Try to find specific voices, fallback to defaults
            const femaleVoice = voices.find(voice => voice.name.toLowerCase().includes('google') && voice.name.toLowerCase().includes('us english') && voice.name.toLowerCase().includes('female')) || voices.find(voice => voice.name.toLowerCase().includes('female')) || voices[0];
            const maleVoice = voices.find(voice => voice.name.toLowerCase().includes('google') && voice.name.toLowerCase().includes('us english') && voice.name.toLowerCase().includes('male')) || voices.find(voice => voice.name.toLowerCase().includes('male')) || (voices.length > 1 ? voices[1] : voices[0]);
            
            // Assign voices to narration script items
            narrationScript.forEach(item => {
                if (item.voiceType === 0) item.assignedVoice = maleVoice;
                if (item.voiceType === 1) item.assignedVoice = femaleVoice;
            });

        // Function to start or resume narration
        function speakNext() {
            if (currentNarrationIndex >= narrationScript.length) {
                isNarrating = false;
                currentHighlight = null;
                appState = maxStates; // End state
                document.getElementById('podcast').textContent = '🔊';
                return;
            }

            const currentSegment = narrationScript[currentNarrationIndex];
            
            // Check if we need to advance the visual state
            if (appState !== currentSegment.state) {
                appState = currentSegment.state;
                stateStartTime = millis();
            }

            currentHighlight = currentSegment.highlight; // Set highlight for current segment

            let utterance = new SpeechSynthesisUtterance(currentSegment.text);
            utterance.voice = currentSegment.assignedVoice; // Use assigned voice
            utterance.pitch = currentSegment.pitch || 1.0;
            utterance.rate = currentSegment.rate || 1.0;

            utterance.onend = () => {
                currentNarrationIndex++;
                currentUtterance = null; // Clear current utterance
                setTimeout(speakNext, currentSegment.delay); // Add delay between sentences
            };

            utterance.onerror = (event) => {
                console.error('SpeechSynthesisUtterance.onerror', event);
                isNarrating = false;
                document.getElementById('podcast').textContent = '🔊';
            };

            synth.speak(utterance);
            currentUtterance = utterance;
            isNarrating = true;
            document.getElementById('podcast').textContent = '⏸️'; // Change button to pause icon
        }

        function stopNarration() {
            if (synth.speaking) {
                synth.cancel();
            }
            isNarrating = false;
            document.getElementById('podcast').textContent = '🔊';
            currentUtterance = null;
        }

        function startNarration() {
            stopNarration(); // Stop any ongoing narration first
            appState = 0; // Reset visual state
            currentNarrationIndex = 0; // Reset narration index
            currentHighlight = null;
            stateStartTime = millis(); // Reset state timer for the new visual start
            speakNext();
        }

        document.getElementById('podcast').addEventListener('click', () => {
            if (isNarrating) {
                stopNarration();
            } else {
                startNarration();
            }
        });

        document.getElementById('replay').addEventListener('click', () => {
            stopNarration(); // Stop narration if running, but don't re-trigger it
            appState = 0;
            stateStartTime = millis();
            currentHighlight = null;
        });

        // P5.js setup
        function setup() {
            canvasWidth = windowWidth;
            canvasHeight = windowHeight;
            let canvas = createCanvas(canvasWidth, canvasHeight);
            canvas.parent(document.body); // Attach canvas to body
            frameRate(30);
            stateStartTime = millis();
            if (voices.length === 0) {
                 // If voices aren't loaded yet, try to load them again
                 synth.onvoiceschanged();
            }
        }

        // P5.js draw loop
        function draw() {
            background(color('var(--bg-color)'));

            // Check if narration is active. If not, manage state by time
            if (!isNarrating) {
                if (millis() - stateStartTime > stateDuration) {
                    appState = (appState + 1) % (maxStates + 1); // Cycle through states, +1 for intro state
                    stateStartTime = millis();
                    currentHighlight = null; // Clear highlight when state changes manually
                }
            }

            switch (appState) {
                case 0: // Introduction
                    drawIntroCard();
                    break;
                case 1: // What is Badshahs/Mughal Empire
                    drawDefinitionCard();
                    break;
                case 2: // Early Emperors: Babur, Humayun, Akbar
                    drawEarlyEmperorsCard();
                    break;
                case 3: // Later Emperors: Jahangir, Shah Jahan, Aurangzeb
                    drawLaterEmperorsCard();
                    break;
                case 4: // Decline: Later Mughals
                    drawDeclineCard();
                    break;
                case 5: // Legacy and Conclusion
                    drawLegacyCard();
                    break;
                default:
                    // Loop back or end
                    appState = 0;
                    stateStartTime = millis();
                    break;
            }
        }

        function windowResized() {
            canvasWidth = windowWidth;
            canvasHeight = windowHeight;
            resizeCanvas(canvasWidth, canvasHeight);
        }

        // Helper function to draw a glassmorphic card
        function drawGlassCard(x, y, w, h) {
            push();
            let c = color(getComputedStyle(document.documentElement).getPropertyValue('--card-bg'));
            // getComputedStyle to read CSS variable's rgba value correctly
            // and apply its alpha to p5 color.
            let alphaValue = c.levels[3]; // Get alpha from RGBA color
            c.setAlpha(alphaValue); 
            fill(c);
            stroke(color(getComputedStyle(document.documentElement).getPropertyValue('--card-border')));
            strokeWeight(1);
            drawingContext.filter = 'blur(10px)'; // p5 specific way for blur
            rect(x, y, w, h, 15); // Rounded corners
            drawingContext.filter = 'none'; // Reset filter
            pop();
        }

        // Helper function for text with highlighting
        function drawTextWithHighlight(textStr, x, y, size, align, highlightKey, targetHighlight, fillCol = 'var(--text-color)') {
            push();
            textSize(size);
            textAlign(align);
            noStroke();

            if (highlightKey === targetHighlight) {
                fill(color('var(--highlight-color)'));
            } else {
                fill(color(fillCol));
            }
            text(textStr, x, y);
            pop();
        }

            let numParticles = 8;
            for (let i = 0; i < numParticles; i++) {
                let angle = (TWO_PI / numParticles) * i + frameCount * 0.02;
                let px = centerX + cos(angle) * orbitRadius;
                let py = centerY + sin(angle) * orbitRadius;
                fill(color('var(--primary-color)'));
                noStroke();
                ellipse(px, py, 10, 10); // Small glowing circles

            textAlign(LEFT);

            let yOffset = cardY + cardH * 0.18;
            for (let i = 0; i < emperorsData.length; i++) {
                let emp = emperorsData[i];
                let currentY = yOffset + i * spacingY;

                // Highlight background behind the name/details
                if (currentHighlight === emp.key) {
                    let highlightAlpha = map(sin(frameCount * 0.1), -1, 1, 0.3, 0.7); // Pulsating alpha
                    let highlightColor = color(getComputedStyle(document.documentElement).getPropertyValue('--highlight-color'));
                    highlightColor.setAlpha(highlightAlpha * 255);
                    fill(highlightColor);
                    noStroke();
                    rect(cardX + cardW * 0.05, currentY - nameSize * 0.8, cardW * 0.9, spacingY * 0.9, 5);

                drawTextWithHighlight(emp.name, textLeftX, currentY, nameSize, LEFT, emp.key, currentHighlight);
                drawTextWithHighlight(emp.reign, textLeftX, currentY + nameSize * 1.1, detailSize, LEFT, emp.key, currentHighlight);
                drawTextWithHighlight(emp.desc, textLeftX + cardW * 0.3, currentY + nameSize * 0.5, detailSize * 1.1, LEFT, emp.key, currentHighlight);
            }
            pop();
        }

            let centerX = cardX + cardW / 2;
            let centerY = cardY + cardH * 0.7;
            let radius = cardH * 0.1;
            let numPoints = 12;

            push();
            noFill();
            stroke(color('var(--primary-color)'));
            strokeWeight(2);
            beginShape();
            for (let i = 0; i <= numPoints; i++) {
                let angle = (TWO_PI / numPoints) * i + frameCount * 0.01;
                let x = centerX + cos(angle) * radius * (1 + sin(frameCount * 0.05 + i) * 0.1);
                let y = centerY + sin(angle) * radius * (1 + cos(frameCount * 0.05 + i) * 0.1);
                curveVertex(x, y);
            }
            endShape(CLOSE); // Use CLOSE to connect last point to first
            pop();
        }
