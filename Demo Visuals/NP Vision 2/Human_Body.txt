    function sketchyPath(vertices, p) {
        for (let i = 0; i < vertices.length - 1; i++) {
            const start = vertices[i];
            const end = vertices[i + 1];
            
            const segmentEndProgress = (i + 1) / (vertices.length - 1);
            if (p < i / (vertices.length - 1)) continue;
            
            const segmentProgress = constrain(map(p, i / (vertices.length-1), segmentEndProgress, 0, 1), 0, 1);
            
            const currentEnd = {
                x: lerp(start.x, end.x, segmentProgress),
                y: lerp(start.y, end.y, segmentProgress)
            };
            
            for(let j = 0; j < 2; j++) {
                line(start.x + random(-1,1), start.y + random(-1,1), currentEnd.x + random(-1,1), currentEnd.y + random(-1,1));
            }
        }
    }
    
    function sketchyShape(vertices, p) {
        beginShape();
        for (let i = 0; i < vertices.length; i++) {
             const start = vertices[i];
             const end = vertices[(i + 1) % vertices.length];
             
             const segmentStartProgress = i / vertices.length;
             const segmentEndProgress = (i + 1) / vertices.length;
             
             if (p < segmentStartProgress) continue;
             
             vertex(start.x, start.y);
             
             if (p < segmentEndProgress) {
                 const segmentProgress = map(p, segmentStartProgress, segmentEndProgress, 0, 1);
                 const currentEnd = {
                     x: lerp(start.x, end.x, segmentProgress),
                     y: lerp(start.y, end.y, segmentProgress)
                 };
                 vertex(currentEnd.x, currentEnd.y);
                 break;
             }
        }
        endShape();
    }
    
    function sketchyCircle(data, p) {
        const chaos = 2;
        const numPoints = 30;
        const maxPoints = floor(p * numPoints);
        beginShape();
        for (let i = 0; i < maxPoints; i++) {
            const angle = map(i, 0, numPoints, 0, 360);
            const r = (data.r / 2) * (1 + 0.1 * sin(angle * 6)); // Make it a bit wobbly
            const x = data.x + (r + random(-chaos, chaos)) * cos(angle);
            const y = data.y + (r + random(-chaos, chaos)) * sin(angle);
            vertex(x, y);
        }
        endShape(CLOSE);
    }

    function sketchyRib(data, p) {
        const endX = lerp(data.x, data.x + data.w, p);
        const endX_mirror = lerp(-data.x, -data.x - data.w, p);

            quadraticVertex(data.x + data.w * 0.5, data.y - 10, endX + random(-1,1), data.y + chaosY);
            endShape();
            // Left side
            beginShape();
            vertex(-data.x + random(-1,1), data.y + chaosY);
            quadraticVertex(-data.x - data.w * 0.5, data.y - 10, endX_mirror + random(-1,1), data.y + chaosY);
            endShape();
        }
    }
