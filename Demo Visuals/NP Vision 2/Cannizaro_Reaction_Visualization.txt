<description>
The Cannizaro reaction is a crucial chemical disproportionation reaction observed in aldehydes lacking an alpha-hydrogen atom (i.e., no hydrogen on the carbon directly adjacent to the carbonyl group), such as Benzaldehyde (C₆H₅CHO) or Formaldehyde (HCHO). This reaction occurs under the influence of a strong, concentrated base, typically aqueous sodium hydroxide (NaOH) or potassium hydroxide (KOH). In this self-redox process, one molecule of the aldehyde undergoes oxidation to form a carboxylic acid (which rapidly deprotonates to its corresponding carboxylate salt in the basic medium), while a second molecule of the same aldehyde is simultaneously reduced to a primary alcohol.

The detailed mechanism commences with the nucleophilic attack of a hydroxide ion (OH⁻) on the electrophilic carbonyl carbon of the first aldehyde molecule. This attack forms a tetrahedral alkoxide intermediate, which possesses a negatively charged oxygen and the aldehyde's original hydrogen atom still attached to the now sp³ hybridized carbonyl carbon. Uniquely, this highly unstable tetrahedral intermediate then undergoes a key, rate-determining intermolecular hydride (H⁻) transfer. This hydride, expelled from the alkoxide, acts as a nucleophile, attacking the carbonyl carbon of a *second* aldehyde molecule. Concomitantly, the first aldehyde molecule, having lost its hydride, transforms into a carboxylic acid, while the second aldehyde, having gained the hydride, becomes an alkoxide. Due to the strongly basic conditions, the carboxylic acid immediately transfers its acidic proton to the alkoxide. This final proton exchange yields the stable carboxylate anion and the primary alcohol, completing the disproportionation of a single aldehyde starting material into two distinct products.
</description>

        // --- Molecular Definition Functions ---
        // These functions define the static structure of each molecule type.
        // x, y coordinates for atoms are relative to the molecule's origin (0,0).
        function createBenzaldehyde(originX, originY, idPrefix) {
            let atoms = [];
            let bonds = [];
            let mol = { pos: { x: originX, y: originY }, atoms: [], bonds: [], id: idPrefix, opacity: 1 };

            let r = 50; // Benzene ring radius
            let atomR_C = ATOM_RADIUS_C;
            let atomR_O = ATOM_RADIUS_O;
            let atomR_H = ATOM_RADIUS_H;

            // Benzene ring carbons (approximate positions for hexagon)
            // C0 is the one attached to the CHO group
            for (let i = 0; i < 6; i++) {
                let angle = TWO_PI / 6 * i - PI/2; // Start from top
                let x = r * cos(angle);
                let y = r * sin(angle);
                atoms.push({ id: `${idPrefix}_C_ring_${i}`, type: 'C', x: x, y: y, radius: atomR_C, color: '#909090', charge: 0 });
            }
            // Connect ring carbons (alternating single/double for resonance visualization)
            for (let i = 0; i < 6; i++) {
                bonds.push({ from: `${idPrefix}_C_ring_${i}`, to: `${idPrefix}_C_ring_${(i + 1) % 6}`, type: (i % 2 === 0 ? 'double' : 'single') });
            }

            // Aldehyde group attached to C_ring_0
            let carbonylC_x = atoms.find(a => a.id === `${idPrefix}_C_ring_0`).x + r + 20;
            let carbonylC_y = atoms.find(a => a.id === `${idPrefix}_C_ring_0`).y;
            let carbonylC = { id: `${idPrefix}_C_CHO`, type: 'C', x: carbonylC_x, y: carbonylC_y, radius: atomR_C, color: '#909090', charge: 0 };
            let carbonylO = { id: `${idPrefix}_O_CHO`, type: 'O', x: carbonylC_x + atomR_C * 1.5, y: carbonylC_y - atomR_O * 1.2, radius: atomR_O, color: '#FF6347', charge: 0 };
            let aldehydeH = { id: `${idPrefix}_H_CHO`, type: 'H', x: carbonylC_x + atomR_C * 1.5, y: carbonylC_y + atomR_H * 1.2, radius: atomR_H, color: '#E0E0E0', charge: 0 };

            atoms.push(carbonylC, carbonylO, aldehydeH);
            bonds.push(
                { id: `${idPrefix}_C_ring_0_C_CHO`, from: `${idPrefix}_C_ring_0`, to: `${idPrefix}_C_CHO`, type: 'single' },
                { id: `${idPrefix}_C_CHO_O_CHO`, from: `${idPrefix}_C_CHO`, to: `${idPrefix}_O_CHO`, type: 'double' },
                { id: `${idPrefix}_C_CHO_H_CHO`, from: `${idPrefix}_C_CHO`, to: `${idPrefix}_H_CHO`, type: 'single' }
            );
            mol.atoms = atoms;
            mol.bonds = bonds;
            return mol;
        }

        function createHydroxide(originX, originY, idPrefix) {
            let atoms = [];
            let bonds = [];
            let mol = { pos: { x: originX, y: originY }, atoms: [], bonds: [], id: idPrefix, opacity: 1 };

            let atomR_O = ATOM_RADIUS_O;
            let atomR_H = ATOM_RADIUS_H;

            let oxygen = { id: `${idPrefix}_O`, type: 'O', x: 0, y: 0, radius: atomR_O, color: '#FF6347', charge: -1 };
            let hydrogen = { id: `${idPrefix}_H`, type: 'H', x: atomR_O + atomR_H * 0.8, y: 0, radius: atomR_H, color: '#E0E0E0', charge: 0 };

            atoms.push(oxygen, hydrogen);
            bonds.push({ id: `${idPrefix}_O_H`, from: `${idPrefix}_O`, to: `${idPrefix}_H`, type: 'single' });

            mol.atoms = atoms;
            mol.bonds = bonds;
            return mol;
        }

        function createTetrahedralIntermediate(originX, originY, idPrefix) {
            let atoms = [];
            let bonds = [];
            let mol = { pos: { x: originX, y: originY }, atoms: [], bonds: [], id: idPrefix, opacity: 1 };

            let r = 50; // Benzene ring radius
            let atomR_C = ATOM_RADIUS_C;
            let atomR_O = ATOM_RADIUS_O;
            let atomR_H = ATOM_RADIUS_H;

            // Benzene ring carbons
            for (let i = 0; i < 6; i++) {
                let angle = TWO_PI / 6 * i - PI/2;
                let x = r * cos(angle);
                let y = r * sin(angle);
                atoms.push({ id: `${idPrefix}_C_ring_${i}`, type: 'C', x: x, y: y, radius: atomR_C, color: '#909090', charge: 0 });
            }
            for (let i = 0; i < 6; i++) {
                bonds.push({ id: `${idPrefix}_C_ring_${i}_C_ring_${(i+1)%6}`, from: `${idPrefix}_C_ring_${i}`, to: `${idPrefix}_C_ring_${(i + 1) % 6}`, type: (i % 2 === 0 ? 'double' : 'single') });
            }

            // Transformed carbonyl C (sp3)
            let carbonylC_x = atoms.find(a => a.id === `${idPrefix}_C_ring_0`).x + r + 20;
            let carbonylC_y = atoms.find(a => a.id === `${idPrefix}_C_ring_0`).y;

            // New incoming OH group
            let newOxygen = { id: `${idPrefix}_O_OH`, type: 'O', x: carbonylC_x - atomR_C * 1.5, y: carbonylC_y - atomR_O * 1.2, radius: atomR_O, color: '#FF6347', charge: 0 };

        }

        // Generic drawing function for any molecule object
        function drawMolecule(mol, opacity = 1) {
            if (!mol || mol.opacity * opacity <= 0.05) return; // Don't draw if too transparent

            push();
            translate(mol.pos.x, mol.pos.y);

            // Draw bonds first
            for (let bond of mol.bonds) {
                let fromAtom = mol.atoms.find(a => a.id === bond.from);
                let toAtom = mol.atoms.find(a => a.id === bond.to);
                if (fromAtom && toAtom) {
                    let isBondHighlighted = (highlightTarget?.type === 'bond' && highlightTarget?.targetId === bond.id) || 
                                            (highlightTarget?.type === 'molecule' && highlightTarget?.targetId === mol.id);
                    drawBond(fromAtom.x, fromAtom.y, toAtom.x, toAtom.y, bond.type, isBondHighlighted, mol.opacity * opacity);
                }
            }

            // Draw atoms
            for (let atom of mol.atoms) {
                let isAtomHighlighted = (highlightTarget?.type === 'atom' && highlightTarget?.targetId === atom.id) || 
                                        (highlightTarget?.type === 'molecule' && highlightTarget?.targetId === mol.id);
                drawAtom(atom.x, atom.y, atom, isAtomHighlighted, mol.opacity * opacity);
            }

            pop();
        }

        function drawAtom(x, y, atom, isHighlighted, opacity) {
            push();
            translate(x, y);

            // Glow effect if highlighted
            if (isHighlighted) {
                fill(red(GLOW_COLOR), green(GLOW_COLOR), blue(GLOW_COLOR), opacity * highlightPulsate * 255 * 0.7);
                drawingContext.filter = `blur(${8 * highlightPulsate}px)`;
                circle(0, 0, atom.radius * 2 + 10 * highlightPulsate);
                drawingContext.filter = 'none';
            }

            // Main atom body (glassmorphism look)
            let fillColor = color(atom.color);
            fillColor.setAlpha(opacity * 255 * 0.6); // Semi-transparent fill
            fill(fillColor);
            
            let strokeColor = lerpColor(color(atom.color), color(GLASS_BORDER), 0.5);
            strokeColor.setAlpha(opacity * 255);
            stroke(strokeColor);
            strokeWeight(2);
            
            circle(0, 0, atom.radius * 2);

            // Subtle inner highlight/gradient for depth
            for(let i=0; i<3; i++) {
                noStroke();
                let innerColor = color(255, 255, 255, 10 + i*10);
                innerColor.setAlpha(opacity * 255 * 0.1 * (3-i));
                fill(innerColor);
                circle(-atom.radius * 0.4, -atom.radius * 0.4, atom.radius * 0.8 * (1.5 - i * 0.2));
            }


            // Charge indicator
            if (atom.charge !== 0) {
                let chargeColor = color(TEXT_COLOR);
                chargeColor.setAlpha(opacity * 255);
                fill(chargeColor);
                noStroke();
                textSize(atom.radius * 0.8);
                textAlign(CENTER, CENTER);
                text(atom.charge > 0 ? '+' : (atom.charge < 0 ? '-' : ''), 0, 0); // Display only if non-zero
            }
            pop();
        }

        function drawBond(x1, y1, x2, y2, type, isHighlighted, opacity) {
            push();
            let bondColor = color(TEXT_COLOR);
            bondColor.setAlpha(opacity * 255);

            if (isHighlighted) {
                let glowBondColor = color(red(GLOW_COLOR), green(GLOW_COLOR), blue(GLOW_COLOR), opacity * highlightPulsate * 255 * 0.7);
                drawingContext.filter = `blur(${5 * highlightPulsate}px)`;
                stroke(glowBondColor);
                strokeWeight(BOND_THICKNESS + 2 + 5 * highlightPulsate);
                line(x1, y1, x2, y2);
                drawingContext.filter = 'none';
            }
            
            strokeWeight(BOND_THICKNESS);
            stroke(bondColor);
            line(x1, y1, x2, y2);

            if (type === 'double') {
                let angle = atan2(y2 - y1, x2 - x1);
                let offset = BOND_THICKNESS * 0.8;
                push();
                translate(x1, y1);
                rotate(angle);
                line(0, -offset, dist(x1, y1, x2, y2), -offset);
                pop();
            } else if (type === 'breaking' || type === 'forming') {
                // Dashed line for breaking/forming bonds
                drawingContext.setLineDash([5, 5]); // Dash pattern
                stroke(bondColor);
                line(x1, y1, x2, y2);
                drawingContext.setLineDash([]); // Reset dash
            }
            pop();
        }
