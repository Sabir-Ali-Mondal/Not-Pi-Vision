<description>
"Indian Hotspots" refer to four major biodiversity hotspots located within India, identified for their exceptionally high levels of biodiversity and endemism, coupled with significant habitat loss, making them critical for global conservation efforts. These regions meet stringent criteria: possessing at least 1,500 endemic species of vascular plants (meaning they are found nowhere else on Earth) and having lost at least 70% of their original natural vegetation.

The four primary Indian hotspots are:

1.  **The Western Ghats:** A mountain range running parallel to the western coast of the Indian peninsula. It is globally recognized as one of the "hottest hotspots" due to its rich endemic flora and fauna, particularly amphibians, reptiles, and flowering plants. It encompasses diverse ecosystems, from tropical evergreen forests to montane grasslands.

2.  **The Himalayas:** Covering the entire Indian Himalayan Region (IHR), this hotspot is characterized by its dramatic altitudinal variations, supporting unique ecosystems like alpine meadows, temperate forests, and glacial environments. It is a treasure trove of endemic medicinal plants and rare mammals such as the snow leopard.

3.  **Indo-Burma Region:** This vast hotspot includes parts of Northeast India (east of the Brahmaputra River valley). It is one of the largest biodiversity hotspots globally, distinguished by extensive plains, hills, and a variety of forest types, sustaining rich aquatic and terrestrial species, including numerous endemic freshwater turtles and fish.

4.  **Sundaland (Nicobar Islands):** While Sundaland primarily spans Southeast Asia, India's Nicobar Islands are considered an extension of this hotspot. These islands showcase unique island ecosystems, featuring pristine mangrove forests, vibrant coral reefs, and rainforests, which host a significant number of endemic bird and marine species.

These hotspots are under severe threat from factors like deforestation, agricultural expansion, pollution, urbanization, and climate change. Conservation initiatives in these regions are paramount to protecting a substantial portion of the world's irreplaceable biodiversity.

            // Helper function to check if a point is inside a polygon (Ray Casting Algorithm)
            function pointInPolygon(point, polygon) {
                let x = point.x, y = point.y;
                let inside = false;
                for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
                    let xi = polygon[i].x, yi = polygon[i].y;
                    let xj = polygon[j].x, yj = polygon[j].y;

                    let intersect = ((yi > y) != (yj > y)) && (x < (xj - xi) * (y - yi) / (yj - yi) + xi);
                    if (intersect) inside = !inside;
                }
                return inside;
            }

            class LifeForm {
                constructor(x, y, type, baseHue, endemic = false) {
                    this.x = x;
                    this.y = y;
                    this.vx = random(-0.5, 0.5);
                    this.vy = random(-0.5, 0.5);
                    this.size = random(4, 10);
                    this.baseHue = baseHue; // HSB hue value
                    this.type = type; // 'plant', 'animal', 'fish' etc.
                    this.endemic = endemic;
                    this.health = 1.0; // 1.0 for healthy, decreases with threat
                }

                update(hotspotPolygon = []) {
                    this.x += this.vx;
                    this.y += this.vy;

                    // Keep life forms generally within the map area
                    let minX = 250, maxX = 580, minY = 50, maxY = 580;
                    if (this.x < minX || this.x > maxX) this.vx *= -1;
                    if (this.y < minY || this.y > maxY) this.vy *= -1;

                    // If a specific hotspot is active, try to keep lifeforms within its bounds
                    if (hotspotPolygon.length > 0) {
                        if (!pointInPolygon(this, hotspotPolygon)) {
                            // If outside, gently nudge towards the hotspot's centroid
                            let centerX = hotspotPolygon.reduce((sum, p) => sum + p.x, 0) / hotspotPolygon.length;
                            let centerY = hotspotPolygon.reduce((sum, p) => sum + p.y, 0) / hotspotPolygon.length;
                            let angle = atan2(centerY - this.y, centerX - this.x);
                            this.vx = lerp(this.vx, cos(angle) * 1.5, 0.1);
                            this.vy = lerp(this.vy, sin(angle) * 1.5, 0.1);
                        }
                    }

                    // Apply threat and conservation effects
                    if (currentState === 5) { // Threats
                        this.health = max(0, this.health - 0.003); // Slowly decrease health
                    } else if (currentState === 6) { // Conservation
                        this.health = min(1, this.health + 0.005); // Slowly increase health
                    } else if (currentState >= 1 && currentState <= 4) { // During hotspot intro, maintain full health
                        this.health = 1.0;
                    } else { // Initial state, general health
                        this.health = 0.5; // Start less vibrant
                    }
                }

                display() {
                    colorMode(HSB, 360, 100, 100, 1); // Use HSB for vibrant colors and alpha
                    let s = map(this.health, 0, 1, 30, 90);
                    let b = map(this.health, 0, 1, 30, 90);
                    let a = map(this.health, 0, 1, 0.1, 0.9);
                    fill(this.baseHue, s, b, a);
                    noStroke();

                    push();
                    translate(this.x, this.y);
                    rotate(frameCount * 0.01 + this.x * 0.001); // Subtle rotation

                    if (this.type === 'plant') {
                        ellipse(0, 0, this.size, this.size * 1.5);
                        triangle(-this.size / 2, this.size / 2, this.size / 2, this.size / 2, 0, -this.size / 2);
                    } else if (this.type === 'animal') {
                        ellipse(0, 0, this.size);
                        rect(-this.size / 4, -this.size / 4, this.size / 2, this.size / 2);
                    } else if (this.type === 'fish') {
                        triangle(-this.size, 0, this.size, -this.size / 2, this.size, this.size / 2);
                    } else { // Generic
                        ellipse(0, 0, this.size);
                    }
                    pop();
                    colorMode(RGB); // Reset to default for other drawings
                }
            }

                frameRate(30);

                    speechSynthesis.onvoiceschanged = () => {

                        // Try to find specific voices, fallback to first available

                        femaleVoice = voices.find(voice => voice.name.includes('Google US English') && voice.name.includes('Female')) || voices.find(voice => voice.name.includes('female')) || voices[1] || voices[0];
                    };
                } else {
                    console.warn("SpeechSynthesis API not supported in this browser.");
                    document.getElementById('podcast').style.display = 'none'; // Hide button if not supported
                }

                initAnimation();

                document.getElementById('replay').addEventListener('click', () => {
                    if (isSpeechSynthAvailable) {
                        speechSynthesis.cancel();
                    }
                    initAnimation();
                });

                    if (!isSpeechSynthAvailable) return;
                    speechSynthesis.cancel();
                    initAnimation();
                    currentNarrationIndex = 0;
                    speakNext();
                });
            }

            function initAnimation() {
                currentState = 0;
                animationStartTime = millis();
                lifeForms = []; // Clear existing life forms
                // Populate initial background life forms (less dense, less vibrant)
                for (let i = 0; i < maxLifeForms / 5; i++) {
                    lifeForms.push(new LifeForm(random(280, 550), random(100, 550), 'generic', random(0, 360)));
                }
                speaking = false;
            }

            function speakNext() {
                if (!isSpeechSynthAvailable) return;

                if (currentNarrationIndex < podcastNarration.length) {
                    let nar = podcastNarration[currentNarrationIndex];
                    let utterance = new SpeechSynthesisUtterance(nar.text);
                    utterance.voice = nar.voice === "male" ? maleVoice : femaleVoice;
                    utterance.pitch = nar.pitch;
                    utterance.rate = nar.rate;
                    utterance.onend = () => {
                        currentNarationIndex++;
                        speakNext();
                    };
                    utterance.onerror = (event) => {
                        console.error('SpeechSynthesisUtterance.onerror', event);
                        currentNarrationIndex++; // Still try to proceed
                        speakNext();
                    };
                    speechSynthesis.speak(utterance);
                    currentState = nar.visualState; // Update visual state immediately
                    animationStartTime = millis(); // Reset animation timer for current visual state
                } else {
                    currentNarrationIndex = 0; // Reset for replay
                    speaking = false;
                    currentState = 0; // Go back to initial state
                    animationStartTime = millis();
                }
            }

                background(getComputedStyle(document.documentElement).getPropertyValue('--bg-color'));

                    vertex(p.x, p.y);

                    push();
                    if (currentState === (i + 1) || currentState === 5 || currentState === 6) { // Highlight active or during threat/conservation phase
                        let highlightColor = color(hotspot.highlightColor);
                        let glowAlpha = map(sin(frameCount * 0.05), -1, 1, 0.4, 1);
                        highlightColor.setAlpha(glowAlpha * 255);
                        fill(highlightColor);
                        stroke(highlightColor);
                        strokeWeight(3);

                        // If this is the currently highlighted hotspot (not threat/conservation phase where all are affected)
                        if (currentState === (i + 1)) {
                            activeHotspotPolygon = hotspot.points;
                        }
                    } else {
                        fill(hotspot.color);
                        noStroke();
                    }
                    beginShape();
                    for (let p of hotspot.points) {
                        vertex(p.x, p.y);
                    }
                    endShape(CLOSE);
                    pop();
                }

                // Manage life forms based on state
                if (currentState >= 1 && currentState <= 4 && lifeForms.length < maxLifeForms) {
                    let hotspot = hotspotsData[currentState - 1];
                    let center = hotspot.center;
                    // Spawn new life forms in the active hotspot
                    for (let i = 0; i < 2; i++) { // Spawn a few each frame
                        let type = random(['plant', 'animal', 'fish']);
                        let hue = random(0, 360); // Random vibrant hue
                        let newLifeForm = new LifeForm(
                            random(center.x - 30, center.x + 30),
                            random(center.y - 30, center.y + 30),
                            type,
                            hue,
                            true // These are endemic to the hotspot
                        );
                        lifeForms.push(newLifeForm);
                        if (lifeForms.length > maxLifeForms) {
                            lifeForms.splice(0, lifeForms.length - maxLifeForms); // Trim oldest if too many
                        }
                    }
                }

                // Update and display life forms
                for (let i = lifeForms.length - 1; i >= 0; i--) {
                    let lf = lifeForms[i];
                    lf.update(activeHotspotPolygon);
                    lf.display();

                    // Remove if health drops too low (threat state)
                    if (currentState === 5 && lf.health < 0.01) {
                        lifeForms.splice(i, 1);
                    }
                }
            }
