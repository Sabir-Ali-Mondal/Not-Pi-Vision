<description>
Hacking, viewed from a scientific perspective, constitutes the systematic identification and exploitation of vulnerabilities within computational systems or networks to achieve unauthorized access or manipulate their intended functions. This multi-stage process typically begins with **reconnaissance**, involving both passive and active intelligence gathering. This phase aims to comprehensively map the target's attack surface, including network topology, active services, open ports, and specific software versions, often leveraging packet analysis and protocol enumeration. Subsequently, **vulnerability analysis** is performed, where the collected data is cross-referenced with known security flaws (e.g., Common Vulnerabilities and Exposures - CVEs), misconfigurations, or design weaknesses in protocols. An **exploit** is then developed or adapted—a meticulously crafted sequence of data or instructions designed to leverage a specific vulnerability, often by manipulating buffer overflows, integer overflows, or race conditions to alter program flow. Upon successful **delivery**, bypassing security mechanisms like firewalls or intrusion detection systems through techniques such as fragmentation or obfuscation, the exploit triggers a **payload execution**. This payload, typically shellcode, grants initial, often low-privileged, access to the system by subverting the operating system's process management or memory allocation. The process frequently escalates to **privilege escalation**, exploiting kernel vulnerabilities or misconfigurations to gain administrative or root access. Subsequent **post-exploitation activities** include data exfiltration, where sensitive information is systematically identified and transferred out, and establishing **persistence** mechanisms like backdoors or rootkits, coupled with anti-forensic techniques to evade detection and ensure future access to the compromised digital architecture.
</description>

        class Packet {
            constructor(startX, startY, endX, endY, speed, type = 'data', size = 8, colorVal) {
                this.startX = startX;
                this.startY = startY;
                this.endX = endX;
                this.endY = endY;
                this.pos = createVector(startX, startY);
                this.target = createVector(endX, endY);
                this.speed = speed;
                this.progress = 0; // 0 to 1
                this.type = type; // 'data', 'exploit', 'exfiltrate'
                this.size = size;
                this.color = colorVal || color(255, 255, 255);
                this.glow = 0;
                this.alpha = 255;
            }

            update() {
                this.progress += this.speed / dist(this.startX, this.startY, this.endX, this.endY);
                this.progress = constrain(this.progress, 0, 1);
                this.pos.x = lerp(this.startX, this.endX, this.progress);
                this.pos.y = lerp(this.startY, this.endY, this.progress);

                this.glow = map(sin(frameCount * 10), -1, 1, 0, 20);

                if (this.type === 'exploit') {
                    this.size = lerp(8, 20, this.progress); // Grow as it travels
                    this.alpha = lerp(255, 255, this.progress); // Stay opaque
                } else if (this.type === 'exfiltrate') {
                    this.alpha = map(this.progress, 0, 1, 255, 50); // Fade out as it arrives
                }
            }

            draw() {
                push();
                translate(this.pos.x, this.pos.y);
                rotate(frameCount * 5); // Rotate for dynamic feel

                drawingContext.shadowBlur = this.glow;
                drawingContext.shadowColor = this.color;

                fill(this.color.levels[0], this.color.levels[1], this.color.levels[2], this.alpha);
                noStroke();

                if (this.type === 'exploit') {
                    beginShape();
                    vertex(0, -this.size);
                    vertex(this.size * 0.866, this.size * 0.5);
                    vertex(-this.size * 0.866, this.size * 0.5);
                    endShape(CLOSE); // Triangle for exploit
                } else {
                    rectMode(CENTER);
                    rect(0, 0, this.size, this.size); // Square for data
                }
                drawingContext.shadowBlur = 0;
                pop();
            }

            isFinished() {
                return this.progress >= 1;
            }
        }

                for (let i = 0; i < logLines.length; i++) {
                    text(logLines[i], 10, 10 + i * 16);
                }
                pop();
            }
        }

        // --- Narration and Synchronization Logic ---

        function loadVoices() {
            speechSynthesis.onvoiceschanged = () => {
                voices = speechSynthesis.getVoices();
                maleVoice = voices.find(voice => voice.name.includes('Google US English') && voice.lang === 'en-US') || voices.find(voice => voice.name.includes('Male')) || voices[0];
                femaleVoice = voices.find(voice => voice.name.includes('Google US English') && voice.lang === 'en-US') || voices.find(voice => voice.name.includes('Female')) || voices[1] || voices[0];
                console.log("Voices loaded:", voices.map(v => v.name));
            };
            // Trigger voice loading if not already loaded
            if (speechSynthesis.getVoices().length === 0) {
                speechSynthesis.getVoices();
            }
        }

        function startNarration() {
            if (isNarrating) return; // Prevent multiple narrations
            isNarrating = true;
            narrationIndex = 0;
            speakNext();
        }

        function stopNarration() {
            if (currentUtterance) {
                speechSynthesis.cancel();
                currentUtterance = null;
            }
            isNarrating = false;
        }

        function speakNext() {
            if (narrationIndex >= narration.length) {
                isNarrating = false;
                initAnimation(); // Reset animation if narration finishes
                return;
            }

            const currentSpeech = narration[narrationIndex];

            // Perform visual action based on narration step
            if (currentSpeech.visualAction) {
                performVisualAction(currentSpeech.visualAction);
            }

            currentUtterance = new SpeechSynthesisUtterance(currentSpeech.text);
            currentUtterance.voice = (currentSpeech.voiceIdx % 2 === 0 ? maleVoice : femaleVoice);
            currentUtterance.pitch = currentSpeech.pitch;
            currentUtterance.rate = currentSpeech.rate;

            currentUtterance.onend = () => {
                narrationIndex++;
                speakNext();
            };
            currentUtterance.onerror = (event) => {
                console.error('SpeechSynthesisUtterance.onerror', event);
                narrationIndex++; // Still try to proceed
                speakNext();
            };

            speechSynthesis.speak(currentUtterance);
        }

        function performVisualAction(action) {
            stageStartTime = millis(); // Reset stage timer for new visual action

            // Reset specific animation elements for new stage
            cleanupStageElements();

            switch (action) {
                case 'reset':
                    initAnimation();
                    break;
                case 'startRecon':
                    currentStage = 1;
                    // Initialize some random packets for reconnaissance
                    for (let i = 0; i < 5; i++) {
                        let startNode = nodes[floor(random(nodes.length))];
                        let endNode = nodes[floor(random(nodes.length))];
                        if (startNode !== endNode) {
                            packets.push(new Packet(startNode.x, startNode.y, endNode.x, endNode.y, random(2, 5), 'data', 8, color(primaryColor)));
                        }
                    }
                    break;
                case 'highlightVulnerability':
                    currentStage = 2;
                    // Pick a random node to highlight, not the attacker or initial server
                    let nonSpecialNodes = nodes.filter(n => n.type !== 'attacker' && n.type !== 'server');
                    highlightedNodeIndex = nodes.indexOf(nonSpecialNodes[floor(random(nonSpecialNodes.length))]);
                    break;
                case 'startExploitCrafting':
                    currentStage = 3;
                    break;
                case 'deliverExploit':
                    currentStage = 4;
                    if (highlightedNodeIndex !== -1) {
                        let targetNode = nodes[highlightedNodeIndex];
                        let attackerNode = nodes.find(n => n.type === 'attacker');
                        if (attackerNode) {
                            exploitPacket = new Packet(attackerNode.x, attackerNode.y, targetNode.x, targetNode.y, 10, 'exploit', 8, color(255, 0, 0));
                        }
                    }
                    firewallAlpha = 0; // Start firewall fade in
                    break;
                case 'executePayload':
                    currentStage = 5;
                    if (highlightedNodeIndex !== -1) {
                        nodes[highlightedNodeIndex].shellAlpha = 0; // Start shell icon fade in
                    }
                    firewallAlpha = 0; // Fade out firewall
                    if(exploitPacket) exploitPacket = null; // Ensure it's gone
                    break;
                case 'escalatePrivileges':
                    currentStage = 6;
                    if (highlightedNodeIndex !== -1) {
                        nodes[highlightedNodeIndex].rootAlpha = 0; // Start root icon fade in
                    }
                    break;
                case 'exfiltrateData':
                    currentStage = 7;
                    dataExfiltrationPackets = []; // Clear any previous exfil packets
                    break;
                case 'establishPersistence':
                    currentStage = 8;
                    if (highlightedNodeIndex !== -1) {
                        nodes[highlightedNodeIndex].textAlpha = 0; // Start text fade in
                        cleanupAlpha = 0; // Start cleanup icon fade in
                    }
                    break;
                case 'endSimulation':
                    currentStage = 9;
                    break;
            }
            console.log(`Visual action: ${action}, new stage: ${currentStage}`);
        }

        function initAnimation() {
            currentStage = 0;
            stageStartTime = millis();
            narrationIndex = 0;
            isNarrating = false;
            stopNarration(); // Ensure no current narration
            resetVisuals();
        }

        function resetVisuals() {
            nodes = [];
            packets = [];
            dataExfiltrationPackets = [];
            exploitPacket = null;
            highlightedNodeIndex = -1;
            vulnerabilityHighlightStrength = 0;
            reconScanProgress = 0;
            shellIconAlpha = 0;
            rootIconAlpha = 0;
            cleanupAlpha = 0;
            firewallAlpha = 0;

            // Generate nodes for the network
            const centerX = width / 2;
            const centerY = height / 2;
            const radius = min(width, height) * 0.3;

            nodes.push(new Node(centerX, centerY - radius, 'server')); // Central server
            nodes.push(new Node(centerX + radius * cos(45), centerY - radius * sin(45), 'workstation'));
            nodes.push(new Node(centerX + radius, centerY, 'workstation'));
            nodes.push(new Node(centerX + radius * cos(315), centerY - radius * sin(315), 'workstation'));
            nodes.push(new Node(centerX, centerY + radius, 'workstation'));
            nodes.push(new Node(centerX - radius * cos(45), centerY + radius * sin(45), 'workstation'));
            nodes.push(new Node(centerX - radius, centerY, 'workstation'));
            nodes.push(new Node(centerX - radius * cos(315), centerY + radius * sin(315), 'attacker')); // Attacker machine

            // Connect nodes (simple full mesh for illustration)
            for (let i = 0; i < nodes.length; i++) {
                for (let j = i + 1; j < nodes.length; j++) {
                    nodes[i].connections.push(nodes[j]);
                    nodes[j].connections.push(nodes[i]);
                }
                nodes[i].isHighlighted = false;
                nodes[i].shellAlpha = 0;
                nodes[i].rootAlpha = 0;
                nodes[i].crackAlpha = 0;
                nodes[i].textAlpha = 0;
            }

            // Set a specific target node for highlightVulnerability for consistency
            highlightedNodeIndex = 1; // Or pick a specific 'workstation' type
            let foundWorkstation = nodes.find(n => n.type === 'workstation');
            if (foundWorkstation) {
                highlightedNodeIndex = nodes.indexOf(foundWorkstation);
            }
        }

        function cleanupStageElements() {
            // Reset highlighting for all nodes
            nodes.forEach(node => {
                node.isHighlighted = false;
                node.pulse = 0;
            });
        }
