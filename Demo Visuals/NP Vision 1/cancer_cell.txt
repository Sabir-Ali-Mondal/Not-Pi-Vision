<description>A cancer cell is a cell that grows uncontrollably and abnormally, defying the body's signals to stop dividing. Unlike healthy cells, cancer cells often exhibit irregular shapes, disorganized behavior, and can invade nearby tissues or spread to other parts of the body. This visualization represents a geometric, animated 2D abstraction of cancer cells proliferating among normal cells, with dynamic motion and clustering behavior to simulate how they grow and disrupt healthy tissue.</description>

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>Cancer Cell Visualization</title>
  <style>
    body {
      margin: 0;
      background: #0e0e0e;
      overflow: hidden;
    }
    canvas {
      display: block;
    }
  </style>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/p5.js/1.6.0/p5.min.js"></script>
  <script>
    let normalCells = [];
    let cancerCells = [];

    function setup() {
      createCanvas(windowWidth, windowHeight);
      for (let i = 0; i < 80; i++) {
        normalCells.push(new Cell(random(width), random(height), false));
      }
      for (let i = 0; i < 5; i++) {
        cancerCells.push(new Cell(random(width), random(height), true));
      }
    }

    function draw() {
      background(14, 14, 14, 60);

      for (let cell of normalCells) {
        cell.update();
        cell.display();
      }

      for (let i = cancerCells.length - 1; i >= 0; i--) {
        cancerCells[i].update();
        cancerCells[i].display();

        // Proliferation
        if (random(1) < 0.005 && cancerCells.length < 100) {
          let newX = cancerCells[i].pos.x + random(-30, 30);
          let newY = cancerCells[i].pos.y + random(-30, 30);
          cancerCells.push(new Cell(newX, newY, true));
        }
      }
    }

    class Cell {
      constructor(x, y, isCancer) {
        this.pos = createVector(x, y);
        this.vel = p5.Vector.random2D().mult(random(0.3, isCancer ? 1.5 : 0.5));
        this.size = isCancer ? random(20, 40) : random(10, 20);
        this.isCancer = isCancer;
        this.angle = random(TWO_PI);
        this.points = floor(random(5, 9));
        this.shapeJitter = [];
        for (let i = 0; i < this.points; i++) {
          this.shapeJitter.push(random(0.8, 1.2));
        }
        this.color = isCancer ? color(255, 80, 120, 180) : color(100, 200, 255, 120);
      }

      update() {
        this.pos.add(this.vel);
        this.edgeBounce();
        this.angle += 0.01;
      }

      edgeBounce() {
        if (this.pos.x < 0 || this.pos.x > width) this.vel.x *= -1;
        if (this.pos.y < 0 || this.pos.y > height) this.vel.y *= -1;
      }

      display() {
        push();
        translate(this.pos.x, this.pos.y);
        rotate(this.angle);
        noStroke();
        fill(this.color);
        beginShape();
        for (let i = 0; i < this.points; i++) {
          let angle = map(i, 0, this.points, 0, TWO_PI);
          let r = this.size * this.shapeJitter[i];
          let x = r * cos(angle);
          let y = r * sin(angle);
          vertex(x, y);
        }
        endShape(CLOSE);
        pop();
      }
    }

    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }
  </script>
</head>
<body></body>
</html>
```
