<description>  
"Downstream Processing Cell Disruption Enzymatic Methods Lysozyme" refers to a specific phase in bioprocessing where lysozyme enzymes are used to break open microbial cells (typically bacteria) to release intracellular products like proteins. This is part of downstream processing, which involves recovering and purifying biological products after fermentation. Lysozyme works by hydrolyzing the peptidoglycan layer in bacterial cell walls, making it a gentle and efficient method of cell disruption compared to mechanical techniques.  
</description>

  <script src="https://cdnjs.cloudflare.com/ajax/libs/p5.js/1.6.0/p5.min.js"></script>
  <script>
    let angle = 0;
    let enzymeCount = 20;
    let enzymes = [];

    function setup() {
      createCanvas(windowWidth, windowHeight, WEBGL);
      for (let i = 0; i < enzymeCount; i++) {
        enzymes.push({
          angle: random(TWO_PI),
          radius: 150,
          yOffset: random(-80, 80),
          speed: random(0.002, 0.005)
        });
      }
    }

    function draw() {
      background(0, 30, 40);
      orbitControl();
      rotateY(angle);
      drawCellStructure();
      drawEnzymes();
      angle += 0.003;
    }

    function drawCellStructure() {
      push();
      noFill();
      stroke(255, 255, 255, 80);
      strokeWeight(2);
      for (let i = 0; i < 10; i++) {
        rotateY(TWO_PI / 10);
        ellipse(0, 0, 300, 300);
      }
      pop();

      // Membrane sphere
      push();
      fill(0, 200, 100, 150);
      noStroke();
      sphere(120);
      pop();

      // Breaking effect
      push();
      stroke(255, 100, 100, 180);
      strokeWeight(3);
      for (let i = 0; i < 30; i++) {
        let theta = random(TWO_PI);
        let phi = random(PI);
        let r = 120;
        let x = r * sin(phi) * cos(theta);
        let y = r * sin(phi) * sin(theta);
        let z = r * cos(phi);
        line(x, y, z, x + random(-20, 20), y + random(-20, 20), z + random(-20, 20));
      }
      pop();
    }

    function drawEnzymes() {
      for (let i = 0; i < enzymes.length; i++) {
        let e = enzymes[i];
        push();
        let x = cos(e.angle) * e.radius;
        let z = sin(e.angle) * e.radius;
        let y = e.yOffset;
        translate(x, y, z);
        fill(0, 120, 255);
        noStroke();
        sphere(10);
        e.angle += e.speed;

        // Draw interaction line
        stroke(0, 120, 255, 80);
        line(0, 0, 0, -x * 0.3, -y * 0.3, -z * 0.3);
        pop();
      }
    }

    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }
  </script>
</body>
