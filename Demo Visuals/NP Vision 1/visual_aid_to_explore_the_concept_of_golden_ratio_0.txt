The visualization is interactive—users can rotate and zoom the camera. It uses dynamic lighting and techy neon outlines to enhance clarity and engagement. This approach helps learners experience the elegant growth pattern visually and spatially.</description>

  <script src="https://cdn.jsdelivr.net/npm/p5.easycam@1.0.1/p5.easycam.min.js"></script>
  <script>
    let phi = (1 + Math.sqrt(5)) / 2;

    function setup() {
      createCanvas(windowWidth, windowHeight, WEBGL);
      createEasyCam({ distance: 800 });
    }

    function draw() {
      background(10);
      ambientLight(50);
      directionalLight(0, 255, 200, 0.5, 1, -1);

      noFill();
      stroke(0, 255, 200);
      strokeWeight(2);

      drawGoldenSpiral3D();
    }

    function drawGoldenSpiral3D() {
      beginShape();
      let r = 5;
      for (let a = 0; a < 8 * PI; a += 0.05) {
        let radius = r * Math.pow(phi, a / (2 * PI));
        let x = radius * cos(a);
        let y = radius * sin(a);
        let z = a * 3;
        vertex(x, y, z);
      }
      endShape();
    }

    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }
  </script>
