<description>
"Pi visualise" refers to the act of graphically representing the mathematical constant Pi (π), which is fundamentally defined as the ratio of a circle's circumference to its diameter, approximately 3.14159. Given Pi's irrational and transcendental nature, its decimal representation extends infinitely without any repeating pattern. At a college level, visualizing Pi involves transforming this endless, non-repeating sequence of digits into perceivable patterns using various mathematical and computational techniques. One prevalent method is the "Pi walk" or "digit spiral," where each digit from 0 to 9 is mapped to a specific angular rotation or step direction on a Cartesian plane. As successive digits of Pi are processed, a point traces a complex, evolving path, often revealing intricate, emergent structures from the digit sequence. For instance, a digit '3' might correspond to a 108-degree turn, while a '1' implies a 36-degree turn (assuming 360 degrees divided by 10 digits). Another technique involves "Pi tapestries" or "color maps," where each digit is assigned a unique color, and a grid or linear arrangement of cells is filled according to Pi's digit sequence. These visualizations serve not only as aesthetic explorations but also as analytical tools to intuitively grasp the statistical distribution of digits within Pi. They can offer insights into the normal number conjecture, which posits that every finite sequence of digits appears with equal frequency in Pi's expansion. By transforming this abstract, infinite number into dynamic, tangible visual forms, such visualizations foster a deeper, more intuitive appreciation of Pi's profound mathematical properties.

        function startPiWalkAnimation() {
            if (piWalkIndex < PI_DIGITS.length && piWalkPoints.length < piWalkMaxPoints) {
                let digit = parseInt(PI_DIGITS[piWalkIndex]);
                if (!isNaN(digit)) {
                    let stepLength = 10; // Fixed step length, adjusted by canvas size
                    let angleIncrement = radians(digit * 36); // Each digit rotates by 36 degrees
                    piWalkAngle += angleIncrement;

                    let newX = piWalkCurrentX + cos(piWalkAngle) * stepLength;
                    let newY = piWalkCurrentY + sin(piWalkAngle) * stepLength;

                    // Keep points somewhat centered
                    if (piWalkPoints.length > 0) {
                        let avgX = piWalkPoints.reduce((sum, p) => sum + p.x, 0) / piWalkPoints.length;
                        let avgY = piWalkPoints.reduce((sum, p) => sum + p.y, 0) / piWalkPoints.length;
                        let maxDist = Math.min(width, height) * 0.4; // Max distance from center
                        if (dist(newX, newY, width/2, height/2) > maxDist) {
                            // Scale down points if they go too far out
                            let scaleFactor = maxDist / dist(newX, newY, width/2, height/2) * 0.95;
                            piWalkPoints = piWalkPoints.map(p => ({
                                x: (p.x - width/2) * scaleFactor + width/2,
                                y: (p.y - height/2) * scaleFactor + height/2
                            }));
                            piWalkCurrentX = (piWalkCurrentX - width/2) * scaleFactor + width/2;
                            piWalkCurrentY = (piWalkCurrentY - height/2) * scaleFactor + height/2;
                            newX = piWalkCurrentX + cos(piWalkAngle) * stepLength;
                            newY = piWalkCurrentY + sin(piWalkAngle) * stepLength;
                        }
                    }

                    piWalkPoints.push({ x: newX, y: newY });
                    piWalkCurrentX = newX;
                    piWalkCurrentY = newY;
                }
                piWalkIndex++;
            }
        }

            pop();
        }

        function startColorMapAnimation() {
            if (colorMapIndex < PI_DIGITS.length) {
                let digit = parseInt(PI_DIGITS[colorMapIndex]);
                if (!isNaN(digit)) {
                    let col = colorMapIndex % colorMapGridSize;
                    let row = floor(colorMapIndex / colorMapGridSize);
                    if (row < colorMapGridSize) {
                        colorMapGrid[row][col] = digit;
                    }
                }
                colorMapIndex++;
            }
        }

        // --- Main Draw Loop ---
        function draw() {
            drawBackground();

            switch (animationState) {
                case 0: // Initial state / Idle
                    drawConclusion(); // Show conclusion as idle state
                    break;
                case 1: // Pi Definition
                    drawPiDefinition(constrain(frameCount / 90, 0, 1)); // Animation over 3 seconds (90 frames)
                    break;
                case 2: // Digits Flow
                    drawDigitsFlow();
                    break;
                case 3: // Pi Walk
                    drawPiWalk();
                    if (currentHighlight.element === 'pi_walk_drawing') {
                        startPiWalkAnimation();
                    }
                    break;
                case 4: // Color Map
                    drawColorMap();
                    if (currentHighlight.element === 'color_map_drawing') {
                        startColorMapAnimation();
                    }
                    break;
                case 5: // Conclusion / Final State
                    drawConclusion();
                    break;
            }
        }

        // --- Narration Controls ---
        function speakNarration() {
            if (!speechSynth.speaking && narrationIndex < narrationSteps.length - 1) {
                narrationIndex++;
                const step = narrationSteps[narrationIndex];
                currentUtterance = new SpeechSynthesisUtterance(step.text);
                if (voices.length > 0) {
                    currentUtterance.voice = voices[step.voiceIndex];
                }
                currentUtterance.pitch = step.pitch;
                currentUtterance.rate = step.rate;

                currentUtterance.onstart = () => {
                    speaking = true;
                    if (step.onstart) step.onstart();
                };
                currentUtterance.onend = () => {
                    if (step.onend) step.onend();
                    if (narrationIndex < narrationSteps.length - 1) {
                        speakNarration(); // Speak next
                    } else {
                        speaking = false;
                        resetHighlight();
                    }
                };
                currentUtterance.onerror = (event) => {
                    console.error('SpeechSynthesisUtterance.onerror', event);
                    speaking = false;
                    resetHighlight();
                };
                speechSynth.speak(currentUtterance);
            } else if (narrationIndex >= narrationSteps.length - 1) {
                speaking = false; // End of narration
                resetHighlight();
            }
        }

        function stopNarration() {
            if (speechSynth.speaking) {
                speechSynth.cancel();
            }
            speaking = false;
            narrationIndex = -1;
            currentUtterance = null;
            resetHighlight();
        }

            animationState = 1; // Start from the beginning of visuals
            resetVisuals();

        });

                stopNarration();

                resetVisuals();
                narrationIndex = -1; // Reset narration index
                speakNarration(); // Start narration
            }
        });
