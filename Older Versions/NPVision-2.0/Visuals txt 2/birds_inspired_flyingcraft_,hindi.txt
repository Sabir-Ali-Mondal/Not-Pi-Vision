<description>
पक्षियों से प्रेरित उड़ने वाले शिल्प (फ्लाइंगक्राफ्ट) का विज्ञान बायोमिमिक्री का एक उत्कृष्ट उदाहरण है, जहाँ हम प्रकृति की लाखों वर्षों की विकास प्रक्रिया से सीखते हैं। स्नातक स्तर (Graduate level) पर, इस विषय में वायुगतिकी (aerodynamics), पंखों की संरचना (wing morphology), प्रणोदन (propulsion), और नियंत्रण (control) के जटिल सिद्धांत शामिल हैं, और ये कैसे विमानन प्रौद्योगिकी (aviation technology) को आकार देते हैं।

मुख्य रूप से, पक्षियों की उड़ान चार वायुगतिकीय बलों पर निर्भर करती है: उत्थापन (Lift), भार (Weight), प्रणोद (Thrust), और कर्षण (Drag)। उनके पंख 'एयरफ़ॉइल' (airfoil) के रूप में कार्य करते हैं, जो पंख के ऊपर की हवा को नीचे की तुलना में तेज़ी से चलने के लिए मजबूर करते हैं, जिससे बर्नौली के सिद्धांत (Bernoulli's principle) के अनुसार ऊपर कम दबाव और नीचे अधिक दबाव बनता है, जिससे उत्थापन उत्पन्न होता है। इसके साथ ही, पंखों का फ्लैप करना (फड़फड़ाना) हवा को पीछे की ओर धकेलता है, जिससे न्यूटन के तीसरे नियम के अनुसार आगे की ओर प्रणोद मिलता है।

पक्षियों के पंखों की संरचना और आकार उनकी उड़ान शैली के अनुसार अनुकूलित होते हैं। उदाहरण के लिए, ऊँची-पहलू-अनुपात (high-aspect ratio) वाले संकीर्ण पंख (जैसे समुद्री पक्षियों में) ग्लाइडिंग और ऊर्जा-कुशल उड़ान के लिए आदर्श होते हैं, जबकि अंडाकार पंख (जैसे गौरैया में) उच्च-कुशलता और त्वरित गतिशीलता प्रदान करते हैं। उनकी पंखों की हड्डियाँ खोखली और हल्की होती हैं, और पंखों का सूक्ष्म ढाँचा (microstructure) - जिसमें बारब्यूल्स और हुकलेट शामिल हैं - लचीलापन और मजबूती प्रदान करता है, साथ ही उड़ान के दौरान पंखों को हवा-रोधी (airtight) बनाए रखता है।

पक्षियों के पंखों में प्राथमिक पंखों के बीच 'स्लॉटिंग' (slotting) एक उन्नत वायुगतिकीय विशेषता है। यह कम गति पर स्टाल (stall) को रोकने और लिफ्ट को बढ़ाने में मदद करता है, एक सिद्धांत जिसे आधुनिक विमानों में स्लैट्स (slats) और फ्लैप्स (flaps) के रूप में अपनाया गया है। पंख अपनी आकृति (कैम्बर, स्वीप, डायहेड्रल) और आक्रमण के कोण (angle of attack) को बदलकर उड़ान की परिस्थितियों के अनुसार ढल सकते हैं, जिससे उन्हें अपनी लिफ्ट और कर्षण को गतिशील रूप से नियंत्रित करने की क्षमता मिलती है।

पक्षियों से प्रेरित डिज़ाइन में 'ऑर्निथॉप्टर' (ornithopters) शामिल हैं, जो सीधे पंखों को फड़फड़ाने की नकल करते हैं। हालाँकि, पारंपरिक विमानों में, पक्षियों से अप्रत्यक्ष प्रेरणाएँ अधिक आम हैं, जैसे कि विंगलेट्स (winglets) जो विंगटिप भंवरों को कम करते हैं (पक्षियों के फैले हुए पंखों के समान), या पंखों को मोड़ने और आकार बदलने की क्षमता (morphing wings) जो पक्षियों की उड़ान की दक्षता की नकल करती है। भविष्य की एयरोस्पेस इंजीनियरिंग सक्रिय प्रवाह नियंत्रण (active flow control) और बायोमिमेटिक सामग्रियों (biomimetic materials) पर ध्यान केंद्रित कर रही है, जो पक्षियों की पंखों की संरचना और उनकी अनुकूलनशीलता से प्रेरणा लेती है ताकि अगली पीढ़ी के विमानों को और अधिक कुशल और चुस्त बनाया जा सके।

        function draw() {
            clear(); // Clear canvas for redraw - important for glassmorphism to show through
            
            let currentTime = millis();
            let deltaTime = currentTime - lastUpdateTime;
            lastUpdateTime = currentTime;

            let targetDuration = narrationData[narrationIndex]?.duration || slideDuration;
            
            animationProgress += deltaTime / targetDuration;

            if (animationProgress >= 1.0) {
                if (isNarrating && narrationIndex < narrationData.length - 1) {
                    narrationIndex++;
                    speakNarrationSegment();
                } else if (isNarrating && narrationIndex >= narrationData.length - 1) {
                    stopNarration(); // End narration

                currentSlide = (currentSlide + 1); // Increment slide
                if (currentSlide >= narrationData.length) { // Loop back if past last slide, or stop
                    currentSlide = 0; // Reset to first slide for replay
                }
                animationProgress = 0; // Reset progress for the new slide
            }

            // Draw current slide based on currentSlide and animationProgress
            drawSlide(currentSlide, animationProgress);
            
            // Apply highlight if narrating and highlight function exists
            if (isNarrating && narrationData[narrationIndex] && narrationData[narrationIndex].highlight) {
                push();
                translate(width / 2, height / 2); // Ensure highlights are also centered
                narrationData[narrationIndex].highlight(animationProgress);
                pop();
            }
        }

        // Helper to get CSS variable color
        function varToRGB(variable) {
            const style = getComputedStyle(document.documentElement);
            const color = style.getPropertyValue(variable);
            // Handle rgba colors
            const rgbaMatch = color.match(/rgba?\((\d+),\s*(\d+),\s*(\d+)(,\s*([\d.]+))?\)/);
            if (rgbaMatch) {
                return {
                    r: parseInt(rgbaMatch[1]),
                    g: parseInt(rgbaMatch[2]),
                    b: parseInt(rgbaMatch[3]),
                    a: parseFloat(rgbaMatch[5] || 1)
                };
            }
            // Fallback for hex colors or other formats (not strictly needed with provided variables)
            const hexMatch = color.match(/^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i);
            if (hexMatch) {
                return {
                    r: parseInt(hexMatch[1], 16),
                    g: parseInt(hexMatch[2], 16),
                    b: parseInt(hexMatch[3], 16),
                    a: 1
                };
            }
            return { r: 255, g: 255, b: 255, a: 1 }; // Default white if parsing fails
        }

        function drawAirfoil(p) {
            let airfoilX = 0;
            let airfoilY = 0;
            let airfoilLength = width * 0.6;
            let airfoilHeight = height * 0.1;

            // Airfoil shape
            fill(50, 50, 80, 180);
            let primaryColor = varToRGB('--primary-color');
            stroke(primaryColor.r, primaryColor.g, primaryColor.b, 200);
            strokeWeight(3);
            beginShape();
            vertex(-airfoilLength / 2, 0);
            bezierVertex(
                -airfoilLength / 2 + airfoilLength * 0.1, -airfoilHeight,
                airfoilLength / 2 - airfoilLength * 0.1, -airfoilHeight,
                airfoilLength / 2, 0
            );
            bezierVertex(
                airfoilLength / 2 - airfoilLength * 0.1, airfoilHeight * 0.2,
                -airfoilLength / 2 + airfoilLength * 0.1, airfoilHeight * 0.2,
                -airfoilLength / 2, 0
            );
            endShape(CLOSE);

            // Airflow lines
            let glowColor = varToRGB('--glow-color');
            stroke(glowColor.r, glowColor.g, glowColor.b, 200);
            strokeWeight(2);
            for (let i = 0; i < 7; i++) {
                let yOffset = map(i, 0, 6, -airfoilHeight * 2, airfoilHeight * 2);
                let startX = -width / 2;
                let endX = width / 2;
                let curveFactor = map(i, 0, 6, 0.5, 1.5);

                let x1 = lerp(startX, -airfoilLength / 2 - 50, p);
                let x2 = lerp(startX + airfoilLength * 0.2, airfoilLength / 2 + 50, p);
                
                noFill();
                curve(
                    startX - 100, yOffset, // Control 1
                    startX, yOffset, // Point 1
                    x2, yOffset + (i > 3 ? -1 : 1) * airfoilHeight * 0.5 * curveFactor * (i==3 ? 0 : 1) * p * 2, // Point 2, curved over airfoil
                    endX + 100, yOffset // Control 2
                );

                // Arrows for airflow direction
                if (p > 0.5) {
                    let arrowX = lerp(x1, x2, 0.7);
                    let arrowY = yOffset + (i > 3 ? -1 : 1) * airfoilHeight * 0.5 * curveFactor * (i==3 ? 0 : 1);
                    push();
                    translate(arrowX, arrowY);
                    rotate(atan2(0, 1)); // Point right
                    line(0, 0, -10, -5);
                    line(0, 0, -10, 5);
                    pop();
                }
            }

            // Pressure indication
            if (p > 0.7) {
                fill(0, 0, 255, map(p, 0.7, 1.0, 0, 150)); // Blue for high pressure
                noStroke();
                ellipse(-airfoilLength * 0.2, airfoilHeight * 0.5, 20, 10);
                ellipse(airfoilLength * 0.2, airfoilHeight * 0.4, 20, 10);
                textSize(12);
                let textColor = varToRGB('--text-color');
                fill(textColor.r, textColor.g, textColor.b, map(p, 0.7, 1.0, 0, 255));
                text("उच्च दबाव", airfoilLength * 0.1, airfoilHeight * 0.8);

                fill(255, 0, 0, map(p, 0.7, 1.0, 0, 150)); // Red for low pressure
                ellipse(-airfoilLength * 0.2, -airfoilHeight * 0.5, 20, 10);
                ellipse(airfoilLength * 0.2, -airfoilHeight * 0.4, 20, 10);
                text("कम दबाव", airfoilLength * 0.1, -airfoilHeight * 0.8);
            }
        }

        function drawArrow(baseX, baseY, dx, dy, label, textOffset = 0) {
            line(baseX, baseY, baseX + dx, baseY + dy);

                translate(baseX + dx / 2, baseY + dy / 2);

                let textY = textOffset * (dy === 0 ? (dx > 0 ? 1 : -1) : 0);

        function drawMorphingWing(p) {
            let wingX = 0;
            let wingY = 0;

            let wingThickness = height * 0.1;

            // Base wing structure
            fill(80, 80, 100, 180);
            rect(wingX - wingLength / 2, wingY - wingThickness / 2, wingLength, wingThickness, 10);

            // Morphing effect
            let morphFactor = sin(frameCount * 0.05) * 0.5 + 0.5; // Smooth 0 to 1 oscillation
            let morphAmplitude = lerp(0, wingThickness * 0.5, p);

            let primaryColor = varToRGB('--primary-color');
            fill(primaryColor.r, primaryColor.g, primaryColor.b, map(p, 0.2, 0.8, 0, 200, true));
            noStroke();

            // Top morphing line
            beginShape();
            vertex(wingX - wingLength / 2, wingY - wingThickness / 2);
            bezierVertex(
                wingX - wingLength / 4, wingY - wingThickness / 2 - morphAmplitude * morphFactor,
                wingX + wingLength / 4, wingY - wingThickness / 2 - morphAmplitude * (1 - morphFactor),
                wingX + wingLength / 2, wingY - wingThickness / 2
            );
            endShape();

            // Bottom morphing line
            beginShape();
            vertex(wingX - wingLength / 2, wingY + wingThickness / 2);
            bezierVertex(
                wingX - wingLength / 4, wingY + wingThickness / 2 + morphAmplitude * (1 - morphFactor),
                wingX + wingLength / 4, wingY + wingThickness / 2 + morphAmplitude * morphFactor,
                wingX + wingLength / 2, wingY + wingThickness / 2
            );
            endShape();

            // Active flow control points (small pulsating circles)
            if (p > 0.6) {
                let dotSize = lerp(5, 15, abs(sin(frameCount * 0.15)));
                let glowColor = varToRGB('--glow-color');
                fill(glowColor.r, glowColor.g, glowColor.b, map(p, 0.6, 1.0, 0, 255));
                ellipse(wingX - wingLength * 0.2, wingY - wingThickness * 0.2, dotSize, dotSize);
                ellipse(wingX + wingLength * 0.2, wingY + wingThickness * 0.2, dotSize, dotSize);
                textSize(12);
                let textColor = varToRGB('--text-color');
                fill(textColor.r, textColor.g, textColor.b, map(p, 0.8, 1.0, 0, 255));
                text("सक्रिय प्रवाह नियंत्रण", wingX, wingY + wingThickness + 20);
            }
        }

        // --- Narration Functions ---

        function startNarration() {
            if (speechSynth.speaking) {
                speechSynth.cancel();
            }
            isNarrating = true;
            narrationIndex = 0;
            lastUpdateTime = millis(); // Reset time for animation sync
            currentSlide = 0; // Ensure visual also resets to first slide
            animationProgress = 0; // Reset animation progress
            speakNarrationSegment();
        }

        function stopNarration() {
            if (speechSynth.speaking) {
                speechSynth.cancel();
            }
            isNarrating = false;
        }

        function speakNarrationSegment() {
            if (narrationIndex >= narrationData.length) {
                stopNarration();
                return;
            }

            const segment = narrationData[narrationIndex];
            currentUtterance = new SpeechSynthesisUtterance(segment.text);
            
            // Get available voices and pick one, or just use default and vary pitch/rate
            let voices = speechSynth.getVoices();
            // Try to find a Hindi voice, or fallback
            let hindiVoice = voices.find(voice => voice.lang === 'hi-IN' || voice.lang === 'hi_IN');
            if (hindiVoice) {
                currentUtterance.voice = hindiVoice;
            } else {
                // If no specific Hindi voice, use default and adjust pitch/rate for variety
                currentUtterance.pitch = segment.pitch || 1.0;
                currentUtterance.rate = segment.rate || 1.0;
                currentUtterance.lang = 'en-US'; // Fallback lang, might pronounce Hindi poorly
                console.warn("Hindi voice not found. Using default voice for narration.");
            }

            currentUtterance.onend = () => {
                // The draw loop manages slide transitions and narrationIndex increment based on `animationProgress`.
                // This `onend` primarily ensures speech finishes.
            };

            currentUtterance.onerror = (event) => {
                console.error('SpeechSynthesisUtterance.onerror', event);
            };

            speechSynth.speak(currentUtterance);
        }

        function resetAnimation() {
            stopNarration(); // Stop any ongoing narration
            currentSlide = 0;
            animationProgress = 0;
            lastUpdateTime = millis();
            narrationIndex = 0; // Reset narration index too
        }
