<description>
The water cycle, also known as the hydrologic cycle, describes the continuous movement of water on, above, and below the surface of the Earth. It is a fundamental biogeochemical cycle powered primarily by solar energy. The cycle begins with **evaporation**, where solar radiation heats liquid water in oceans, lakes, and rivers, converting it into water vapor that rises into the atmosphere. Simultaneously, **transpiration** contributes water vapor from plants through their leaves. As the warm, moist air rises, it cools, leading to **condensation**, where water vapor transforms back into tiny liquid droplets or ice crystals, forming clouds.

When these cloud droplets or crystals grow sufficiently large, they fall to the Earth's surface as **precipitation** in various forms, including rain, snow, sleet, or hail. Upon reaching the ground, precipitation can take several paths. Some water flows over the land surface as **runoff**, eventually collecting in streams, rivers, and ultimately returning to oceans. A significant portion **infiltrates** the ground, replenishing groundwater aquifers, which can then discharge into surface water bodies or be drawn upon by plants. Some groundwater can also resurface as springs. Finally, **collection** occurs in large bodies of water like oceans, lakes, and reservoirs, from which the cycle can begin anew. This ceaseless process continuously recycles the Earth's water supply, influencing weather patterns, climate, and supporting all life.

            if (!narrationStarted || narrationIndex >= narrationQueue.length) {

                currentHighlight = null;

            msg.voice = currentUtteranceData.voice;
            msg.pitch = currentUtteranceData.pitch;
            msg.rate = currentUtteranceData.rate;

            msg.onend = () => {
                narrationIndex++;
                speakNextNarration();

                console.error('SpeechSynthesis Utterance Error:', e);

                speakNextNarration();

            currentHighlight = currentUtteranceData.highlight;

        function updateAnimation() {
            if (!narrationStarted) { // Only advance animation automatically if not in podcast mode
                animProgress += animSpeed;
                if (animProgress >= 1.0) {
                    animProgress = 0;
                    animPhase = (animPhase + 1) % 8; // Cycle through 0-7 phases
                    if (animPhase === 0) {
                        // If cycle restarts, re-init particles for fresh start
                        initAnimation();
                        animPhase = 0; // Ensure it starts from 0 again
                    }
                }
            } else {
                // When narration is active, animPhase is controlled by speechNextNarration
                // We only advance animProgress within the current animPhase
                animProgress += animSpeed * 0.5; // Slower progress during narration
                if (animProgress >= 1.0) animProgress = 1.0; // Cap at 1
            }

            // Phase-specific logic
            switch (animPhase) {
                case 0: // Idle / Initial state / Return to ocean
                    // Water molecules are at ocean level, cloud coverage is minimal, no rain
                    cloudCoverage = lerp(cloudCoverage, 0, 0.05);
                    // Ensure all molecules are in water
                    for (let mol of waterMolecules) {
                        mol.state = 0; // In water
                        mol.y = oceanLevel + random(5, 15);
                    }
                    raindrops = []; // Clear any residual raindrops
                    break;
                case 1: // Evaporation
                    // Sun comes into play, water molecules rise
                    sunY = lerp(sunY, height * 0.1, 0.02); // Sun higher
                    for (let i = 0; i < MAX_WATER_MOLECULES / 5; i++) { // Generate some rising molecules
                        let mol = waterMolecules[int(random(waterMolecules.length))];
                        if (mol.state === 0) { // If in water
                            mol.x = random(width * 0.1, width * 0.9);
                            mol.y = oceanLevel + random(5, 15);
                            mol.state = 1; // Rising (evaporation)
                        }
                    }
                    break;
                case 2: // Transpiration (some molecules from trees)
                    // Continue evaporation + add from trees
                    for (let i = 0; i < MAX_WATER_MOLECULES / 10; i++) {
                        let mol = waterMolecules[int(random(waterMolecules.length))];
                        if (mol.state === 0) { // If in water (re-use molecule)
                            mol.x = random(width * 0.6, width * 0.8); // Near trees
                            mol.y = height * 0.7 + random(-10, 10);
                            mol.state = 1; // Rising (transpiration)
                        }
                    }
                    break;
                case 3: // Condensation
                    // Molecules move towards clouds and cloud grows
                    cloudCoverage = lerp(cloudCoverage, 1, 0.02); // Cloud grows
                    for (let mol of waterMolecules) {
                        if (mol.state === 1 || mol.state === 0) { // If rising or idle
                            mol.targetX = width * 0.5 + random(-100, 100);
                            mol.targetY = cloudHeight + random(-50, 50);
                            mol.state = 2; // Condensing
                        }
                    }
                    break;
                case 4: // Precipitation
                    // Raindrops fall from clouds
                    cloudCoverage = lerp(cloudCoverage, 0.8, 0.01); // Clouds thin out slightly
                    for (let i = 0; i < MAX_RAINDROPS / 2; i++) { // Generate raindrops
                        if (raindrops.length < MAX_RAINDROPS) {
                            raindrops.push(new Raindrop(
                                random(width * 0.3, width * 0.7),
                                cloudHeight + random(-50, 50)
                            ));
                        }
                    }
                    // Some water molecules in cloud state convert back
                    for (let mol of waterMolecules) {
                        if (mol.state === 2 && random(1) < 0.01) { // Small chance to turn into raindrop
                            raindrops.push(new Raindrop(mol.x, mol.y));
                            mol.state = 0; // Return to water pool conceptually
                            mol.y = oceanLevel;
                        }
                    }
                    break;
                case 5: // Runoff
                    // Raindrops hit land and visually flow (handled by raindrop update)
                    break;
                case 6: // Infiltration
                    // Some raindrops disappear into ground (handled by raindrop update)
                    break;
                case 7: // Collection / Return to ocean
                    // No new evaporation/rain, all water returns to ocean
                    cloudCoverage = lerp(cloudCoverage, 0, 0.02);
                    for (let mol of waterMolecules) {
                        if (mol.state === 1 || mol.state === 2) { // If rising or in cloud
                            mol.targetX = random(width * 0.1, width * 0.9);
                            mol.targetY = oceanLevel + random(5, 15);
                            mol.state = 0; // Return to water
                        }
                    }
                    raindrops = []; // Clear any residual raindrops as they complete their cycle
                    break;
            }
        }

        // --- Particle Classes ---

        class WaterMolecule {
            constructor(x, y, state) {
                this.x = x;
                this.y = y;
                this.z = random(0.5, 1.5); // For depth effect
                this.size = 3 * this.z;
                this.speed = random(0.5, 2) * this.z;
                this.state = state; // 0: in water, 1: rising (vapor), 2: in cloud (condensed)
                this.targetX = this.x;
                this.targetY = this.y;
                this.alpha = 200; // Transparency for vapor
            }

            update() {
                if (this.state === 1) { // Rising (evaporation/transpiration)
                    this.y -= this.speed;
                    this.x += sin(this.y * 0.05 + this.z) * 0.5; // Slight horizontal wiggle
                    if (this.y < cloudHeight + 50) { // Near cloud level, start transition
                        this.state = 2; // Move to condensing state
                        this.targetX = width * 0.5 + random(-100, 100);
                        this.targetY = cloudHeight + random(-50, 50);
                    }
                } else if (this.state === 2) { // Condensing / In cloud
                    this.x = lerp(this.x, this.targetX, 0.05);
                    this.y = lerp(this.y, this.targetY, 0.05);
                    this.alpha = lerp(this.alpha, 255, 0.1); // Become less transparent
                    this.size = lerp(this.size, 5 * this.z, 0.1); // Grow slightly
                } else if (this.state === 0) { // In water, idle or waiting to evaporate
                    // Stay near ocean level, subtle movement
                    this.x += sin(frameCount * 0.02 + this.z) * 0.1;
                    this.y = oceanLevel + map(sin(frameCount * 0.05 + this.z), -1, 1, 5, 15);
                    this.alpha = 150;
                    this.size = 3 * this.z;
                }
            }

            display() {
                noStroke();
                let moleculeColor = color(150, 200, 255, this.alpha);
                if (currentHighlight && currentHighlight.element === 'ocean' && this.state === 0) {
                    moleculeColor = color(primaryColor);
                    moleculeColor.setAlpha(150 + highlightPulse * 100);
                } else if (currentHighlight && currentHighlight.element === 'clouds' && this.state === 2) {
                    moleculeColor = color(primaryColor);
                    moleculeColor.setAlpha(150 + highlightPulse * 100);
                }
                fill(moleculeColor);
                ellipse(this.x, this.y, this.size, this.size);
            }

            isDone() {
                // Molecules never truly 'done' for water cycle, they just change state or get recycled
                return false;
            }
        }

        class Raindrop {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.speed = random(5, 10);
                this.length = random(10, 20);
                this.alpha = random(150, 255);
                this.isGroundwater = false;
                this.splashCount = 0;
            }

            update() {
                if (this.isGroundwater) {
                    this.y += random(0.1, 0.5); // Slow movement underground
                    this.x += sin(frameCount * 0.03 + this.y) * 0.2;
                    return;
                }

                // Check if hit ocean
                if (this.y + this.length > oceanLevel && this.x < width * 0.3) {
                     this.splash();
                     this.y = height + 100; // Mark as "returned to ocean"
                     return;
                }
                
                // Check if hit land
                let landYAtX = calculateLandY(this.x);
                if (this.y + this.length > landYAtX) {
                    if (animPhase === 6 || (narrationStarted && narrationQueue[narationIndex]?.animPhaseTarget === 6)) {
                        if (random(1) < 0.2) { // 20% chance to infiltrate in phase 6
                            this.isGroundwater = true;
                            this.y = landYAtX + random(5, 15); // Start underground
                            this.x += random(-5, 5); // Slight horizontal drift underground
                            return;
                        }
                    }
                    this.splash();
                    this.y = height + 100; // Mark as "runoff" off screen (eventually back to ocean)
                    return;
                }

                this.y += this.speed;
            }

            display() {
                if (this.y > height + 50 && !this.isGroundwater) return; // Don't draw if off-screen/returned

                let dropColor = color(100, 180, 255, this.alpha);
                if (currentHighlight && currentHighlight.element === 'rain') {
                    dropColor = color(primaryColor);
                    dropColor.setAlpha(150 + highlightPulse * 100);
                }
                if (this.isGroundwater) {
                    dropColor = color(0, 180, 255, 50 + highlightPulse * 100); // Groundwater specific color
                    if (currentHighlight && currentHighlight.element === 'groundwater') {
                        dropColor = color(primaryColor);
                        dropColor.setAlpha(100 + highlightPulse * 150);
                    }
                }
                
                stroke(dropColor);
                strokeWeight(1.5);
                if (!this.isGroundwater) {
                    line(this.x, this.y, this.x, this.y + this.length);
                } else {
                    // Represent groundwater as subtle, slower-moving dots
                    noStroke();
                    fill(dropColor);
                    ellipse(this.x, this.y, 4, 4);
                }

                // Draw splash effect if triggered
                if (this.splashCount > 0) {
                    this.splashCount--;
                    noFill();
                    stroke(200, 220, 255, map(this.splashCount, 10, 0, 100, 0));
                    ellipse(this.x, this.y - this.length, 5 + (10 - this.splashCount) * 2, 5 + (10 - this.splashCount) * 2);
                }
            }

            splash() {
                this.splashCount = 10; // Trigger a small splash animation
            }

            isDone() {
                return (this.y > height + 200) || (this.isGroundwater && this.y > height * 0.95 + 20); // Off-screen or deep underground
            }
        }

        function calculateLandY(x) {
            // Re-evaluates the land bezier path to get Y at a given X
            // This is a simplified linear interpolation for the example's complex bezier.
            // For true bezier, a more complex function like de Casteljau's algorithm would be needed.
            // For this visualization, we approximate based on fixed points.
            if (x < mountainShape[1].x) return mountainShape[1].y;
            if (x < mountainShape[4].x) {
                return map(x, mountainShape[1].x, mountainShape[4].x, mountainShape[1].y, mountainShape[4].y);
            }
            if (x < mountainShape[7].x) {
                return map(x, mountainShape[4].x, mountainShape[7].x, mountainShape[4].y, mountainShape[7].y);
            }
            return mountainShape[7].y;
        }

        function windowResized() {
            canvasWidth = windowWidth;
            canvasHeight = windowHeight - 4;
            resizeCanvas(canvasWidth, canvasHeight);
            initAnimation(); // Re-initialize elements positions based on new size
            precalculateMountainShape(); // Re-calculate mountain shape for new size
        }
    </script>
