<description> The "carbon bond pyramid structure" refers to a pyramidal carbocation configuration where a single carbon atom is positioned at the apex of a pyramid formed by a polygonal base (usually four or five sides). This unique arrangement showcases the carbon atom's rare coordination number of five or six, which is uncommon compared to its usual four. The apical carbon atom is typically sp-hybridized and interacts with the molecular orbitals of the base, creating a stable yet unusual geometry. </description>

        function setup() {
            createCanvas(600, 600);
            noLoop();
        }

        function draw() {
            background(255);

            // Draw base polygon (square)
            let baseSize = 200;
            let centerX = width / 2;
            let centerY = height / 2 + baseSize / 4;

            fill(200, 200, 255);
            stroke(50);
            strokeWeight(2);
            beginShape();
            for (let i = 0; i < 4; i++) {
                let angle = PI / 4 + i * HALF_PI;
                let x = centerX + cos(angle) * baseSize / sqrt(2);
                let y = centerY + sin(angle) * baseSize / sqrt(2);
                vertex(x, y);
            }
            endShape(CLOSE);

            // Draw apical carbon atom
            let apexX = centerX;
            let apexY = centerY - baseSize / sqrt(2) - 80;

            fill(255, 100, 100);
            stroke(50);
            ellipse(apexX, apexY, 30);

            // Connect apex to base vertices
            for (let i = 0; i < 4; i++) {
                let angle = PI / 4 + i * HALF_PI;
                let x = centerX + cos(angle) * baseSize / sqrt(2);
                let y = centerY + sin(angle) * baseSize / sqrt(2);

                stroke(100);
                line(apexX, apexY, x, y);
            }

            // Label apex and base atoms
            noStroke();
            fill(0);
            textAlign(CENTER, CENTER);
            
            text("C", apexX, apexY); // Apical carbon
            for (let i = 0; i < 4; i++) {
                let angle = PI / 4 + i * HALF_PI;
                let x = centerX + cos(angle) * baseSize / sqrt(2);
                let y = centerY + sin(angle) * baseSize / sqrt(2);

                text("C", x, y); // Base carbons
            }
        }
