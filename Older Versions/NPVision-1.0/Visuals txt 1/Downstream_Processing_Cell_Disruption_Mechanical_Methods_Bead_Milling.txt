<description>Bead milling is a mechanical method used in downstream processing for disrupting microbial or cell walls to release intracellular products. This technique involves agitating cells with tiny beads (typically made of glass, ceramic, or steel) inside a chamber. As the beads are rapidly stirred, they collide with the cells, breaking them open through high-shear forces and grinding actions. This method is efficient for large-scale applications, providing high cell disruption rates, especially for tough cell walls like those of yeast or bacteria. In a 3D scientific visualization, this can be represented by a rotating chamber filled with moving beads smashing against cells, illustrating the dynamic mechanical disruption process.</description>

  <script>
    let beads = [];
    let cells = [];

    function setup() {
      createCanvas(windowWidth, windowHeight, WEBGL);
      for (let i = 0; i < 80; i++) {
        beads.push(new Particle(random(-200, 200), random(-200, 200), random(-200, 200), 'bead'));
      }
      for (let i = 0; i < 20; i++) {
        cells.push(new Particle(random(-200, 200), random(-200, 200), random(-200, 200), 'cell'));
      }
    }

    function draw() {
      background(240);
      rotateY(frameCount * 0.003);
      rotateX(frameCount * 0.002);

      ambientLight(100);
      directionalLight(255, 255, 255, 0.25, 0.25, -1);

      // Milling chamber
      push();
      noFill();
      stroke(100, 120);
      strokeWeight(1);
      box(450, 300, 300);
      pop();

      for (let b of beads) b.update(), b.display();
      for (let c of cells) c.update(), c.display();
    }

    class Particle {
      constructor(x, y, z, type) {
        this.pos = createVector(x, y, z);
        this.vel = p5.Vector.random3D().mult(random(0.5, 2));
        this.type = type;
        this.radius = this.type === 'bead' ? random(5, 10) : random(12, 18);
        this.color = this.type === 'bead' ? color(60, 60, 150) : color(200, 50, 50, 200);
      }

      update() {
        this.pos.add(this.vel);

        // Bounce inside the box
        if (abs(this.pos.x) > 225) this.vel.x *= -1;
        if (abs(this.pos.y) > 150) this.vel.y *= -1;
        if (abs(this.pos.z) > 150) this.vel.z *= -1;

        // Simulate disruption if a bead hits a cell
        if (this.type === 'bead') {
          for (let c of cells) {
            if (p5.Vector.dist(this.pos, c.pos) < this.radius + c.radius) {
              c.radius *= 0.98;
              c.color = color(220, 100, 100, 180);
            }
          }
        }
      }

      display() {
        push();
        translate(this.pos.x, this.pos.y, this.pos.z);
        ambientMaterial(this.color);
        noStroke();
        sphere(this.radius);
        pop();
      }
    }

    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }
  </script>
</body>
