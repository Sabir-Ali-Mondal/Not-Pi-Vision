<description>PDA (Pushdown Automaton) for the language L = { aⁿ bᵐ cⁿ | m,n > 0 } is a machine that accepts strings containing:

One or more 'a's,

One or more 'b's,

One or more 'c's,
where the number of 'a's and 'c's are equal (n > 0), but the number of 'b's (m) can be any positive number, unrelated to n.

This language is non-regular but context-free because it needs a PDA to match the counts of 'a's and 'c's while allowing any count of 'b's in between.

How the PDA works step by step:
1️⃣ It starts by reading a series of 'a's (at least one). Each time it reads an 'a', it pushes a symbol (like 'A') onto the stack to remember it.
2️⃣ After finishing reading the 'a's, it moves to read 'b's. It ignores 'b's—meaning no push or pop is done for them. This part is unconstrained in count, just requiring at least one 'b'.
3️⃣ Then, when 'c's appear, for each 'c' read, it pops one 'A' from the stack, matching each 'c' with a previously seen 'a'.
4️⃣ If the input is fully read and the stack is empty (except for the start marker), the string is accepted.

For example:

✅ Input: aa bb cc → matches (n=2, m=2)

✅ Input: a bbb c → matches (n=1, m=3)

❌ Input: aa b c → rejected (n=2 a’s but only 1 c)

The PDA ensures a balanced "sandwich" of equal 'a's and 'c's with any length 'b' middle.

Key PDA states:

q0: Start state (reads 'a's and pushes)

q1: Reading 'b's (no stack change)

q2: Reading 'c's (pops from stack)

q_accept: Accepts if the stack is balanced

This PDA’s main focus is on using the stack to balance the counts of 'a's and 'c's while ignoring 'b's.

Such PDA models are crucial to understand context-free languages where stack memory plays a key role, beyond what finite automata can handle.

<script>
let states = ['q0', 'q1', 'q2', 'q_accept'];
let currentState = 0;
let input = ['a', 'a', 'b', 'b', 'c', 'c'];
let stack = [];
let stackSymbol = 'A';
let tapeIndex = 0;
let radius = 50;
let replaying = false;

function setup() {
  createCanvas(windowWidth, windowHeight);
  frameRate(1);
}

function draw() {
  background('#f4f7fb');
  drawStates();
  drawStack();
  drawTape();
  
  if (tapeIndex < input.length) {
    let symbol = input[tapeIndex];
    processSymbol(symbol);
    tapeIndex++;
  } else if (currentState !== 3) {
    if (stack.length === 0) {
      currentState = 3;
    }
  }
}

function drawStates() {
  let cx = width / 2;
  let cy = height / 3;
  for (let i = 0; i < states.length; i++) {
    stroke('#07f');
    strokeWeight(3);
    fill(i === currentState ? '#07f' : '#fff');
    ellipse(cx + (i - 1.5) * 160, cy, radius * 2);
    fill(i === currentState ? '#fff' : '#07f');
    noStroke();
    textAlign(CENTER, CENTER);
    textSize(16);
    text(states[i], cx + (i - 1.5) * 160, cy);
  }
}

function drawStack() {
  let sx = width - 100;
  let sy = height / 2 + 100;
  fill('#07f');
  textSize(16);
  textAlign(CENTER);
  text('Stack', sx, sy - stack.length * 20 - 20);
  for (let i = 0; i < stack.length; i++) {
    fill('#fff');
    stroke('#07f');
    rect(sx - 20, sy - (i + 1) * 20, 40, 20);
    fill('#07f');
    noStroke();
    text(stack[i], sx, sy - (i + 1) * 10 - 10);
  }
}

function drawTape() {
  let tx = 100;
  let ty = height - 100;
  textSize(18);
  textAlign(CENTER);
  fill('#07f');
  text('Input Tape', tx + input.length * 30 / 2, ty - 50);
  for (let i = 0; i < input.length; i++) {
    fill(i === tapeIndex ? '#07f' : '#fff');
    stroke('#07f');
    rect(tx + i * 30, ty, 30, 30);
    fill(i === tapeIndex ? '#fff' : '#07f');
    noStroke();
    text(input[i], tx + i * 30 + 15, ty + 20);
  }
}

function processSymbol(sym) {
  if (currentState === 0) {
    if (sym === 'a') {
      stack.push(stackSymbol);
    } else if (sym === 'b') {
      currentState = 1;
    }
  }
  if (currentState === 1) {
    if (sym === 'b') {
    } else if (sym === 'c') {
      currentState = 2;
      stack.pop();
    }
  } else if (currentState === 2) {
    if (sym === 'c') {
      stack.pop();
    }
  }
}

function resetPDA() {
  currentState = 0;
  stack = [];
  tapeIndex = 0;
}
