<description>A Möbius strip is a non-orientable surface with only one side and one edge, formed by taking a rectangular strip, giving it a half-twist, and joining the ends together. It challenges conventional notions of geometry and topology, making it a fascinating object in both mathematics and art. The following HTML file visualizes a Möbius strip interactively in 3D using p5.js with WEBGL mode, styled in a geometric and visually refined manner.</description>

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>Mobius Strip Visualization</title>
  <style>
    html, body {
      margin: 0;
      padding: 0;
      overflow: hidden;
      background: #0a0a0a;
    }
    canvas {
      display: block;
    }
  </style>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/p5.js/1.7.0/p5.min.js"></script>
</head>
<body>
  <script>
    let angle = 0;

    function setup() {
      createCanvas(windowWidth, windowHeight, WEBGL);
    }

    function draw() {
      background(10);
      orbitControl();
      rotateY(angle * 0.2);
      rotateX(angle * 0.1);

      let radius = 200;
      let stripWidth = 40;
      let detail = 200;

      stroke(255);
      noFill();
      strokeWeight(1.5);
      lights();

      beginShape(TRIANGLE_STRIP);
      for (let i = 0; i <= detail; i++) {
        let t = map(i, 0, detail, 0, TWO_PI);
        for (let j = 0; j <= 1; j++) {
          let s = map(j, 0, 1, -1, 1) * stripWidth;
          let phi = t / 2;

          let x = (radius + s * cos(phi)) * cos(t);
          let y = (radius + s * cos(phi)) * sin(t);
          let z = s * sin(phi);

          let c = map(i, 0, detail, 100, 255);
          fill(c, c * 0.8, 255 - c, 180);
          vertex(x, y, z);
        }
      }
      endShape();
      
      angle += 0.003;
    }

    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }
  </script>
</body>
</html>
```
