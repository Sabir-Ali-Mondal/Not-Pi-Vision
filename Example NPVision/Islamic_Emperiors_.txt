    let empires = [
      { name: "Umayyad", year: "661–750", x: 100, y: 300, color: "#ba8c63" },
      { name: "Abbasid", year: "750–1258", x: 300, y: 200, color: "#a55c43" },
      { name: "Ottoman", year: "1299–1924", x: 500, y: 250, color: "#7c4c3a" },
      { name: "Safavid", year: "1501–1736", x: 700, y: 200, color: "#9e7463" },
      { name: "Mughal", year: "1526–1857", x: 900, y: 300, color: "#b88c64" }
    ];

    function setup() {
      createCanvas(windowWidth, windowHeight);
      textFont('Georgia');
    }

    function draw() {
      background('#f9f5ee');

      stroke(80, 60, 50);
      strokeWeight(2);
      noFill();
      bezier(100, 300, 200, 100, 800, 100, 900, 300);

      for (let i = 0; i < empires.length; i++) {
        let e = empires[i];
        fill(e.color);
        noStroke();
        ellipse(e.x, e.y, 80, 80);
        fill(0);
        noStroke();
        textAlign(CENTER, CENTER);
        textSize(14);
        text(e.name, e.x, e.y - 10);
        textSize(12);
        text(e.year, e.x, e.y + 10);
      }
    }

    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }
  </script>
