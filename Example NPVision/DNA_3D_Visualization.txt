**What is DNA?**

DNA (Deoxyribonucleic Acid) is the molecule that carries the genetic instructions used in the growth, development, functioning, and reproduction of all known living organisms and many viruses. Structurally, DNA is shaped like a double helix, resembling a twisted ladder. Its sequence of four chemical bases—adenine (A), thymine (T), cytosine (C), and guanine (G)—encodes the information needed to build and maintain an organism.

    let angle = 0;
    let helixLength = 300;
    let numPairs = 100;
    let spacing = helixLength / numPairs;
    let bases = ['A', 'T', 'C', 'G'];
    let basePairs = [];

    function setup() {
      createCanvas(windowWidth, windowHeight, WEBGL);
      generateBasePairs();
    }

    function generateBasePairs() {
      for (let i = 0; i < numPairs; i++) {
        let base1 = random(bases);
        let base2;
        if (base1 === 'A') base2 = 'T';
        else if (base1 === 'T') base2 = 'A';
        else if (base1 === 'C') base2 = 'G';
        else base2 = 'C';

        basePairs.push([base1, base2]);
      }
    }

    function draw() {
      background(11, 12, 16);
      rotateY(mouseX * 0.003);
      rotateX(mouseY * 0.003);
      translate(0, -helixLength / 2, 0);

      for (let i = 0; i < numPairs; i++) {
        let y = i * spacing;
        let x1 = 60 * cos(angle + i * 0.3);
        let z1 = 60 * sin(angle + i * 0.3);
        let x2 = -x1;
        let z2 = -z1;

        stroke(102, 255, 255);
        strokeWeight(2);
        line(x1, y, z1, x2, y, z2); // base pair line

        push();
        translate(x1, y, z1);
        fill(0, 255, 255);
        noStroke();
        sphere(5);
        pop();

        push();
        translate(x2, y, z2);
        fill(255, 100, 255);
        noStroke();
        sphere(5);
        pop();

        // backbone spirals
        stroke(102, 255, 102);
        if (i > 0) {
          let prevY = (i - 1) * spacing;
          let prevX1 = 60 * cos(angle + (i - 1) * 0.3);
          let prevZ1 = 60 * sin(angle + (i - 1) * 0.3);
          let prevX2 = -prevX1;
          let prevZ2 = -prevZ1;
          line(prevX1, prevY, prevZ1, x1, y, z1);
          line(prevX2, prevY, prevZ2, x2, y, z2);
        }
      }

      angle += 0.01;
    }

    function windowResized() {
      resizeCanvas(windowWidth, windowHeight);
    }

  </script>
