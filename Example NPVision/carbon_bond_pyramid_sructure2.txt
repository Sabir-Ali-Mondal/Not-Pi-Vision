<description>The "carbon bond pyramid structure" typically refers to the tetrahedral geometry formed by a carbon atom bonded to four other atoms (e.g., hydrogen), as in methane (CH₄). In this structure, the carbon sits at the center of a pyramid-like shape with four bonds pointing to the corners of a tetrahedron. The bond angles are approximately 109.5°, creating a 3D molecular shape crucial in organic chemistry. The visualization below renders this structure in an interactive 3D space using p5.js with a "Scientific" style and a complexity of 9, highlighting each atom and bond dynamically and intuitively.</description>

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Carbon Bond Pyramid Structure</title>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/p5.js/1.4.2/p5.min.js"></script>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/p5.js/1.4.2/addons/p5.dom.min.js"></script>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/p5.js/1.4.2/addons/p5.sound.min.js"></script>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/p5.js/1.4.2/p5.min.js"></script>
  <style>
    html, body {
      margin: 0;
      padding: 0;
      overflow: hidden;
      background: #0d1117;
      color: white;
      font-family: 'Helvetica Neue', sans-serif;
    }
    canvas {
      display: block;
    }
  </style>
</head>
<body>
<script>
let angleX = 0;
let angleY = 0;
let dragging = false;
let lastX, lastY;

function setup() {
  createCanvas(windowWidth, windowHeight, WEBGL);
}

function draw() {
  background(13, 17, 23);
  orbitControl();
  rotateX(angleX);
  rotateY(angleY);

  let carbonRadius = 20;
  let hydrogenRadius = 12;
  let bondLength = 120;

  // Tetrahedral vectors
  let vectors = [
    createVector(1, 1, 1),
    createVector(-1, -1, 1),
    createVector(-1, 1, -1),
    createVector(1, -1, -1)
  ];

  // Normalize and scale vectors
  for (let i = 0; i < vectors.length; i++) {
    vectors[i].normalize().mult(bondLength);
  }

  ambientLight(100);
  directionalLight(255, 255, 255, 0.5, 1, -0.5);

  // Draw Carbon atom
  push();
  ambientMaterial(0, 150, 255); // Carbon: Blue
  sphere(carbonRadius);
  pop();

  for (let i = 0; i < vectors.length; i++) {
    let v = vectors[i];

    // Draw bond
    push();
    let bondVec = p5.Vector.sub(v, createVector(0, 0, 0));
    let bondMag = bondVec.mag();
    translate(v.x / 2, v.y / 2, v.z / 2);
    rotateZ(atan2(v.y, v.x));
    rotateY(acos(v.z / bondMag));
    ambientMaterial(200);
    cylinder(4, bondMag);
    pop();

    // Draw Hydrogen atom
    push();
    translate(v.x, v.y, v.z);
    ambientMaterial(255, 255, 255); // Hydrogen: White
    sphere(hydrogenRadius);
    pop();
  }
}

function mousePressed() {
  dragging = true;
  lastX = mouseX;
  lastY = mouseY;
}

function mouseReleased() {
  dragging = false;
}

function mouseDragged() {
  if (dragging) {
    let dx = mouseX - lastX;
    let dy = mouseY - lastY;
    angleY += dx * 0.01;
    angleX += dy * 0.01;
    lastX = mouseX;
    lastY = mouseY;
  }
}
</script>
</body>
</html>
```
